   {\small \flushleft \shadebox{$\upDelta; \upGamma \vdash \sigma: \elimTy{A}{E}$}%
   \hfill \textbfit{$\sigma$ has eliminator type $\elimTy{A}{E}$ in $\upGamma$, where $A$ and $C$ are well-kinded in $\upDelta$}}
   \begin{smathpar}
      \inferrule*
      {
         \upDelta; \cxtExtend{\upGamma}{\bind{x}{A}} \vdash P: E
      }
      {
         \upDelta; \upGamma \vdash \elimVar{x}{P}: \elimTy{A}{E}
      }
      %
      \and
      %
      \inferrule*
      {
         \upDelta; \upGamma \vdash P: E
      }
      {
         \upDelta; \upGamma
         \vdash
         \elimRecord{\seqEmpty}{P}:
         \elimTy{\tyRecord{R}}{E}
      }
      %
      \and
      %
      \inferrule*
      {
         \upDelta; \upGamma
         \vdash
         \elimRecord{\seq{\ell}}{P}:
         \elimTy{\tyRecord{R}}{\elimTy{B}{E}}
      }
      {
         \upDelta; \upGamma
         \vdash
         \elimRecord{\seq{\ell}, \ell'}{P}:
         \elimTy{\tyRecord{\rowExtend{R}{\bind{\ell'}{B}}}}{E}
      }
      %
      \and
      %
      \inferrule*
      {
         \upDelta; \upGamma \vdash P_i: \elimTy{A_i}{E}
         \quad
         (\forall i \leq \length{\seq{P}})
         \\
         \upDelta; \upGamma \vdash P': \elimTy{\tyVariant{R}}{E}
      }
      {
         \upDelta; \upGamma
         \vdash
         \elimVariant{\ell}{P}{P'}: \elimTy{\tyVariant{\rowExtend{R}{\seq{\bind{\ell}{A}}}}}{E}
      }
      %
      \and
      %
      \inferrule*
      {
         \upDelta; \upGamma \vdash P_i: \elimTy{A_i}{\tyFun{(\tyFun{B_i}{C})}{C}}
         \quad
         (\forall i \in \length{\seq{P}})
         \\
         \upDelta; \upGamma \vdash P': \elimTy{A'}{C}
      }
      {
         \upDelta; \upGamma
         \vdash
         \elimComp{\ell}{P; \elimBind{\kw{return}}{P'}}: \elimTy{\tyComp{A'}{\seq{\bind{\ell}{\tyFun{A}{B}}}}}{C}
      }
   \end{smathpar}

   {\small \flushleft \shadebox{$\upDelta; \upGamma \vdash V: A$}%
   \hfill \textbfit{$V$ has type $A$ in $\upGamma$, where $A$ is well-kinded in $\upDelta$}}
   \begin{smathpar}
      \inferrule*
      {
         \strut
      }
      {
         \upDelta; \upGamma \vdash \exInt{n}: \tyInt
      }
      %
      \and
      %
      \inferrule*
      {
         \bind{x}{A} \in \upGamma
      }
      {
         \upDelta; \upGamma \vdash \exVar{x}: A
      }
      %
      \and
      %
      \inferrule*
      {
         \upDelta; \upGamma \vdash V: A
      }
      {
         \upDelta; \upGamma \vdash \exVariant{\ell}{V}: \tyVariant{\rowExtend{R}{\bind{\ell}{A}}}
      }
      %
      \and
      %
      \inferrule*
      {
         \strut
      }
      {
         \upDelta; \upGamma \vdash \exRecordEmpty: \tyRecord{}
      }
      %
      \and
      %
      \inferrule*
      {
         \upDelta; \upGamma \vdash U: \tyRecord{R}
         \\
         \upDelta; \upGamma \vdash V_i: A_i
         \quad
         (\forall i \in \length{\seq{A}})
      }
      {
         \upDelta; \upGamma \vdash \exRecordExtend{U}{\seq{\bind{\ell}{V}}}: \tyRecord{\rowExtend{R}{\seq{\bind{\ell}{A}}}}
      }
      %
      \and
      %
      \inferrule*
      {
         \upDelta; \upGamma \vdash \sigma: \elimTy{A}{C}
      }
      {
         \upDelta; \upGamma \vdash \exFun{\sigma}: \tyFun{A}{C}
      }
      %
      \and
      %
      \inferrule*
      {
         \cxtExtend{\upDelta}{\bind{\alpha}{K}}; \upGamma \vdash M: C
         \\
         \alpha \notin \FV(\upGamma)
      }
      {
         \upDelta; \upGamma \vdash \exPolyFun{\alpha}{M}: \tyForall{\alpha}{K}{C}
      }
   \end{smathpar}
   \\[2mm]
   {\small \flushleft \shadebox{$\upDelta; \upGamma \vdash M: C$}%
   \hfill \textbfit{$M$ has computation type $C$ in $\upGamma$, where $C$ is well-kinded in $\upDelta$}}
   \begin{smathpar}
      \inferrule*
      {
         \upDelta; \upGamma \vdash V: A
      }
      {
         \upDelta; \upGamma \vdash \exReturn{V}: \tyComp{A}{R}
      }
      %
      \and
      %
      \inferrule*
      {
         \upDelta; \upGamma \vdash V: \tyFun{A}{C}
         \\
         \upDelta; \upGamma \vdash U: A
      }
      {
         \upDelta; \upGamma \vdash \exApp{V}{U}: C
      }
      %
      \and
      %
      \inferrule*
      {
         \upDelta; \upGamma \vdash V: A
         \\
         \upDelta; \upGamma \vdash \sigma: \elimTy{A}{C}
      }
      {
         \upDelta; \upGamma \vdash \exMatch{V}{\sigma}: C
      }
      %
      \and
      %
      \inferrule*
      {
         \upDelta; \upGamma \vdash V: \tyForall{\alpha}{K}{C}
         \\
         \upDelta \vdash T: K
      }
      {
         \upDelta; \upGamma \vdash \exPolyApp{V}{T}: \subst{C}{T}{\alpha}
      }
      %
      \and
      %
      \inferrule*
      {
         \upDelta; \upGamma \vdash V: A
      }
      {
         \upDelta; \upGamma \vdash \exOp{\ell}{V}: \tyComp{B}{\rowExtend{R}{\bind{\ell}{\tyFun{A}{B}}}}
      }
      %
      \and
      %
      \inferrule*
      {
         \upDelta; \upGamma \vdash M: C
         \\
         \upDelta; \upGamma \vdash V: \tyHandler{C}{D}
      }
      {
         \upDelta; \upGamma \vdash \exHandle{M}{V}: D
      }
   \end{smathpar}
