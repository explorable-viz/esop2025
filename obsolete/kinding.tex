   {
      \bind{\alpha}{K} \in \upDelta
   }

      \upDelta \vdash \tyVar{\alpha}: K

   \and

   \inferrule*

      \upDelta \vdash \tyInt: \kiType

   \inferrule*
   {
      \upDelta \vdash A: \kiType
      \\
      \upDelta \vdash C: \kiComp
   }
   {
      \upDelta \vdash \tyFun{A}{C}: \kiType
   }
   %
   \and
   %
   \inferrule*
   {
      \upDelta \vdash C: \kiComp
      \\
      \upDelta \vdash D: \kiComp

   {
      \upDelta \vdash \tyHandler{C}{D}: \kiType
   }
   %
   \and
   %
   \inferrule*
   {
      \upDelta \vdash C: \kiComp
      \\
      \upDelta \vdash D: \kiComp
   }
   {
      \upDelta \vdash \tyHandler{C}{D}: \kiType
   }
   %
   \and
   %
   \inferrule*
   {
      \upDelta \vdash A_i: \kiType

   {
      \upDelta \vdash \seq{\bind{\ell}{A}}: \kiRow
   }
   %
   \and
   %
   \inferrule*
   {
      \upDelta \vdash R: \kiRow

      \upDelta \vdash \tyRecord{R}: \kiType

   \and
   %
   \inferrule*
   {
      \upDelta \vdash R: \kiRow
   }

   \inferrule*
   {
      \upDelta \vdash R: \kiRow
      \\
      \upDelta \vdash A: \kiType
   }
   {
      \upDelta \vdash \tyComp{A}{R}: \kiComp
   }
   %
   \and
   %
   \inferrule*
   {
      \cxtExtend{\upDelta}{\bind{\alpha}{K}} \vdash C: \kiComp

      \upDelta \vdash \tyForall{\alpha}{K}{C}: \kiType
   }

   {
      \upDelta \vdash A: \kiType
      \\
      \upDelta \vdash E: \kiCont

      \upDelta \vdash \elimTy{A}{E}: \kiCont
   }
