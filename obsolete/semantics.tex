      }

   {\small \flushleft \shadebox{$M \reduce N$}%
   \hfill \textbfit{$M$ reduces to $N$}}
   \begin{salign}
      \exApp{(\exFun{\upsigma})}{V}
      &\reduce
      M
      \quad
      \textit{if }V, \upsigma \match M
      \\
      \exPolyApp{(\exPolyFun{\alpha}{M})}{T}
      &\reduce
      \subst{M}{T}{\alpha}
      \\
      \exMatch{V}{\upsigma}
      &\reduce
      M
      \quad
      \textit{if }V, \upsigma \match M
      \\
      \exHandle{\exReturn{V}}{\exFun{\upsigma}}
      &\reduce
      M
      \quad
      \textit{if }\exReturn{V}, \upsigma \match M
      \\
      \begin{split}
      \exHandle{\plug{\mathcal{E}}{\exOp{\ell}{V}}}{\exFun{\upsigma}}
      &\reduce
      \exApp{M}{(\exFun{\elimVar{y}{\exHandle{\plug{\mathcal{E}}{\exReturn{y}}}{\exFun{\upsigma}}}})}
      \\
      &\quad\;\;\,
      \textit{if }\exOp{\ell}{V}, \upsigma \match M
      \textit{ and }
      \ell \notin \BL(\mathcal{E})
      \end{split}
      \\
      M &\reduce N
      \quad\hfill
      \textit{if }\plug{\mathcal{E}}{M} \reduce \plug{\mathcal{E}}{N}
   \end{salign}
