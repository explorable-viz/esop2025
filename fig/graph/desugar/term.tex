   {\flushleft \shadebox{$s \desugar s', e, E$}\hfill}
   \begin{smathpar}
      \inferrule*[lab={\ruleName{$\desugar$-edge}}]
      {
         \raw{s} \desugar \raw{s}', \raw{e}, E
         \\
         \alpha, \beta\textit{ fresh}
      }
      {
         \annot{\raw{s}}{\addrNull}
         \desugar
         \annot{\raw{s}'}{\alpha}, \annot{\raw{e}}{\beta}, E \cup \set{(\alpha, \beta)}
      }
   \end{smathpar}
   {\flushleft \shadebox{$\raw{s} \desugar \raw{s}', \raw{e}, E$}\hfill}
   \begin{smathpar}
      \inferrule*[lab={\ruleName{$\desugar$-binary-apply}}]
      {
         s_1 \desugar s_1', e_1, E_1
         \\
         s_2 \desugar s_2', e_2, E_2
      }
      {
         \exBinaryApp{s_1}{\oplus}{s_2}
         \desugar
         \exBinaryApp{s_1'}{\oplus}{s_2'},
         \exForeignApp{\oplus}{e, e'},
         E_1 \cup E_2
      }
      %
      \and
      %
      \inferrule*[lab={\ruleName{$\desugar$-let-rec}}]
      {
         g_i = \bind{x_i}{\mu_i}
         \\
         g'_i = \bind{x_i}{\mu_i'}
         \\
         \mu_i \desugar \mu_i', \sigma_i, E_i
         \quad
         (\forall i \numleq \length{\seq{g}})
         \\
         s \desugar s', e, E'
      }
      {
         \exLetRecPiecewise{\seq{g}}{s}
         \desugar
         \exLetRecPiecewise{\seq{g}'}{s'},
         \exLetRecPiecewise{\seq{\bind{x}{\sigma}}}{e},
         \bigcup{\seq{E}} \cup E'
      }
      %
      \and
      %
      \inferrule*[lab={\ruleName{$\desugar$-match}}]
      {
         s \desugar s', e, E
         \\
         \mu \desugar \mu', \sigma, E'
      }
      {
         \exMatch{s}{\mu}
         \desugar
         \exMatch{s'}{\mu'},
         \exApp{\exFun{\sigma}}{e},
         E \cup E'
      }
      %
      \and
      %
      \inferrule*[lab={\ruleName{$\desugar$-non-empty-list}}]
      {
         s \desugar s', e, E
         \\
         l \desugar l', e', E'
      }
      {
         \exList{s}{l}
         \desugar
         \exList{s'}{l'},
         \exConstr{\cCons}{e, e'},
         E \cup E'
      }
      %
      \and
      %
      \inferrule*[lab={\ruleName{$\desugar$-if}}]
      {
         s \desugar s', e, E
         \\
         s_1 \desugar s_1', e_1, E_1
         \\
         s_2 \desugar s_2', e_2, E_2
         \\
         \sigma = \elimConstr{\elimBind{\cTrue}{e_1}, \elimBind{\cFalse}{e_2}}
      }
      {
         \exIfThenElse{s}{s_1}{s_2}
         \desugar
         \exIfThenElse{s'}{s_1'}{s_2'},
         \exApp{\exFun{\sigma}}{e},
         E \cup E_1 \cup E_2
      }
      %
      \and
      %
      \inferrule*[lab={\ruleName{$\desugar$-list-enum}}]
      {
         s_1 \desugar s_1', e, E
         \\
         s_2 \desugar s_2', e', E'
      }
      {
         \exListEnum{s_1}{s_2}
         \desugar
         \exListEnum{s_1'}{s_2'},
         \exApp{\exApp{\varEnumFromTo}{e}}{e'},
         E \cup E'
      }
      %
      \and
      %
      \inferrule*[lab={\ruleName{$\desugar$-list-comp}}]
      {
         \seq{q}, s \desugar \seq{q}', s', e
      }
      {
         \exListComp{s}{\seq{q}}
         \desugar
         \exListComp{s'}{\seq{q}'}, e
      }
      %
      \and
      %
      \inferrule*[lab={\ruleName{$\desugar$-do}}]
      {
         \seq{d} \desugar \annot{\seq{\raw{d}}}{\alpha}, e, E
      }
      {
         \exDo{\seq{d}}
         \desugar
         \annot{(\exDo{\annot{\seq{\raw{d}}}{\addrNull}})}{\alpha}, e, E
      }
   \end{smathpar}

      }
