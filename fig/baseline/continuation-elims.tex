   {\small \flushleft \shadebox{$\upGamma \vdash \sigma: \elimTy{A}{K}$}%
   \hfill \textbfit{$\sigma$ has type $\elimTy{A}{K}$ under $\upGamma$}}
   \begin{smathpar}
   \inferrule*
   {
      \upGamma \cons \bind{x}{A} \vdash \kappa: K
   }
   {
      \upGamma \vdash (\elimVar{x}{\kappa}): \elimTy{A}{K}
   }
   %
   \and
   %
   \inferrule*
   [
      right={$\datatype{\upSigma}{c} = \datatype{\upSigma}{c} = D$}
   ]
   {
      \upGamma \vdash \kappa: K
      \\
      \upGamma \vdash \kappa': K
   }
   {
      \upGamma \vdash \elimPair{\seq{\elimBind{c}{\upkappa}}}{\seq{\elimBind{c'}{\upkappa'}}}: \elimTy{D}{K}
   }
   %
   \and
   %
   \inferrule*
   {
      \upGamma \vdash \kappa: K
   }
   {
      \upGamma
      \vdash
      \elimRecEmpty{\kappa}: \elimTy{\tyRecEmpty}{K}
   }
   %
   \and
   %
   \inferrule*
   {
      \upGamma \vdash \elimRec{\seq{x}}{\sigma}: \elimTy{\tyRec{\seq{\bind{x}{A}}}}{\elimTy{B}{K}}
   }
   {
      \upGamma
      \vdash
      \elimRec{\seq{x} \cons y}{\sigma}: \elimTy{\tyRec{\seq{\bind{x}{A}} \cons \bind{y}{B}}}{K}
   }
   \end{smathpar}
