\begin{figure}
   {\flushleft \shadebox{$\kappa \disjjoin \kappa' = \kappa^{\dag}$}
   \begin{smathpar}
      \inferrule*[
         lab={\ruleName{$\disjjoin$-lambda}}
      ]
      {
         \sigma \disjjoin \sigma' = \tau
      }
      {
         (\exApp{\lambda}{\sigma}) \disjjoin (\exApp{\lambda}{\sigma'}) = \exApp{\lambda}{\tau}
      }
      %
      \and
      %
      \inferrule*[
         lab={\ruleName{$\disjjoin${-var}}}
      ]
      {
         \kappa_1 \disjjoin \kappa_2 = \kappa
      }
      {
         (\elimBind{\exVar{x}}{\kappa_1}) \disjjoin (\elimBind{\exVar{x}}{\kappa_2}) = \elimBind{\exVar{x}}{\kappa}
      }
      %
      \and
      %
      \inferrule*[
         lab={\ruleName{$\disjjoin$-record}}
      ]
      {
         \kappa_1 \disjjoin \kappa_2 = \kappa
      }
      {
         (\elimRecord{\seq{x}}{\kappa_1}) \disjjoin (\elimRecord{\seq{x}}{\kappa_2}) = \elimRecord{\seq{x}}{\kappa}
      }
      %
      \and
      %
      \inferrule*[
         lab={\ruleName{$\disjjoin$-overlap}}
      ]
      {
         \sigma \disjjoin \sigma' = (\elimBind{c}{\kappa_2})\cons \tau
         \\
         \kappa_1 \disjjoin \kappa_2 = \kappa
      }
      {
         ((\elimBind{c}{\kappa_1}) \cons \sigma) \disjjoin \sigma' = (\elimBind{c}{\kappa})\cons \tau
      }
      %
      \and
      %
      \inferrule*[
         lab={\ruleName{$\disjjoin$-no-overlap}}
      ]
      {
         \seq{c} \cap \seq{c}' = \varnothing
      }
      {
         \elimConstr{\seq{\elimBind{c}{\sigma}}} \disjjoin \elimConstr{\seq{\elimBind{c'}{\sigma'}}} = \elimConstr{\seq{\elimBind{c}{\sigma}}} \cup \elimConstr{\seq{\elimBind{c'}{\sigma'}}}
      }
   \end{smathpar}}
   \caption{Disjoint join of (partial) continuations}
\end{figure}
