         lab={\ruleName{$\orElse$-done}}

         (\seqEmpty, s), s' \orElse \clause{\seqEmpty}{s}
      }
      %
      \and

         lab={\ruleName{$\orElse$-var}}
      ]

         k, s \orElse \seqRange{k_1}{k_j}
      }
      {
         \pattVar{x} \unzipWith{\cons} k, s
         \orElse
         \seqRange{\pattVar{x} \unzipWith{\cons} k_1}
                  {\pattVar{x} \unzipWith{\cons} k_j}
      }
      %
      \and
      %
      \inferrule*[
         lab={\ruleName{$\orElse$-record}}
      ]
      {
         \seq{p} \unzipWith{\concat} k, s
         \orElse
         \seqRange{\seq{p_1} \unzipWith{\concat} k_1}
                  {\seq{p_j} \unzipWith{\concat} k_j}
      }
      {
         \pattRecord{\seq{\bind{x}{p}}} \unzipWith{\cons} k, s
         \orElse
         \seqRange{\pattRecord{\seq{\bind{x}{p_1}}} \unzipWith{\cons} k_1}
                  {\pattRecord{\seq{\bind{x}{p_j}}} \unzipWith{\cons} k_j}
      }
      %
      \and
      %
      \inferrule*[
         lab={\ruleName{$\orElse$-constr}},
         right={$\datatype{c} = D$}
      ]
      {
         \pi \unzipWith{\concat} k, s
         \orElse
         \seqRange{\pi_1 \unzipWith{\concat} k_1}
                  {\pi_j \unzipWith{\concat} k_j}
         \\
         \set{\seq{c}} \eqdef \ctrsF(D) \setminus \set{c}
         \\
         k_i =
         (\clause{\pattConstr{c_i}{\pattVar{\varAnon} \mid n \numleq n_i} \cons
         (\pattVar{\varAnon} \mid n \numleq \length{\pi'})}{s})
         \quad
         (\forall i \numleq \length{\seq{c}}.\arity{c_i} = n_i)
      }
      {
         \pattConstr{c}{\pi} \unzipWith{\cons} k, s
         \orElse
         \seqRange{\pattConstr{c}{\pi_1} \unzipWith{\cons} k_1}
                  {\pattConstr{c}{\pi_j} \unzipWith{\cons} k_j}
         \concat
         \seq{k}
      }
   \end{smathpar}}
