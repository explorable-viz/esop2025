   {\small \flushleft \shadebox{$\Delta; \Gamma \vdash \sigma: \elimTy{A}{E}$}%
   \hfill \textbfit{$\sigma$ has eliminator type $\elimTy{A}{E}$ in $\Gamma$, where $A$ and $C$ are well-kinded in $\Delta$}}
   \begin{smathpar}
      \inferrule*
      {
         \Delta; \cxtExtend{\Gamma}{\bind{x}{A}} \vdash P: E
      }
      {
         \Delta; \Gamma \vdash \elimVar{x}{P}: \elimTy{A}{E}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \Gamma \vdash P: E
      }
      {
         \Delta; \Gamma
         \vdash
         \elimRecord{P}{\seqEmpty}:
         \elimTy{\tyRecord{R}}{E}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \Gamma
         \vdash
         \elimRecord{P}{\seq{\ell}}:
         \elimTy{\tyRecord{R}}{\elimTy{B}{E}}
      }
      {
         \Delta; \Gamma
         \vdash
         \elimRecord{P}{\seq{\ell}, \ell'}:
         \elimTy{\tyRecord{\rowExtend{R}{\bind{\ell'}{B}}}}{E}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \Gamma \vdash P_i: \elimTy{A_i}{E}
         \quad
         (\forall i \leq \length{\seq{P}})
         \\
         \Delta; \Gamma \vdash P': \elimTy{\tyVariant{R}}{E}
      }
      {
         \Delta; \Gamma
         \vdash
         \elimVariant{\ell}{P}{P'}: \elimTy{\tyVariant{\rowExtend{R}{\seq{\bind{\ell}{A}}}}}{E}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \Gamma \vdash P_i: \elimTy{A_i}{\tyFun{(\tyFun{B_i}{C})}{C}}
         \quad
         (\forall i \in \length{\seq{P}})
         \\
         \Delta; \Gamma \vdash P': \elimTy{A'}{C}
      }
      {
         \Delta; \Gamma
         \vdash
         \elimComp{\ell}{P}{P'}: \elimTy{\tyEffect{A'}{\seq{\bind{\ell}{\tyFun{A}{B}}}}}{C}
      }
   \end{smathpar}

   {\small \flushleft \shadebox{$\Delta; \Gamma \vdash V: A$}%
   \hfill \textbfit{$V$ has type $A$ in $\Gamma$, where $A$ is well-kinded in $\Delta$}}
   \begin{smathpar}
      \inferrule*
      {
         \strut
      }
      {
         \Delta; \Gamma \vdash \exInt{n}: \tyInt
      }
      %
      \and
      %
      \inferrule*
      {
         \bind{x}{A} \in \Gamma
      }
      {
         \Delta; \Gamma \vdash \exVar{x}: A
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \Gamma \vdash V: A
      }
      {
         \Delta; \Gamma \vdash \exVariant{\ell}{V}: \tyVariant{\rowExtend{R}{\bind{\ell}{A}}}
      }
      %
      \and
      %
      \inferrule*
      {
         \strut
      }
      {
         \Delta; \Gamma \vdash \exRecordEmpty: \tyRecord{}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \Gamma \vdash U: \tyRecord{R}
         \\
         \Delta; \Gamma \vdash V_i: A_i
         \quad
         (\forall i \in \length{\seq{A}})
      }
      {
         \Delta; \Gamma \vdash \exRecordExtend{U}{\seq{\bind{\ell}{V}}}: \tyRecord{\rowExtend{R}{\seq{\bind{\ell}{A}}}}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \Gamma \vdash \sigma: \elimTy{A}{C}
      }
      {
         \Delta; \Gamma \vdash \exFun{\sigma}: \tyFun{A}{C}
      }
      %
      \and
      %
      \inferrule*
      {
         \cxtExtend{\Delta}{\bind{\alpha}{K}}; \Gamma \vdash M: C
         \\
         \alpha \notin \FV(\Gamma)
      }
      {
         \Delta; \Gamma \vdash \exPolyFun{\alpha}{M}: \tyForall{\alpha}{K}{C}
      }
   \end{smathpar}

   {\small \flushleft \shadebox{$\Delta; \Gamma \vdash M: C$}%
   \hfill \textbfit{$M$ has computation type $C$ in $\Gamma$, where $C$ is well-kinded in $\Delta$}}
   \begin{smathpar}
      \inferrule*
      {
         \Delta; \Gamma \vdash V: A
      }
      {
         \Delta; \Gamma \vdash \exReturn{V}: \tyEffect{A}{R}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \Gamma \vdash V: \tyFun{A}{C}
         \\
         \Delta; \Gamma \vdash U: A
      }
      {
         \Delta; \Gamma \vdash \exApp{V}{U}: C
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \Gamma \vdash V: A
         \\
         \Delta; \Gamma \vdash \sigma: \elimTy{A}{C}
      }
      {
         \Delta; \Gamma \vdash \exMatch{V}{\sigma}: C
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \Gamma \vdash V: \tyForall{\alpha}{K}{C}
         \\
         \Delta \vdash T: K
      }
      {
         \Delta; \Gamma \vdash \exPolyApp{V}{T}: \subst{C}{T}{\alpha}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \Gamma \vdash V: A
      }
      {
         \Delta; \Gamma \vdash \exOp{\ell}{V}: \tyEffect{B}{\rowExtend{R}{\bind{\ell}{\tyFun{A}{B}}}}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \Gamma \vdash M: C
         \\
         \Delta; \Gamma \vdash V: \tyHandler{C}{D}
      }
      {
         \Delta; \Gamma \vdash \exHandle{M}{V}: D
      }
   \end{smathpar}
