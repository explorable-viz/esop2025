   {\small \flushleft \shadebox{$\Delta; \Gamma \vdash V: A$}%
   \hfill \textbfit{$V$ has well-kinded value type $A$ in $\Gamma$, where $A$ is well-kinded in $\Delta$}}
   \begin{smathpar}
      \inferrule*
      {
         \strut
      }
      {
         \Delta; \Gamma \vdash \exInt{n}: \tyInt
      }
      %
      \and
      %
      \inferrule*
      {
         \bind{x}{A} \in \Gamma
      }
      {
         \Delta; \Gamma \vdash \exVar{x}: A
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \Gamma \vdash V: A
      }
      {
         \Delta; \Gamma \vdash \exVariant{\ell}{V}: \tyVariant{\rowExtend{R}{\ell}{A}}
      }
      %
      \and
      %
      \inferrule*
      {
         \strut
      }
      {
         \Delta; \Gamma \vdash \exRecord{}: \tyRecord{}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \cxtExtend{\Gamma}{x}{A} \vdash M: C
      }
      {
         \Delta; \Gamma \vdash \exLambda{x}{M}: \tyFun{A}{C}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \cxtExtend{\Gamma}{x}{A} \vdash M: D
      }
      {
         \Delta; \Gamma \vdash \exHandler{\hReturn{x}{M}}: \tyHandler{A}{D}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta; \cxtExtend{\cxtExtend{\Gamma}{x}{A}}{k}{\tyFun{B}{D}} \vdash M: D
         \\
         \Delta; \Gamma \vdash H: \tyHandler{\tyEffect{R}{A}}{D}
      }
      {
         \Delta; \Gamma
         \vdash
         (\hEffect{\ell}{x}{k}{M}) \cons H:
         \tyHandler{\tyEffect{\rowExtend{R}{\ell}{\tyFun{A}{B}}}{A}}{D}
      }
   \end{smathpar}
   \\[2mm]
   {\small \flushleft \shadebox{$\Delta, \Gamma \vdash M: C$}%
   \hfill \textbfit{$M$ has computation type $C$ in $\Gamma$, where $C$ is well-kinded in $\Delta$}}
   \begin{smathpar}
      \inferrule*
      {
         \Delta, \Gamma \vdash V: A
      }
      {
         \Delta, \Gamma \vdash \exReturn{V}: \tyEffect{E}{A}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta, \Gamma \vdash U: \tyFun{A}{C}
         \\
         \Delta, \Gamma \vdash V: A
      }
      {
         \Delta, \Gamma \vdash \exApp{U}{V}: C
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta, \Gamma \vdash V: A
      }
      {
         \Delta, \Gamma \vdash \exOp{\ell}{V}: \tyEffect{\rowExtend{R}{\ell}{\tyFun{A}{B}}}{B}
      }
      %
      \and
      %
      \inferrule*
      {
         \Delta, \Gamma \vdash M: C
         \\
         \Delta, \Gamma \vdash V: \tyHandler{C}{D}
      }
      {
         \Delta, \Gamma \vdash \exHandle{M}{V}: D
      }
   \end{smathpar}

\end{figure}
