   \begin{salign}
      %========================================
      \intertext{
      \flushleft\shadebox{$\orElseF :: \Cont\;\Gamma\;A \times \Expr\;\cxtEmpty\;A \partialGCto \Cont\;\Gamma\;A$}
      }
      %========================================
      \orElseFwdF
      &::
      (1 + \exists B, \Gamma.\PattSeq\;B\;\Gamma) \to \Cont\;\Gamma\;A \times \Expr\;\cxtEmpty\;A \to \Cont\;\Gamma\;A
      \\
      %========================================
      \orElseFwdF(\seqEmpty)(e, e')
      &=
      e
      \\
      \orElseFwdF(\pattVar{x} \cons \seq{\pi})(\elimVar{x}{\kappa}, e)
      &=
      \elimVar{x}{\orElseFwdF(\seq{\pi})(\kappa, e)}
      \\
      \orElseFwdF(\pattRecord{} \cons \seq{\pi})(\elimRecord{\seqEmpty}{\kappa}, e)
      &=
      \elimRecord{\seqEmpty}{\orElseFwdF(\seq{\pi})(\kappa, e)}
      \\
      \orElseFwdF(\pattRecord{\seq{\bind{x}{p}} \cons \bind{y}{p'}} \cons \seq{\pi})(\elimRecord{\seq{x} \cons y}{\sigma}, e)
      &=
      \elimRecord{\seq{x} \cons y}{\orElseFwdF(\seq{p} \concat (p' \cons \seq{\pi}))(\sigma, e)}
      \\
      \begin{split}
      \orElseFwdF(\pattConstr{c}{\seq{p}} \cons \seq{\pi})(\elimConstr{\elimBind{c}{\kappa}}, e)
      &=
      \elimConstr{\elimBind{c}{\orElseFwdF(\seq{p} \concat \seq{\pi})(\kappa, e)}} \disjjoin \sigma
      \\
      &
      \textit{where }
      \sigma = \elimConstr{\elimBind{c'}{\opName{elimArgs}(\seq{B})} \mid \bind{c'}{\seq{B}} \in \datatype{\Delta}{c} \setminus \set{c}}
      \\
      &
      \phantom{\textit{where }}
      \opName{elimArgs}(\seqEmpty) = e
      \\
      &
      \phantom{\textit{where }}
      \opName{elimArgs}(B' \cons \seq{B}) = \elimVar{x}{\opName{elimArgs}(\seq{B})}
      \end{split}
      \\[2mm]
      %========================================
      \orElseBwdF
      &::
      (1 + \exists B, \Gamma.\PattSeq\;B\;\Gamma) \to \Raw{\Expr}\;\cxtEmpty\;A \to \Cont\;\Gamma\;A \times \Expr\;\cxtEmpty\;A
      \\
      %========================================
      \orElseBwdF(\seqEmpty)(\raw{e}, e)
      &=
      (e, \bot_{\raw{e}})
      \\
      \orElseBwdF(\pattVar{x} \cons \seq{\pi})(\raw{e}, \elimVar{x}{\kappa})
      &=
      (\elimVar{x}{\kappa'}, e)
      \textit{ where }
      (\kappa', e) = \orElseBwdF(\seq{\pi})(\raw{e}, \kappa)
   \end{salign}
   \caption{Desugaring: $\orElseF$ auxiliary function}
