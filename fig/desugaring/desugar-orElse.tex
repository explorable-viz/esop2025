      %========================================
      \intertext{
      \flushleft\shadebox{$\orElseF :: \Cont\;\Gamma\;A \times \Expr\;\cxtEmpty\;A \partialGCto \Cont\;\Gamma\;A$}
      }
      %========================================
      \orElseFwdF
      &::
      (1 + \exists B, \Gamma.\PattSeq\;B\;\Gamma) \to \Cont\;\Gamma\;A \times \Expr\;\cxtEmpty\;A \to \Cont\;\Gamma\;A
      \\
      %========================================
      \orElseFwdF(\seqEmpty)(e, e')
      &=
      e
      \\
      \orElseFwdF(\pattVar{x} \cons \seq{\pi})(\elimVar{x}{\kappa}, e)
      &=
      \elimVar{x}{\orElseFwdF(\seq{\pi})(\kappa, e)}
      \\
      \orElseFwdF(\elimRecord{\seqEmpty}{\kappa}, e)
      &=
      (\pattRecord{} \cons \seq{\pi}, \elimRecord{\seqEmpty}{\kappa'})
      \textit{ where }
      (\seq{\pi}, \kappa') = \orElseFwdF(\kappa, e)
      \\
      \orElseFwdF(\elimRecord{\seq{x} \cons y}{\sigma}, e)
      &=
      (\pattRecord{\seq{\bind{x}{p}} \cons \bind{y}{p'}} \cons \seq{\pi}, \elimRecord{\seq{x} \cons y}{\sigma'})
      \textit{ where }
      (\seq{p} \cons p' \cons \seq{\pi}, \sigma') = \orElseFwdF(\sigma, e)
      \\
      \begin{split}
      \orElseFwdF(\elimConstr{\elimBind{c}{\kappa}}, e)
      &=
      \elimConstr{\elimBind{c}{\orElseFwdF(\kappa, e)}}
      \disjjoin
      \elimConstr{\elimBind{c'}{\opName{elimArgs}(n)} \mid \bind{c'}{n} \in \datatype{\Delta}{c} \setminus \set{c}}
      \\
      &
      \textit{where }
      \opName{elimArgs}(0) = e
      \\
      &
      \phantom{\textit{where }}
      \opName{elimArgs}(n + 1) = \elimVar{x}{\opName{elimArgs}(n)}
      \end{split}
      \\[2mm]
      %========================================
      \orElseBwdF
      &::
      (1 + \exists B, \Gamma'.\PattSeq\;B\;\Gamma') \to \Raw{\Expr}\;\cxtEmpty\;A \to \Cont\;\Gamma\;A \times \Expr\;\cxtEmpty\;A
      \\
      %========================================
   \end{salign}
