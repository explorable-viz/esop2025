      %========================================
      \intertext{
      \flushleft\shadebox{$\orElseF :: \Cont\;\Gamma\;A \times \Expr\;\cxtEmpty\;A \partialGCto \Cont\;\Gamma\;A$}

      %========================================
      \orElseFwdF
      &::
      \Cont\;\Gamma\;A \times \Expr\;\cxtEmpty\;A \to (1 + \exists B, \Gamma'.\PattSeq\;B\;\Gamma') \times \Cont\;\Gamma\;A
      \\
      %========================================
      \orElseFwdF(e, e')
      &=
      (\seqEmpty, e)
      \\
      \orElseFwdF(\elimVar{x}{\kappa}, e)
      &=
      (\pattVar{x} \cons \seq{\pi}, \elimVar{x}{\kappa'})
      \textit{ where }
      (\seq{\pi}, \kappa') = \orElseFwdF(\kappa, e)
      \\
      \orElseFwdF(\elimRecord{\seqEmpty}{\kappa}, e)
      &=
      (\pattRecord{} \cons \seq{\pi}, \elimRecord{\seq{x}}{\kappa'})
      \textit{ where }
      (\seq{\pi}, \kappa') = \orElseFwdF(\kappa, e)
      \\
      \orElseFwdF(\elimRecord{\seq{x} \cons y}{\sigma}, e)
      &=
      (\pattRecord{\seq{\bind{x}{p}} \cons \bind{y}{p'}} \cons \seq{\pi}, \elimRecord{\seq{x}}{\sigma'})
      \textit{ where }
      (p' \cons \seq{\pi}, \sigma') = \orElseFwdF(\sigma, e)
      \\
      \begin{split}
      \orElseFwdF(\elimConstr{\elimBind{c}{\kappa}}, e)
      &=
      \elimConstr{\elimBind{c}{\orElseFwdF(\kappa, e)}}
      \disjjoin
      \elimConstr{\elimBind{c'}{\opName{elimArgs}(n)} \mid \bind{c'}{n} \in \datatype{\Delta}{c} \setminus \set{c}}
      \\
      &
      \textit{where }
      \opName{elimArgs}(0) = e
      \\
      &
      \phantom{\textit{where }}
      \opName{elimArgs}(n + 1) = \elimVar{x}{\opName{elimArgs}(n)}
      \end{split}
   \end{salign}
