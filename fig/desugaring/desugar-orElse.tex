   \begin{salign}
      %========================================
      \PattSeqOpt &:: \Set{Set}
      \\
      \PattSeqOpt &\eqdef 1 + \exists A, \Gamma.\PattSeq\;A\;\Gamma
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\orElseF :: \Cont\;\Gamma\;A \times \Expr\;\cxtEmpty\;A \partialGCto \Cont\;\Gamma\;A$}
      }
      %========================================
      \orElseFwdF
      &::
      \PattSeqOpt \to \Cont\;\Gamma\;A \times \Expr\;\cxtEmpty\;A \to \Cont\;\Gamma\;A
      \\
      %========================================
      \orElseFwdF(\seqEmpty)(e, e')
      &=
      e
      \\
      \orElseFwdF(\pattVar{x} \cons \seq{\pi})(\elimVar{x}{\kappa}, e)
      &=
      \elimVar{x}{\orElseFwdF(\seq{\pi})(\kappa, e)}
      \\
      \orElseFwdF(\pattRecord{} \cons \seq{\pi})(\elimRecord{\seqEmpty}{\kappa}, e)
      &=
      \elimRecord{\seqEmpty}{\orElseFwdF(\seq{\pi})(\kappa, e)}
      \\
      \orElseFwdF(\pattRecord{\seq{\bind{x}{p}} \cons \bind{y}{p'}} \cons \seq{\pi})(\elimRecord{\seq{x} \cons y}{\sigma}, e)
      &=
      \elimRecord{\seq{x} \cons y}{\orElseFwdF(\seq{p} \concat (p' \cons \seq{\pi}))(\sigma, e)}
      \\
      \orElseFwdF(\pattConstr{c}{\seq{p}} \cons \seq{\pi})(\elimConstr{\elimBind{c}{\kappa}}, e)
      &=
      \elimConstr{\elimBind{c}{\orElseFwdF(\seq{p} \concat \seq{\pi})(\kappa, e)}} \disjjoin \unlessFwdF(c)(e)
      \\
      \orElseFwdF(\pattList{p}{o} \cons \seq{\pi})(\elimConstr{\elimBind{\cCons}{\sigma}}, e)
      &=
      \elimConstr{\elimBind{\cCons}{\orElseFwdF(p \cons o \cons \seq{\pi})(\sigma)}} \disjjoin \unlessFwdF(\cCons)(e)
      \\
      \orElseFwdF(\pattListNext{p}{o} \cons \seq{\pi})(\elimConstr{\elimBind{\cCons}{\sigma}}, e)
      &=
      \elimConstr{\elimBind{\cCons}{\orElseFwdF(p \cons o \cons \seq{\pi})(\sigma)}} \disjjoin \unlessFwdF(\cCons)(e)
      \\
      \orElseFwdF(\pattListEnd \cons \seq{\pi})(\elimConstr{\elimBind{\cNil}{\kappa}}, e)
      &=
      \elimConstr{\elimBind{\cNil}{\orElseFwdF(\seq{\pi})(\sigma)}} \disjjoin \unlessFwdF(\cNil)(e)
      \\[2mm]
      %========================================
      \orElseBwdF
      &::
      \PattSeqOpt \to \Raw{\Expr}\;\cxtEmpty\;A \times \Cont\;\Gamma\;A \to \Cont\;\Gamma\;A \times \Expr\;\cxtEmpty\;A
      \\
      %========================================
      \orElseBwdF(\seqEmpty)(\raw{e}, e)
      &=
      (e, \bot_{\raw{e}})
      \\
      \orElseBwdF(\pattVar{x} \cons \seq{\pi})(\raw{e}, \elimVar{x}{\kappa})
      &=
      (\elimVar{x}{\kappa'}, e)
      \textit{ where }
      (\kappa', e) = \orElseBwdF(\seq{\pi})(\raw{e}, \kappa)
      \\
      \orElseBwdF(\pattRecord{} \cons \seq{\pi})(\raw{e}, \elimRecord{\seqEmpty}{\kappa})
      &=
      (\elimRecord{\seqEmpty}{\kappa'}, e)
      \textit{ where }
      (\kappa', e) = \orElseBwdF(\seq{\pi})(\raw{e}, \kappa)
      \\
      \orElseBwdF(\pattRecord{\seq{\bind{x}{p}} \cons \bind{y}{p'}} \cons \seq{\pi})(\raw{e}, \elimRecord{\seq{x} \cons y}{\sigma})
      &= (\elimRecord{\seq{x} \cons y}{\sigma'}, e)
      \textit{ where }
      (\sigma', e) = \orElseBwdF(\seq{p} \concat (p' \cons \seq{\pi}))(\raw{e}, \kappa)
      \\
      \orElseBwdF(\pattConstr{c}{\seq{p}} \cons \seq{\pi})(\raw{e}, \elimConstr{\elimBind{c}{\kappa}} \disjjoin \sigma)
      &=
      (\elimConstr{\elimBind{c}{\kappa'}}, e \join e')
      \textit{ where }
      (\kappa', e) = \orElseBwdF(\seq{p} \concat \seq{\pi})(\raw{e}, \kappa)
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\unlessF :: \Ctr \to (\Expr\;\seqEmpty\;B \partialGCto \Elim\;\seqEmpty\;A\;B)$}
      }
      %========================================
      \unlessFwdF
      &:: \Ctr \to \Expr\;\seqEmpty\;B \to \Elim\;\seqEmpty\;\seq{A}\;B
      \\
      %========================================
      \begin{split}
         \unlessFwdF(c)(e)
         &=
         \elimConstr{\elimBind{c'}{\opName{elimArgs}(\seq{B})} \mid \bind{c'}{\seq{B}} \in \datatype{\Delta}{c} \setminus \set{c}}
         \\
         &
         \textit{where }
         \opName{elimArgs}(\seqEmpty) = e
         \\
         &
         \phantom{\textit{where }}
         \opName{elimArgs}(B' \cons \seq{B}) = \elimVar{x}{\opName{elimArgs}(\seq{B})}
      \end{split}
      \\[2mm]
      %========================================
      \unlessBwdF
      &:: \Ctr \to \Raw{\Expr}\;\seqEmpty\;B \times \Elim\;\seqEmpty\;A\;B \to \Expr\;\seqEmpty\;B
      \\
      \unlessBwdF(c)(\raw{e}, \elimConstr{\elimBind{c}{\kappa}})
      &=
      \bot_{\raw{e}}
      %========================================
   \end{salign}
