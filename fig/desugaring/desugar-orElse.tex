      %========================================
      \intertext{
      \flushleft\shadebox{$\orElseF :: \PattSeqOpt\;\Gamma' \to (\Cont\;(\Gamma \concat \Gamma')\;A \times \Expr\;\cxtEmpty\;A \partialGCto \Cont\;\Gamma\;A)$}
      }
      %========================================
      \begin{split}
      \orElseFwdF
      &::
      \PattSeqOpt\;\Gamma' \to\\
      &\phantom{aj.}
      \Cont\;(\Gamma \concat \Gamma')\;A \times \Expr\;\cxtEmpty\;A \to \Cont\;(\Gamma \concat \Gamma')\;A
      \end{split}
      \\
      %========================================
      \orElseFwdF(\seqEmpty)(e, e')
      &=
      e
      \\
      \orElseFwdF(\pattVar{x} \cons \seq{\pi})(\elimVar{x}{\kappa}, e)
      &=
      \elimVar{x}{\orElseFwdF(\seq{\pi})(\kappa, e)}
      \\
      \orElseFwdF(\pattRecord{\seq{\bind{x}{p}}} \cons \seq{\pi})(\elimRecord{\seq{x}}{\kappa}, e)
      &=
      \elimRecord{\seq{x}}{\orElseFwdF(\seq{p} \concat \seq{\pi})(\kappa, e)}
      \\
      \orElseFwdF(\pattConstr{c}{\seq{p}} \cons \seq{\pi})(\elimConstr{\elimBind{c}{\kappa}}, e)
      &=
      \elimConstr{\elimBind{c}{\orElseFwdF(\seq{p} \concat \seq{\pi})(\kappa, e)}} \disjjoin \unlessFwdF(c)(e)
      \\
      \orElseFwdF(\pattList{p}{o} \cons \seq{\pi})(\elimConstr{\elimBind{\cCons}{\sigma}}, e)
      &=
      \elimConstr{\elimBind{\cCons}{\orElseFwdF(p \cons o \cons \seq{\pi})(\sigma)}} \disjjoin \unlessFwdF(\cCons)(e)
      \\
      \orElseFwdF(\pattListNext{p}{o} \cons \seq{\pi})(\elimConstr{\elimBind{\cCons}{\sigma}}, e)
      &=
      \elimConstr{\elimBind{\cCons}{\orElseFwdF(p \cons o \cons \seq{\pi})(\sigma)}} \disjjoin \unlessFwdF(\cCons)(e)
      \\
      \orElseFwdF(\pattListEnd \cons \seq{\pi})(\elimConstr{\elimBind{\cNil}{\kappa}}, e)
      &=
      \elimConstr{\elimBind{\cNil}{\orElseFwdF(\seq{\pi})(\sigma)}} \disjjoin \unlessFwdF(\cNil)(e)
      \\[2mm]
      %========================================
      \begin{split}
      \orElseBwdF
      &::
      \PattSeqOpt\;\Gamma' \to\\
      &\phantom{aj.}
      \Raw{\Expr}\;\cxtEmpty\;A \times \Cont\;(\Gamma \concat \Gamma')\;A \to \Cont\;(\Gamma \concat \Gamma')\;A \times \Expr\;\cxtEmpty\;A
      \end{split}
      \\
      %========================================
      \orElseBwdF(\seqEmpty)(\raw{e}, e)
      &=
      (e, \bot_{\raw{e}})
      \\
      \orElseBwdF(\pattVar{x} \cons \seq{\pi})(\raw{e}, \elimVar{x}{\kappa})
      &=
      (\elimVar{x}{\kappa'}, e)
      \textit{ where }
      (\kappa', e) = \orElseBwdF(\seq{\pi})(\raw{e}, \kappa)
      \\
      \orElseBwdF(\pattRecord{\seq{\bind{x}{p}}} \cons \seq{\pi})(\raw{e}, \elimRecord{\seq{x}}{\kappa})
      &= (\elimRecord{\seq{x}}{\kappa'}, e)
      \textit{ where }
      (\kappa', e) = \orElseBwdF(\seq{p} \concat \seq{\pi})(\raw{e}, \kappa)
      \\
      \orElseBwdF(\pattConstr{c}{\seq{p}} \cons \seq{\pi})(\raw{e}, \elimConstr{\elimBind{c}{\kappa}} \disjjoin \sigma)
      &=
      (\elimConstr{\elimBind{c}{\kappa'}}, e \join e')
      \textit{ where }
      (\kappa', e) = \orElseBwdF(\seq{p} \concat \seq{\pi})(\raw{e}, \kappa)
      \\
      \orElseBwdF(\pattList{p}{o} \cons \seq{\pi})(\raw{e}, \elimConstr{\elimBind{\cCons}{\sigma}} \disjjoin \sigma')
      &=
      (\elimConstr{\elimBind{\cCons}{\sigma^\dagger}}, e \join e')
      \textit{ where }
      (\sigma^\dagger, e) = \orElseFwdF(p \cons o \cons \seq{\pi})(\raw{e}, \sigma)
      \\
      \orElseBwdF(\pattListNext{p}{o} \cons \seq{\pi})(\raw{e}, \elimConstr{\elimBind{\cCons}{\sigma}} \disjjoin \sigma')
      &=
      (\elimConstr{\elimBind{\cCons}{\sigma^\dagger}}, e \join e')
      \textit{ where }
      (\sigma^\dagger, e) = \orElseBwdF(p \cons o \cons \seq{\pi})(\raw{e}, \sigma)
      \\
      \orElseBwdF(\pattListEnd \cons \seq{\pi})(\raw{e}, \elimConstr{\elimBind{\cNil}{\kappa}} \disjoin \sigma)
      &=
      \elimConstr{\elimBind{\cNil}{\sigma'}}
      \textit{ where }
      (\sigma', e) = \orElseBwdF(\seq{\pi})(\raw{e}, \sigma)
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\unlessF :: \Pi(c: \Ctr).\Expr\;\seqEmpty\;A \partialGCto \Elim\;\Gamma\;d\;A
      \quad\textit{ where }d = \datatype{\Delta}{c}$}
      }
      %========================================
      \unlessFwdF
      &:: \Pi(c: \Ctr).\Expr\;\seqEmpty\;A \to \Elim\;\Gamma\;d\;A
      \\
      %========================================
      \unlessFwdF(c)(e)
      &=
      \elimConstr{\elimBind{c'}{\opName{elimArgs}(\seq{B})} \mid \bind{c'}{\seq{B}} \in d \setminus \set{c}}
      \\[2mm]
      %========================================
      \unlessBwdF
      &:: \Pi(c: \Ctr).\Raw{\Expr}\;\seqEmpty\;A \times \Elim\;\Gamma\;d\;A \to \Expr\;\seqEmpty\;A
      \\
      \unlessBwdF(c)(\raw{e}, \elimConstr{\elimBind{c'}{\kappa'}} \disjjoin \elimConstr{\seq{\elimBind{c}{\kappa}}})
      &=
      \bot_{\raw{e}} \join \seqRangeOp{\elimArgsBwdF(\kappa_1)}{\elimArgsBwdF(\kappa_n)}{\join}
      %========================================
      \intertext{
      \flushleft\shadebox{$\elimArgsF :: \Type^* \to (\Expr\;\seqEmpty\;A \partialGCto \Cont\;\Gamma\;A)$}
      }
      %========================================
      \elimArgsFwdF
      &:: \Type^* \to \Expr\;\seqEmpty\;A \to \Cont\;\Gamma\;A
      \\
      \elimArgsFwdF(\seqEmpty)(e) &= e
      \\
      \elimArgsFwdF(A' \cons \seq{A})(e) &= \elimVar{x}{\opName{elimArgs}(\seq{A}, e)}
      \\[2mm]
      %========================================
      \elimArgsBwdF
      &:: \Type^* \to \Cont\;\Gamma\;A \to \Expr\;\seqEmpty\;A
      \\
      %========================================
      \elimArgsBwdF(\seqEmpty)(e) &= e
      \\
      \elimArgsBwdF(A' \cons \seq{A})(\elimVar{x}{\kappa}) &= \elimArgsBwdF(\seq{A})(\kappa)
   %========================================
