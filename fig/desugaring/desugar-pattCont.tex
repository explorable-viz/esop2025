      %========================================
      \PattSeqOpt &:: \Cxt \to \Set{Set}
      \\
      \PattSeqOpt\;\Gamma &\eqdef 1 + \exists A.\PattSeq\;A\;\Gamma
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\pattContF :: \Patt\;A\;\Gamma' \to (\Cont\;(\Gamma \cons \Gamma')\;B \partialGCto \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \pattContFwdF
      &::
      \Patt\;A\;\Gamma' \to \Cont\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \pattContFwdF(\pattVar{x})(\kappa)
      &=
      \elimVar{x}{\kappa}
      \\
      \pattContFwdF(\pattConstr{c}{\seq{p}})(\kappa)
      &=
      \elimConstr{\elimBind{c}{\pattArgsFwdF(\seq{p})(\kappa)}}
      \\
      \pattContFwdF(\pattList{p}{o})(\kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\pattArgsFwdF(p \cons o)(\kappa)}}
      \\
      \pattContFwdF(\pattRecord{\seq{\bind{x}{p}}})(\kappa)
      &=
      \elimRecord{\seq{x}}{\pattArgsFwdF(\seq{p})(\kappa)}
      \\[2mm]
      %========================================
      \pattContBwdF
      &::
      \Patt\;A\;\Gamma' \to \Elim\;\Gamma\;A\;B \to \Cont\;(\Gamma \cons \Gamma')\;B
      \\
      \pattContBwdF(\pattVar{x})(\elimVar{x}{\kappa})
      &=
      \kappa
      \\
      \pattContBwdF(\pattConstr{c}{\seq{p}})(\elimConstr{\elimBind{c}{\kappa}})
      &=
      \pattArgsBwdF(\seq{p})(\kappa)
      \\
      \pattContBwdF(\pattList{p}{o})(\elimConstr{\elimBind{\cCons}{\sigma}})
      &=
      \pattArgsBwdF(p \cons o)(\sigma)
      \\
      \pattContBwdF(\pattRecord{\seq{\bind{x}{p}}})(\elimRecord{\seq{x}}{\kappa})
      &=
      \pattArgsBwdF(\seq{p})(\kappa)
      \\
      %========================================
      \intertext{
      \flushleft\shadebox{$\pattContF :: \ListRestPatt\;A\;\Gamma' \to (\Cont\;(\Gamma \cons \Gamma')\;B \partialGCto \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \pattContFwdF
      &::
      \ListRestPatt\;A\;\Gamma' \to \Cont\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \pattContFwdF(\pattListEnd)(\kappa)
      &=
      \elimConstr{\elimBind{\cNil}{\kappa}}
      \\
      \pattContFwdF(\pattListNext{p}{o})(\kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\pattArgsFwdF(p \cons o)(\kappa)}}
      \\[2mm]
      %========================================
      \pattContBwdF
      &::
      \ListRestPatt\;A\;\Gamma' \to \Elim\;\Gamma\;A\;B \to \Cont\;(\Gamma \cons \Gamma')\;B
      \\
      \pattContBwdF(\pattListEnd)(\elimConstr{\elimBind{\cNil}{\kappa}})
      &=
      \kappa
      \\
      \pattContBwdF(\pattListNext{p}{o})(\elimConstr{\elimBind{\cCons}{\sigma}})
      &=
      \pattArgsBwdF(p \cons o)(\sigma)
      %========================================
      \intertext{
      \flushleft\shadebox{$\pattArgsF :: \PattSeq\;A\;\Gamma' \to (\Cont\;(\Gamma \cons \Gamma')\;B \partialGCto \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \pattArgsFwdF
      &::
      \PattSeq\;A\;\Gamma' \to \Cont\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \pattArgsFwdF(\seqEmpty)(\kappa)
      &=
      \kappa
      \\
      \pattArgsFwdF(\pi \cons \seq{\pi})(\sigma)
      &=
      \pattContFwdF(\pi)(\pattArgsFwdF(\seq{\pi})(\sigma))
      \\[2mm]
      %========================================
      \pattArgsBwdF
      &::
      \PattSeq\;A\;\Gamma' \to \Elim\;\Gamma\;A\;B \to \Cont\;(\Gamma \cons \Gamma')\;B
      \\
      %========================================
      \pattArgsBwdF(\seqEmpty)(\kappa)
      &=
      \kappa
      \\
      \pattArgsBwdF(\pi \cons \seq{\pi})(\sigma)
      &=
      \pattArgsBwdF(\seq{\pi})(\pattContBwdF(\pi)(\sigma))
   \end{salign}
