{\flushleft \shadebox{$e \desugarBwdR{\raw{s}} s$\text{ }}
\hfill \textbfit{$e$ backward-desugars to $s$ of shape $\raw{s}$}}
\begin{smathpar}
   \inferrule*[
      lab={\ruleName{$\desugarBwdR{}$-binary-apply}}
   ]
   {
      \strut
   }
   {
      \exBinaryApp{s}{\oplus}{s'}
      \desugarBwdR{\exBinaryApp{\raw{s}}{\oplus}{\raw{s}'}}
      \exApp{\exApp{(\oplus)}{e}}{e'}
   }
   %
   \and
   %
   \inferrule*[
      lab={\ruleName{$\desugarBwdR{}$-let-rec}}
   ]
   {
      \sigma_i = \clausesBwdF(\seq{\raw{c}_i}, \seq{c_i})
      \quad
      (\forall i \leq j)
   }
   {
      \exLetRecPiecewise{\seqRange{\bind{x_1}{\seq{c_1}}}{\bind{x_j}{\seq{c_j}}}}{e}
      \desugarBwdR{\exLetRecPiecewise{\seqRange{\bind{x_1}{\seq{\raw{c}_1}}}{\bind{x_j}{\seq{\raw{c}_j}}}}{\raw{s}}}
      \exLetRecElim{\seq{\bind{x}{\sigma}}}{e}
   }
   %
   \and
   %
   \inferrule*[
      lab={\ruleName{$\desugarBwdR{}$-match}}
   ]
   {
       \strut
   }
   {
      \exMatch{e}{\clausesBwdF(\seq{\raw{c}}, \sigma)}
      \desugarBwdR{\exMatch{\raw{e}}{\seq{\raw{c}}}}
      \exApp{\exFun{\sigma}}{e}
   }
   %
   \and
   %
   \inferrule*[
      lab={\ruleName{$\desugarBwdR{}$-non-empty-list}}
   ]
   {
      r \desugarBwdR{\raw{r}} e'
   }
   {
      \annList{e}{r}{\alpha}
      \desugarBwdR{\exList{\raw{e}}{\raw{r}}}
      \annCons{e}{e'}{\alpha}
   }
   %
   \and
   %
   \inferrule*[
      lab={\ruleName{$\desugarBwdR{}$-if}}
   ]
   {
      \strut
   }
   {
      \exApp{\exFun{\elimConstr{\elimBind{\cTrue}{e_1}, \elimBind{\cFalse}{e_2}}}}{e}
      \desugarBwdR{\exIfThenElse{\raw{e}}{\raw{e}_1}{\raw{e}_2}}
      \exIfThenElse{e}{e_1}{e_2}
   }
\end{smathpar}
