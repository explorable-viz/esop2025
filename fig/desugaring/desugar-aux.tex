      %========================================
      \intertext{
      \flushleft\shadebox{$\clausesF :: (\Clause\;\Gamma\;A\;B)^+ \partialGCto \Elim\;\Gamma\;A\;B$}
      }
      %========================================
      \clausesFwdF
      &::
      (\Clause\;\Gamma\;A\;B)^+ \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \clausesFwdF(\clause{\seq{p}}{e})
      &= \clauseFwdF(\seq{p}, e)
      \\
      \clausesFwdF((\clause{\seq{p}}{e}) \cons \seq{c}, e)
      &=
      \clauseFwdF(\seq{p}, e) \disjjoin \clausesFwdF(\seq{c})
      \\[2mm]
      %========================================
      \clausesBwdF
      &::
      (\Raw{\Clause}\;\Gamma\;A\;B)^+ \times \Elim\;\Gamma\;A\;B \to (\Clause\;\Gamma\;A\;B)^+
      \\
      %========================================
      \clausesBwdF(\clause{\seq{p}}{\raw{e}}, (\seq{p}, e))
      &=
      \clauseBwdF(\seq{p},e)
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\clauseF :: \Patt\;A\;\Gamma' \times \Cont\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B$}
      }
      %========================================
      \clauseFwdF
      &::
      \Patt\;A\;\Gamma' \times \Cont\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \clauseFwdF(\pattVar{x}, \kappa)
      &=
      \elimVar{x}{\kappa}
      \\
      \clauseFwdF(\pattList{p}{o}, \kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\clauseFwdF(p, \clauseFwdF(o, \kappa))}}
      \\
      \clauseFwdF(\pattRecord{\seqEmpty}, \kappa)
      &=
      \elimRecord{\seqEmpty}{\kappa}
      \\
      \clauseFwdF(\pattRecord{\seq{\bind{x}{p}} \cons \bind{y}{p'}}, \kappa)
      &=
      \elimRecord{\seq{x} \cons y}{\sigma}
      \quad\textit{where }
      \clauseFwdF(\pattRecord{\seq{\bind{x}p}}, \clauseFwdF(p', \kappa)) = \elimRecord{\seq{x}}{\sigma}
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\clauseF :: \ListRestPatt\;A\;\Gamma' \times \Cont\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B$}
      }
      %========================================
      \clauseFwdF
      &::
      \ListRestPatt\;A\;\Gamma' \times \Cont\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \clauseFwdF(\pattListEnd, \kappa)
      &=
      \elimConstr{\elimBind{\cNil}{\kappa}}
      \\
      \clauseFwdF((\pattListNext{p}{o}), \kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\clauseFwdF(p, \clauseFwdF(o, \kappa))}}
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\clauseF :: \PattSeq\;A\;\Gamma' \times \Expr\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B$}

      %========================================
      %========================================
      \clauseFwdF
      &::
      \PattSeq\;A\;\Gamma' \times \Expr\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \clauseFwdF(p \cons \seq{p}, e)
      &=
      \clauseFwdF(p, \exFun{(\clauseFwdF(\seq{p}, e))})
