      \flushleft\shadebox{$\clausesF :: (\Clause\;\Gamma\;A\;B)^+ \partialGCto \Elim\;\Gamma\;A\;B$}

      %========================================
      \clausesFwdF
      &::
      (\Clause\;\Gamma\;A\;B)^+ \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \clausesFwdF(\clause{\seq{p}}{e})
      &= \pattContFwdF(\seq{p}, e)
      \\

      &=
      \pattContFwdF(\seq{p}, e) \disjjoin \clausesFwdF(\seq{c})
      \\[2mm]

      \clausesBwdF
      &::
      (\Raw{\Clause}\;\Gamma\;A\;B)^+ \times \Elim\;\Gamma\;A\;B \to (\Clause\;\Gamma\;A\;B)^+
      \\
      %========================================
      \clausesBwdF(\clause{\seq{p}}{\raw{e}}, (\seq{p}, e))
      &=
      \pattContBwdF(\seq{p},e)

      %========================================
      \intertext{
      \flushleft\shadebox{$\pattContF :: \Patt\;A\;\Gamma' \times \Cont\;(\Gamma \cons \Gamma')\;B \partialGCto \Elim\;\Gamma\;A\;B$}
      }
      %========================================
      \pattContFwdF
      &::
      \Patt\;A\;\Gamma' \times \Cont\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \pattContFwdF(\pattVar{x}, \kappa)
      &=
      \elimVar{x}{\kappa}
      \\
      \pattContFwdF(\pattList{p}{o}, \kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\pattContFwdF(p, \pattContFwdF(o, \kappa))}}
      \\
      \pattContFwdF(\pattRecord{\seqEmpty}, \kappa)
      &=
      \elimRecord{\seqEmpty}{\kappa}
      \\
      \pattContFwdF(\pattRecord{\seq{\bind{x}{p}} \cons \bind{y}{p'}}, \kappa)
      &=
      \elimRecord{\seq{x} \cons y}{\sigma}
      \quad\textit{where }
      \pattContFwdF(\pattRecord{\seq{\bind{x}p}}, \pattContFwdF(p', \kappa)) = \elimRecord{\seq{x}}{\sigma}
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\pattContF :: \ListRestPatt\;A\;\Gamma' \times \Cont\;(\Gamma \cons \Gamma')\;B \partialGCto \Elim\;\Gamma\;A\;B$}
      }
      %========================================
      \pattContFwdF
      &::
      \ListRestPatt\;A\;\Gamma' \times \Cont\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \pattContFwdF(\pattListEnd, \kappa)
      &=
      \elimConstr{\elimBind{\cNil}{\kappa}}
      \\
      \pattContFwdF((\pattListNext{p}{o}), \kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\pattContFwdF(p, \pattContFwdF(o, \kappa))}}
      \\[2mm]
      %========================================
      \pattContBwdF
      &::
      \ListRestPatt\;A\;\Gamma' \times \Raw{\Cont}\;(\Gamma \cons \Gamma')\;B \times \Elim\;\Gamma\;A\;B \to
      \\
      &\phantom{an}
      \ListRestPatt\;A\;\Gamma' \times \Cont\;(\Gamma \cons \Gamma')\;B
      \\
      %========================================
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\pattsExprF :: \PattSeq\;A\;\Gamma' \to (\Expr\;(\Gamma \cons \Gamma')\;B \partialGCto \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \pattsExprFwdF
      &::
      \PattSeq\;A\;\Gamma' \times \Expr\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \pattsExprFwdF(p, e)
      &=
      \pattContFwdF(p, e)
      \\
      \pattsExprFwdF(p_1 \cons p_2 \cons \seq{p}', e)
      &=
      \pattContFwdF(p_1, \exFun{(\pattsExprFwdF(p_2 \cons \seq{p}', e))})
      \\[2mm]
      %========================================
      \pattsExprBwdF
      &::
      \PattSeq\;A\;\Gamma' \times \Raw{\Expr}\;(\Gamma \cons \Gamma')\;B \times \Elim\;\Gamma\;A\;B \to
      \\
      &
      \phantom{an}
      \PattSeq\;A\;\Gamma' \times \Expr\;(\Gamma \cons \Gamma')\;B
      \\
      \pattsExprBwdF((p, \raw{e}), \sigma)
      &=
      (p, \pattContBwdF((p, \raw{e}), \sigma))
      \\
      \pattsExprBwdF((p_1 \cons p_2 \cons \seq{p}', \raw{e}), \sigma)
      &=
      (p_1 \cons p_2 \cons \seq{p}', \pi_2(\pattsExprBwdF((p_2 \cons \seq{p}', \raw{e}), \sigma')))
      \\
      &\quad\textit{ where }
      \pattContBwdF(p_1, \sigma) = \exFun{\sigma'}
