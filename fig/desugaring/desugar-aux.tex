   \begin{salign}
      %========================================
      \PattSeqOpt &:: \Cxt \to \Set{Set}
      \\
      \PattSeqOpt\;\Gamma &\eqdef 1 + \exists A.\PattSeq\;A\;\Gamma
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\pattsExprF :: \PattSeq\;A\;\Gamma' \to (\Expr\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \pattsExprF
      &::
      \PattSeq\;A\;\Gamma' \to \Expr\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \pattsExprF(p)(e)
      &=
      \pattContF(p)(e)
      \\
      \pattsExprF(p' \cons \seq{p})(e)
      &=
      \pattContF(p')(\exFun{(\pattsExprF(\seq{p})(e))})
      \quad\textit{ if }\seq{p} \neq \seqEmpty
      %========================================
      \intertext{
      \flushleft\shadebox{$\pattContF :: \Patt\;A\;\Gamma' \to (\Cont\;(\Gamma \concat \Gamma')\;B \to \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \pattContF
      &::
      \Patt\;A\;\Gamma' \to \Cont\;(\Gamma \concat \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \pattContF(\pattVar{x})(\kappa)
      &=
      \elimVar{x}{\kappa}
      \\
      \pattContF(\pattConstr{c}{\seq{p}})(\kappa)
      &=
      \elimConstr{\elimBind{c}{\pattArgsF(\seq{p})(\kappa)}}
      \\
      \pattContF(\pattList{p}{o})(\kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\pattArgsF(p \cons o)(\kappa)}}
      \\
      \pattContF(\pattRecord{\seq{\bind{x}{p}}})(\kappa)
      &=
      \elimRecord{\seq{x}}{\pattArgsF(\seq{p})(\kappa)}
      \\
      %========================================
      \intertext{
      \flushleft\shadebox{$\pattContF :: \ListRestPatt\;A\;\Gamma' \to (\Cont\;(\Gamma \concat \Gamma')\;B \to \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \pattContF
      &::
      \ListRestPatt\;A\;\Gamma' \to \Cont\;(\Gamma \concat \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \pattContF(\pattListEnd)(\kappa)
      &=
      \elimConstr{\elimBind{\cNil}{\kappa}}
      \\
      \pattContF(\pattListNext{p}{o})(\kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\pattArgsF(p \cons o)(\kappa)}}
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\pattArgsF :: \PattSeqOpt\;\Gamma' \to (\Cont\;(\Gamma \concat \Gamma')\;A \to \Cont\;\Gamma\;A)$}
      }
      %========================================
      \pattArgsF
      &::
      \PattSeqOpt\;\Gamma' \to \Cont\;(\Gamma \concat \Gamma')\;A \to \Cont\;\Gamma\;A
      \\
      %========================================
      \pattArgsF(\seqEmpty)(\kappa)
      &=
      \kappa
      \\
      \pattArgsF(\pi \cons \seq{\pi})(\sigma)
      &=
      \pattContF(\pi)(\pattArgsF(\seq{\pi})(\sigma))
   \end{salign}
