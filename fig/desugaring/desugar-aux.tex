      \intertext{
      \flushleft\shadebox{$\clausesF :: (\Clause\;\Gamma\;A\;B)^+ \partialGCto \Elim\;\Gamma\;A\;B$}
      }
      %========================================
      \clausesFwdF
      &::
      (\Clause\;\Gamma\;A\;B)^+ \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \clausesFwdF(\clause{\seq{p}}{e})
      &= \pattContFwdF(\seq{p})(e)
      \\
      \clausesFwdF((\clause{\seq{p}}{e}) \cons \seq{c})
      &=
      \pattContFwdF(\seq{p})(e) \disjjoin \clausesFwdF(\seq{c})
      \quad\textit{ if }\seq{c} \neq \seqEmpty
      \\[2mm]
      %========================================
      \clausesBwdF
      &::
      (\Raw{\Clause}\;\Gamma\;A\;B)^+ \times \Elim\;\Gamma\;A\;B \to (\Clause\;\Gamma\;A\;B)^+
      \\
      %========================================
      \clausesBwdF(\clause{\seq{p}}{\raw{e}}, \sigma)
      &=
      \clause{\seq{p}}{\pattContBwdF(\seq{p})(\raw{e}, \sigma)}
      \\
      \clausesBwdF((\clause{\seq{p}}{\raw{e}}) \cons \seq{\raw{c}}, \sigma \disjjoin \tau)
      &=
      (\clause{\seq{p}}{\pattContBwdF(\seq{p})(\raw{e}, \sigma)}) \cons \clausesBwdF(\seq{\raw{c}}, \tau)
      \quad\textit{ if }\seq{c} \neq \seqEmpty
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\pattContF :: \Patt\;A\;\Gamma' \to (\Cont\;(\Gamma \cons \Gamma')\;B \partialGCto \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \pattContFwdF
      &::
      \Patt\;A\;\Gamma' \to \Cont\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \pattContFwdF(\pattVar{x})(\kappa)
      &=
      \elimVar{x}{\kappa}
      \\
      \pattContFwdF(\pattList{p}{o})(\kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\pattContFwdF(p)(\pattContFwdF(o)(\kappa))}}
      \\
      \pattContFwdF(\pattRecord{\seqEmpty})(\kappa)
      &=
      \elimRecord{\seqEmpty}{\kappa}
      \\
      \pattContFwdF(\pattRecord{\seq{\bind{x}{p}} \cons \bind{y}{p'}})(\kappa)
      &=
      \elimRecord{\seq{x} \cons y}{\sigma}
      \quad\textit{where }
      \pattContFwdF(\pattRecord{\seq{\bind{x}p}})(\pattContFwdF(p')(\kappa)) = \elimRecord{\seq{x}}{\sigma}
      \\[2mm]
      %========================================
      \pattContBwdF
      &::
      \Patt\;A\;\Gamma' \to \Raw{\Cont}\;(\Gamma \cons \Gamma')\;B \times \Elim\;\Gamma\;A\;B \to \Cont\;(\Gamma \cons \Gamma')\;B
      \\
      \pattContBwdF(\pattVar{x})(\elimVar{x}{\kappa})
      &=
      \kappa
      \\
      \pattContBwdF(\pattList{p}{o})(\elimConstr{\elimBind{\cCons}{\sigma}})
      &=
      \pattContBwdF(o, \pattContBwdF(p, \sigma))
      \\
      %========================================
      \intertext{
      \flushleft\shadebox{$\pattContF :: \ListRestPatt\;A\;\Gamma' \to (\Cont\;(\Gamma \cons \Gamma')\;B \partialGCto \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \pattContFwdF
      &::
      \ListRestPatt\;A\;\Gamma' \to \Cont\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \pattContFwdF(\pattListEnd)(\kappa)
      &=
      \elimConstr{\elimBind{\cNil}{\kappa}}
      \\
      \pattContFwdF(\pattListNext{p}{o})(\kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\pattContFwdF(p)(\pattContFwdF(o)(\kappa))}}
      \\[2mm]
      %========================================
      \pattContBwdF
      &::
      \ListRestPatt\;A\;\Gamma' \to \Raw{\Cont}\;(\Gamma \cons \Gamma')\;B \times \Elim\;\Gamma\;A\;B \to \Cont\;(\Gamma \cons \Gamma')\;B
      \\
      \pattContBwdF(\pattListEnd)(\elimConstr{\elimBind{\cNil}{\kappa}})
      &=
      \kappa
      \\
      \pattContBwdF(\pattListNext{p}{o})(\elimConstr{\elimBind{\cCons}{\sigma}})
      &=
      \pattContBwdF(o, \pattContBwdF(p, \sigma))
      %========================================
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\pattsExprF :: \PattSeq\;A\;\Gamma' \to (\Expr\;(\Gamma \cons \Gamma')\;B \partialGCto \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \pattsExprFwdF
      &::
      \PattSeq\;A\;\Gamma' \to \Expr\;(\Gamma \cons \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \pattsExprFwdF(p)(e)
      &=
      \pattContFwdF(p)(e)
      \\
      \pattsExprFwdF(p_1 \cons p_2 \cons \seq{p}')(e)
      &=
      \pattContFwdF(p_1)(\exFun{(\pattsExprFwdF(p_2 \cons \seq{p}')(e))})
      \\[2mm]
      %========================================
      \pattsExprBwdF
      &::
      \PattSeq\;A\;\Gamma' \to \Raw{\Expr}\;(\Gamma \cons \Gamma')\;B \times \Elim\;\Gamma\;A\;B \to \Expr\;(\Gamma \cons \Gamma')\;B
      \\
      \pattsExprBwdF(p)(\raw{e}, \sigma)
      &=
      \pattContBwdF(p)(\raw{e}, \sigma)
      \\
      \pattsExprBwdF(p_1 \cons p_2 \cons \seq{p}')(\raw{e}, \sigma)
      &=
      \pattsExprBwdF(p_2 \cons \seq{p}')(\raw{e}, \sigma')
      \quad\textit{ where }
      \pattContBwdF(p_1)(\sigma) = \exFun{\sigma'}
