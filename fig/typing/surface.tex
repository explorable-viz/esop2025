   \flushleft \shadebox{$\upGamma \vdash \seq{g}: \upDelta$}
   \begin{smathpar}
      \inferrule*[right={
         \textnormal{$\upDelta = \seq{\bind{x}{\tyFun{A}{B}}}$}
      }]
      {
         \upGamma \concat \upDelta \vdash \upmu_i: \tyFun{A_i}{B_i}
         \quad
         (\forall i \numleq j)
      }
      {
         \upGamma \vdash \seqRange{\bind{x_1}{\upmu_1}}{\bind{x_j}{\upmu_j}}: \upDelta
      }
   \end{smathpar}
   
   \vspace{5pt}
   \flushleft \shadebox{$\upGamma \vdash s: A$}
   \begin{smathpar}
      \inferrule*[right={$\primOp: \tyForeign{(A,B)}{C} \in \upGamma$}]
      {
         \upGamma \vdash s: A
         \\
         \upGamma \vdash s': B
      }
      {
         \upGamma \vdash \exBinaryApp{s}{\oplus}{s'}: C
      }
      %
      \and
      %
      \inferrule*[]
      {
         \upGamma \vdash \upmu: \tyFun{A}{B}
      }
      {
         \upGamma \vdash \exApp{\uplambda}{\upmu}: \tyFun{A}{B}
      }
      %
      \and
      %
      \inferrule*
      {
         \upGamma \vdash \seq{g}: \upDelta
         \\
         \upGamma \concat \upDelta \vdash s: A
      }
      {
         \upGamma \vdash \exLetRec{\seq{g}}{s}: A
      }
      %
      \and
      %
      \inferrule*
      {
         \upGamma \vdash s: \tyBool
         \\
         \upGamma \vdash s_1: A
         \\
         \upGamma \vdash s_2: A
      }
      {
         \upGamma \vdash \exIfThenElse{s}{s_1}{s_2}: A
      }
      %
      \and
      %
      \inferrule*
      {
         \upGamma \vdash s: A
         \\
         \upGamma \vdash \upmu: \tyFun{A}{B}
      }
      {
         \upGamma \vdash \exMatch{s}{\upmu}: B
      }
      %
      \and
      %
      \inferrule*
      {
         p: A \dashv \upGamma'
         \\
         \upGamma \vdash s: A
         \\
         \upGamma \concat \upGamma' \vdash s': B
      }
      {
         \upGamma \vdash \exLet{p}{s}{s'}: B
      }
      %
      \and
      %
      \inferrule*
      {
         \upGamma \vdash s : A
         \\
         \upGamma \vdash r : \tyList{A}
      }
      {
         \upGamma \vdash \exList{s}{r} : \tyList{A}
      }
      %
      \and
      %
      \inferrule*
      {
         \upGamma \vdash s: \tyInt
         \\
         \upGamma \vdash s': \tyInt
      }
      {
         \upGamma \vdash \exListEnum{s}{s'}: \tyList{\tyInt}
      }
      %
      \and
      %
      \inferrule*
      {
         \upGamma \vdash \seq{q} \dashv \upDelta
         \\
         \upGamma \concat \upDelta \vdash s: A
      }
      {
         \upGamma \vdash \exListComp{s}{\seq{q}}: \tyList{A}
      }
      \end{smathpar}
   
      % List rest
      \vspace{5pt}
      \flushleft \shadebox{$\upGamma \vdash r: \tyList{A}$}
      \begin{smathpar}
      \inferrule*
      {
         \strut
      }
      {
         \upGamma \vdash \exListEnd : \tyList{A}
      }
      %
      \and
      %
      \inferrule*
      {
         \upGamma \vdash s : A
         \\
         \upGamma \vdash r : \tyList{A}
      }
      {
         \upGamma \vdash \exListNext{s}{r} : \tyList{A}
      }
      \end{smathpar}

      }
