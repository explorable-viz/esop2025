   \small{\flushleft \shadebox{$\upGamma \vdash e: A$}%
   \hfill \textbfit{$e$ has type $A$ under $\upGamma$}}
   \begin{smathpar}
   \inferrule*[right={$x : A \in \upGamma$}]
   {
      \strut
   }
   {
      \upGamma \vdash \exVar{x}: A
   }
   %
   \and
   %
   \inferrule*
   {
      \strut
   }
   {
      \upGamma \vdash \exInt{n}: \tyInt
   }
   %
   \and
   %
   \inferrule*[]
   {
      \upGamma \vdash \seq{e}: \seq{A}
      \\
      d: (c: \tyFun{\seq{A}}{D})
   }
   {
      \upGamma \vdash \exConstr{c}{\seq{e}}: d
   }
   %
   \and
   %
   \inferrule*
   {
      \upGamma \vdash e_i: \tyStr
      \\
      \upGamma \vdash e'_i: A
      \quad
      (\forall i \numleq \length{\seq{e}})
   }
   {
      \upGamma \vdash \exDict{\seq{\bind{e}{e}}'}: \tyDict{A}
   }
   %
   \and
   %
   \inferrule*
   {
      \upGamma \vdash e_i: A_i
      \quad
      (\forall i \numleq \length{\seq{x}})
   }
   {
      \upGamma \vdash \exRec{\seq{\bind{x}{e}}}: \tyRec{\seq{\bind{x}{A}}}
   }
   %
   \and
   %
   \inferrule*[
      right={$i \numleq \length{\seq{x}}$}
   ]
   {
      \upGamma \vdash e: \tyRec{\seq{\bind{x}{A}}}
   }
   {
      \upGamma \vdash \exRecProj{e}{x_i}: A_i
   }
   %
   \and
   %
   \inferrule*
   {
      \upGamma \vdash \upsigma : \elimTy{A}{B}
   }
   {
      \upGamma \vdash \exFun{\upsigma} : \tyFun{A}{B}
   }
   %
   \and
   %
   \inferrule*[
      right={$f: \tyForeign{\seq{A}}{B} \in \Phi$}
   ]
   {
      \upGamma \vdash e_i: A
      \quad
      (\forall i \numleq j)
   }
   {
      \upGamma \vdash f(\seq{e}): B
   }
   %
   \and
   %
   \inferrule*
   {
      \upGamma \vdash e: \tyFun{A}{B}
      \\
      \upGamma \vdash e': A
   }
   {
      \upGamma \vdash \exApp{e}{e'}: B
   }
   %
   \and
   %
   \inferrule*
   {
      \strut
   }
   {
      \upGamma \vdash \exNil: \tyList{A}
   }
   %
   \and
   %
   \inferrule*
   {
      \upGamma \vdash e: A
      \\
      \upGamma \vdash e': \tyList{A}
   }
   {
      \upGamma \vdash (\exCons{e}{e'}): \tyList{A}
   }
   %
   \and
   %
   \inferrule*
   {
      \upGamma \vdash \uprho: \upDelta
      \\
      \upGamma \concat \upDelta \vdash e: A
   }
   {
      \upGamma \vdash \exLetRec{\uprho}{e}: A
   }
   \end{smathpar}

   {\small \flushleft \shadebox{$\vdash v: A$}%
   \hfill \textbfit{$v$ has type $A$}}
   \begin{smathpar}
   \inferrule*
   {
      \strut
   }
   {
      \vdash \exInt{n}: \tyInt
   }
   %
   \and
   %
   \inferrule*
   {
      \strut
   }
   {
      \upGamma \vdash \exTrue: \tyBool
   }
   %
   \and
   %
   \inferrule*
   {
      \strut
   }
   {
      \upGamma \vdash \exFalse: \tyBool
   }
   %
   \and
   %
   \inferrule*
   {
      \vdash v_i: A_i
      \quad
      (\forall i \numleq \length{\seq{x}})
   }
   {
      \vdash \exRec{\seq{\bind{x}{v}}}: \tyRec{\seq{\bind{x}{A}}}
   }
   %
   \and
   %
   \inferrule*
   {
      \strut
   }
   {
      \vdash \exNil: \tyList{A}
   }
   %
   \and
   %
   \inferrule*
   {
      \vdash u: A
      \\
      \vdash v: \tyList{A}
   }
   {
      \vdash (\exCons{u}{v}): \tyList{A}
   }
   %
   \and
   %
   \inferrule*
   {
      \vdash \upgamma: \upGamma
      \\
      \upGamma \vdash \uprho: \upDelta
      \\
      \upGamma \concat \upDelta \vdash \upsigma: \elimTy{A}{B}
   }
   {
      \vdash \exClosure{\upgamma}{\uprho}{\upsigma}: \tyFun{A}{B}
   }
   %
   \and
   %
   \inferrule*[]
   {
      \upGamma \vdash \seq{v}: \seq{A}
      \\
      d: (c: \tyFun{\seq{A}}{D})
   }
   {
      \upGamma \vdash \exConstr{c}{\seq{v}}: d
   }
   \end{smathpar}
