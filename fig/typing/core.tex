   }

   \and

   \inferrule*

      \\
      \Gamma \vdash h: \Delta
      \\

   }
   {
      \vdash \exClosure{\upgamma}{h}{\sigma}: \tyFun{A}{B}
   }
   %
   \and
   %
   \inferrule*[]
   {
      \Gamma \vdash \seq{v}: \seq{A}
      \\
      \Delta(c)= \tyFun{\seq{A}}{D}
   }
   {
      \Gamma \vdash \exConstr{c}{\seq{v}}: D
   }
   \end{smathpar}
