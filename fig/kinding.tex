{\small \flushleft \shadebox{$\Delta \vdash T: K$}%
\hfill \textbfit{$T$ has kind $K$ in $\Delta$}}
\begin{smathpar}
   \inferrule*
   {
      \bind{\alpha}{K} \in \Delta
   }
   {
      \Delta \vdash \tyVar{\alpha}: K
   }
   %
   \and
   %
   \inferrule*
   {
      \strut
   }
   {
      \Delta \vdash \tyInt: \kiType
   }
   %
   \and
   %
   \inferrule*
   {
      \Delta \vdash A: \kiType
      \\
      \Delta \vdash C: \kiComp
   }
   {
      \Delta \vdash \tyFun{A}{C}: \kiType
   }
   %
   \and
   %
   \inferrule*
   {
      \Delta \vdash C: \kiComp
      \\
      \Delta \vdash D: \kiComp
   }
   {
      \Delta \vdash \tyHandler{C}{D}: \kiType
   }
   %
   \and
   %
   \inferrule*
   {
      \Delta \vdash C: \kiComp
      \\
      \Delta \vdash D: \kiComp
   }
   {
      \Delta \vdash \tyHandler{C}{D}: \kiType
   }
   %
   \and
   %
   \inferrule*
   {
      \Delta \vdash A_i: \kiType
      \quad
      (\forall i \in \length{\seq{A}})
   }
   {
      \Delta \vdash \seq{\bind{\ell}{A}}: \kiRow
   }
   %
   \and
   %
   \inferrule*
   {
      \Delta \vdash R: \kiRow
   }
   {
      \Delta \vdash \tyRecord{R}: \kiType
   }
   %
   \and
   %
   \inferrule*
   {
      \Delta \vdash R: \kiRow
   }
   {
      \Delta \vdash \tyVariant{R}: \kiType
   }
   %
   \and
   %
   \inferrule*
   {
      \Delta \vdash R: \kiRow
      \\
      \Delta \vdash A: \kiType
   }
   {
      \Delta \vdash \tyEffect{A}{R}: \kiComp
   }
   %
   \and
   %
   \inferrule*
   {
      \cxtExtend{\Delta}{\bind{\alpha}{K}} \vdash C: \kiComp
   }
   {
      \Delta \vdash \tyForall{\alpha}{K}{C}: \kiType
   }
   %
   \and
   %
   \inferrule*
   {
      \Delta \vdash C: \kiComp
   }
   {
      \Delta \vdash C: \kiCont
   }
   %
   \and
   %
   \inferrule*
   {
      \Delta \vdash A: \kiType
      \\
      \Delta \vdash E: \kiCont
   }
   {
      \Delta \vdash \elimTy{A}{E}: \kiCont
   }
\end{smathpar}
