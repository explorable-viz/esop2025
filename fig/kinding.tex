\begin{figure}
{\small \flushleft \shadebox{$\Delta \vdash T: K$}%
\hfill \textbfit{$T$ has kind $K$ in $\Delta$}}
\begin{smathpar}
   \inferrule*
   {
      \bind{\alpha}{K} \in \Delta
   }
   {
      \Delta \vdash \tyVar{\alpha}: K
   }
   %
   \and
   %
   \inferrule*
   {
      \strut
   }
   {
      \Delta \vdash \tyInt: \kiVal
   }
   %
   \and
   %
   \inferrule*
   {
      \Delta \vdash A: \kiVal
      \\
      \Delta \vdash C: \kiComp
   }
   {
      \Delta \vdash \tyFun{A}{C}: \kiVal
   }
   %
   \and
   %
   \inferrule*
   {
      \Delta \vdash C: \kiComp
      \\
      \Delta \vdash D: \kiComp
   }
   {
      \Delta \vdash \tyHandler{C}{D}: \kiVal
   }
   %
   \and
   %
   \inferrule*
   {
      \Delta \vdash E: \kiComp
      \\
      \Delta \vdash A: \kiVal
   }
   {
      \Delta \vdash \tyEffect{E}{A}: \kiVal
   }
\end{smathpar}
\caption{Kinding rules}
\end{figure}
