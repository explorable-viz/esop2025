   {\small \flushleft \shadebox{$V, \sigma \match \gamma, P$}%
   \hfill \textbfit{$\sigma$ matches $V$ and yields $\rho$ and $P$}}
   \begin{smathpar}
      \inferrule*
      {
         \strut

      }

      \and

      \inferrule*
      {
         i \leq \length{\seq{P}}

         V, P_i \match \gamma, Q
      }

         \exVariant{\ell_i}{V}, \elimVariant{\ell}{P}{P'} \match \gamma, Q
      }
      %
      \and
      %
      \inferrule*
      {
         \ell' \notin \set{\seq{\ell}}
         \quad
         \exVariant{\ell'}{V}, P' \match \gamma, Q
      }
      {
         \exVariant{\ell'}{V}, \elimVariant{\ell}{P}{P'} \match \gamma, Q
      }
      %
      \and
      %
      \inferrule*
      {
         \strut
      }
      {
         V, \elimRecord{P}{\seqEmpty} \match \seqEmpty, P
      }
      %
      \and
      %
      \inferrule*
      {
         V, \elimRecord{P}{\seq{\ell}} \match \gamma, \sigma
         \\
         U, \sigma \match \gamma', Q
      }

         \exRecordExtend{V}{\bind{\ell'}{U}}, \elimRecord{P}{\seq{\ell}, \ell'}
         \match
         \gamma \concat \gamma', Q
      }
   \end{smathpar}

   {\small \flushleft \shadebox{$M \reduce N$}%
   \hfill \textbfit{$M$ reduces to $N$}}
   \begin{salign}
      \exApp{(\exFun{\sigma})}{V}
      &\reduce
      \substNew{\rho}{M}
      \quad
      \textit{if }V, \sigma \match \rho, M
      \\
      \exPolyApp{(\exPolyFun{\alpha}{M})}{T}
      &\reduce
      \substNew{(\seqEmpty,\bind{\alpha}{T})}{M}
      \\
      \exMatch{V}{\sigma}
      &\reduce
      \substNew{\rho}{M}
      \quad
      \textit{if }V, \sigma \match \rho, M
      \\
      \exHandle{\exReturn{V}}{(\hReturn{x}{M}) \cons H}
      &\reduce
      \subst{M}{V}{x}
      \\
      \begin{split}
      \exHandle{\plug{\mathcal{E}}{\exOp{\ell}{V}}}{H}
      &\reduce
      \subst{\subst{M}{V}{x}}{\exFun{\elimVar{y}{\exHandle{\plug{\mathcal{E}}{\exReturn{y}}}{H}}}}{k}
      \\
      &\quad\,\,\,\;\textit{if }
      H = (\hEffect{\ell}{x}{k}{M}) \cons H'
      \text{ and }
      \ell \notin \BL(\mathcal{E})
      \end{split}
      \\
      M &\reduce N
      \quad\hfill
      \textit{if }\plug{\mathcal{E}}{M} \reduce \plug{\mathcal{E}}{N}
   \end{salign}
