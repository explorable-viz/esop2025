Start with the following assumptions:
\begin{equation}
   \compatPatt{c}{\upmu}
\end{equation}
\begin{equation}
   \locallabel{desugar-left}
   c \desugarFwdS \upsigma
\end{equation}
\begin{equation}
   \locallabel{desugar-right}
   \upmu \desugarFwdS \upsigma'
\end{equation}
We proceed by induction on the derivation of $\compatPattSym$
\begin{proof}
\small
\begin{flalign}
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[]
         {
            \exists i.\; (\compatPatt{p_i}{p_i'})
            \opAnd
            \forall j < i.\; p_j = p'_j
         }
         {
            \compatPatt{(\clause{\seqRangeOp{p_1}{p_{i-1}}{\cdot}\cdot p_i \cdot \seq{p}}{s})}{(\clause{\seqRangeOp{p_1}{p_{i-1}}{\cdot}\cdot p'_i \cdot \seq{p}'}{s'})} 
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \text{Have three cases for $\compatPatt{p_i}{p'_i}$}
   &
   \notag
   \\
   \subcase{constr-neq}
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPatt{}{}$-constr-neq}}
         ]
         {
            \exVar{c} \neq \exVar{c'}
         }
         {
            \compatPatt{\exConstr{c}{\seq{p}''}}{\exConstr{c'}{\seq{p}'''}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \text{Let $\uptau = \seqRangeOp{p''_1}{p''_n}{\elimmapsto} \elimmapsto (\varepsilon \disjjoin \clause{\seq{p}}{s})$}
   &
   \notag
   \\
   &
   \text{Let $\uptau' = \seqRangeOp{p'''_1}{p'''_n}{\elimmapsto} \elimmapsto (\varepsilon \disjjoin \clause{\seq{p}'}{s})$}
   &
   \notag
   \\
   &
   (\clause{\seqRangeOp{p_1}{p_{i-1}}{\cdot}\cdot p_i \cdot \seq{p}}{s}) \desugarFwdS \upsigma = \seqRangeOp{p_1}{p_{i-1}}{\elimmapsto} \elimmapsto c \elimmapsto \uptau
   &
   \text{Assumption \localref{desugar-left}}
   \notag
   \\
   &
   (\clause{\seqRangeOp{p_1}{p_{i-1}}{\cdot}\cdot p'_i \cdot \seq{p}'}{s'}) \desugarFwdS \upsigma' = \seqRangeOp{p_1}{p_{i-1}}{\elimmapsto} \elimmapsto c' \elimmapsto \uptau'
   &
   \text{Assumption \localref{desugar-right}}
   \notag
   \\
   &
   \qedLocal
   \upsigma \disjjoin (\clause{\seqRangeOp{p_1}{p_{i-1}}{\cdot}\cdot p'_i \cdot \seq{p}'}{s'}) = \seqRangeOp{p_1}{p_{i-1}}{\elimmapsto} \elimmapsto \elimPair{\elimBind{c}{\uptau}}{\elimBind{c'}{\uptau'}}
   &
   \text{$\disjjoin$-constr-neq}
   \notag
   \\
   \subcase{constr-eq}
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPatt{}{}$-constr-eq}}
         ]
         {
            \compatPatt{\seq{p}''}{\seq{p}'''}
         }
         {
            \compatPatt{\exConstr{c}{\seq{p}''}}{\exConstr{c}{\seq{p}'''}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \text{Let $\uptau = \seqRangeOp{p''_1}{p''_n}{\elimmapsto} \elimmapsto (\varepsilon \disjjoin \clause{\seq{p}}{s})$}
   &
   \notag
   \\
   &
   \text{Let $\uptau' = \seqRangeOp{p'''_1}{p'''_n}{\elimmapsto} \elimmapsto (\varepsilon \disjjoin \clause{\seq{p}'}{s})$}
   &
   \notag
   \\
   &
   (\clause{\seqRangeOp{p_1}{p_{i-1}}{\cdot}\cdot p_i \cdot \seq{p}}{s}) \desugarFwdS \upsigma = \seqRangeOp{p_1}{p_{i-1}}{\elimmapsto} \elimmapsto c \elimmapsto \uptau
   &
   \text{Assumption \localref{desugar-left}}
   \notag
   \\
   &
   (\clause{\seqRangeOp{p_1}{p_{i-1}}{\cdot}\cdot p'_i \cdot \seq{p}'}{s'}) \desugarFwdS \upsigma' = \seqRangeOp{p_1}{p_{i-1}}{\elimmapsto} \elimmapsto c \elimmapsto \uptau'
   &
   \text{Assumption \localref{desugar-right}}
   \notag
   \\
   &
   \exists i'.\; \compatPatt{p''_{i'}}{p'''_{i'}} \opAnd \forall j' < i'.\; p''_{j'} = p'''_{j'} 
   &
   \text{inversion on $\compatPatt{\seq{p}}{\seq{p}'}$}
   \\
   &
   \text{If $p''_{i'} = \exConstr{c}{\seq{p}^{\dag}} \opAnd p'''_{i'} = \exConstr{c'}{\seq{p}'^{\dag}}$ with $\upSigma(c) = \upSigma(c') \opAnd c \neq c'$ then:}
   &
   \\
   &
   \uptau = \seqRangeOp{p''_1}{p''_{i'-1}}{\elimmapsto} \elimmapsto c \elimmapsto \seqRangeOp{p^{\dag}_1}{p^{\dag}_{m}}{\elimmapsto} \elimmapsto \seqRangeOp{p''_{i'+1}}{p''_{n}}{\elimmapsto} \elimmapsto (\varepsilon \disjjoin \clause{\seq{p}}{s})
   &
   \text{4}
   \notag
   \\
   &
   \uptau' = \seqRangeOp{p''_1}{p''_{i'-1}}{\elimmapsto} \elimmapsto c' \elimmapsto \seqRangeOp{p'^{\dag}_1}{p'^{\dag}_{m}}{\elimmapsto} \elimmapsto \seqRangeOp{p'''_{i'+1}}{p'''_{n}}{\elimmapsto} \elimmapsto (\varepsilon \disjjoin \clause{\seq{p}'}{s'})
   &
   \text{4}
   \notag
   \\
   &
   \text{Let $\uptau^{\dag} = \seqRangeOp{p^{\dag}_1}{p^{\dag}_{m}}{\elimmapsto} \elimmapsto \seqRangeOp{p''_{i'+1}}{p''_{n}}{\elimmapsto} \elimmapsto (\varepsilon \disjjoin \clause{\seq{p}}{s})$}
   &
   \notag
   \\
   &
   \text{Let $\uptau'^{\dag} = \seqRangeOp{p'^{\dag}_1}{p'^{\dag}_{m}}{\elimmapsto} \elimmapsto \seqRangeOp{p'''_{i'+1}}{p'''_{n}}{\elimmapsto} \elimmapsto (\varepsilon \disjjoin \clause{\seq{p}'}{s'})$}
   &
   \notag
   \\
   &
   \upsigma \disjjoin (\clause{\seqRangeOp{p_1}{p_{i-1}}{\cdot}\cdot p'_i \cdot \seq{p}'}{s'}) =
   &
   \notag
   \\
   &
   \quad\seqRangeOp{p_1}{p_{i-1}}{\elimmapsto} \elimmapsto c \elimmapsto \seqRangeOp{p''_1}{p''_{i'-1}}{\elimmapsto} \elimmapsto \elimPair{\elimBind{c}{\uptau^{\dag}}}{\elimBind{c'}{\uptau'^{\dag}}}
   &
   \notag
   \\
   &
   \text{If ¬$(5)$, repeat until find non-equal constructors, then construct join accordingly} 
   &
   \notag
   \\
   \subcase{record}
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPatt{}{}$-record}}
         ]
         {
            \compatPatt{\seq{p}}{\seq{p}'}
         }
         {
            \compatPatt{\exRecord{\seq{\bind{\exVar{x}}{p}}}}{\exRecord{\seq{\bind{\exVar{x}}{p}}'}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \text{Isomorphic proof to the constr-eq case}
   &
   \notag
\end{flalign}      
\begin{flalign}
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            right={$\length{\upmu}\ge2$}
         ]
         {
            \forall c_i \in \upmu. \; \compatPatt{c}{c_i}
         }
         {
            \compatPatt{c}{\upmu}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   c \desugarFwdS \upsigma
   &
   \text{assumption \localref{desugar-left}}
   \notag
   \\
   &
   \forall c_i \in \upmu.\; \exists \uptau_i.\; \upsigma \disjjoin \upsigma_i = \uptau_i
   &
   \text{premise}
   \\
   &
   \forall c_i,c_j \in \upmu.\; c_i \desugarFwdS \upsigma_i,\; c_k \desugarFwdS \upsigma_j,\; \exists \uptau_{i,j}.\;\upsigma_i \disjjoin \upsigma_j = \uptau_{i,j}
   &
   \text{premise}
   \\
   &
   \upmu \desugarFwdS \upsigma'
   &
   \text{assumption \localref{desugar-right}}
   \notag
   \\
   &
   \qedLocal
   \exists \uptau.\;\upsigma \disjjoin \upsigma' = \uptau
   &
   \thmref{join-preserves-join}
   \notag
\end{flalign}
\end{proof}
