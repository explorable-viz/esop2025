By case analysis and induction on derivations of $\compatPattSym$. For our first pass, we will only consider pairs of clauses. Then, we will generalize this to arbitrary numbers of clauses. We disallow certain combinations of patterns which might otherwise be allowed. For example, we do not allow clauses formed in the following fashion:
\begin{lstlisting}[frame=single]
   foo :: forall a b. List a -> List a -> b
   foo (Cons y ys) (Cons z zs) = ...
   foo  x           Nil        = ...

   bar :: a -> ... -> c
   bar x ... = ...
   bar y ... = ...
\end{lstlisting}
IE when we have variable patterns, the appropriately indexed arguments must all be variable patterns with the same name.
Pattern matching proceeds from left to right so the following \textit{is} allowed:
\begin{lstlisting}[frame=single]
   baz :: forall a b. List a -> List a -> b
   baz (Cons y ys) (Cons z zs) = ...
   baz  Nil         x          = ...
\end{lstlisting}
\begin{proof}
\small
\begin{flalign}
   \intertext{\crossrule}
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPattSym$-head}}
         ]
         {
            \compatPatt{p}{p'}
         }
         {
            \compatPatt{\clause{p\cdot\seq{p_{rest}}}{s}}{p'\cdot\clause{\seq{p_{rest}}'}{s'}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \clause{p\cdot\seq{p_{rest}}}{s}\desugarFwdS \upsigma
   &
   \\
   &
   \clause{p'\cdot\seq{p_{rest}}}{s'}\desugarFwdS \upsigma'
   &
   \\
   &
   \text{$\clause{\seq{p_{rest}}}{s}\desugarFwdS \upsigma_{rest}$}
   &
   \text{(1) and $\desugarFwdS$-clause-curried}
   \\
   &
   \text{$\clause{\seq{p_{rest}}'}{s'}\desugarFwdS \upsigma_{rest}'$}
   &
   \text{(2) and $\desugarFwdS$-clause-curried}
   \\
   &
   \text{Have three cases for $\compatPatt{p}{p'}$}
   &
   \notag
   \\
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPatt{}{}$-constr-neq}}
         ]
         {
            \exVar{c} \neq \exVar{c'}
         }
         {
            \compatPatt{\exConstr{c}{\seq{p}}}{\exConstr{c'}{\seq{p}'}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \qedLocal
   \upsigma \disjjoin \upsigma' = \uptau
   &
   \text{(IH) and $\disjjoin$-constr-neq}
   \notag
   \\
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPatt{}{}$-constr-eq}}
         ]
         {
            \compatPatt{\seq{p}}{\seq{p}'}
         }
         {
            \compatPatt{\exConstr{c}{\seq{p}}}{\exConstr{c}{\seq{p}'}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \upkappa := \pattArgsF(\seq{p})(\exApp{\uplambda}{\upsigma_{rest}}),\;\upsigma := c \mapsto \upkappa
   &
   \text{Def of $\pattContF$}
   \notag
   \\
   &
   \upkappa' := \pattArgsF(\seq{p}')(\exApp{\uplambda}{\upsigma_{rest}'}),\;\upsigma' := c \mapsto \upkappa'
   &
   \text{Def of $\pattContF$}
   \notag
   \\
   &
   \upkappa \disjjoin \upkappa' = \upkappa^{\dag}
   &
   \text{(IH)}
   \notag
   \\
   &
   \qedLocal
   \upsigma \disjjoin \upsigma' = c \mapsto \upkappa^{\dag}
   &
   \text{$\disjjoin$-constr-eq}
   \notag
   \\
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPatt{}{}$-record}}
         ]
         {
            \compatPatt{\seq{p}}{\seq{p}'}
         }
         {
            \compatPatt{\exRecord{\seq{\bind{\exVar{x}}{p}}}}{\exRecord{\seq{\bind{\exVar{x}}{p}}'}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \upkappa := \pattArgsF(\seq{p})(\exApp{\uplambda}{\upsigma_{rest}}),\;\upsigma := \elimRec{\seq{x}}{\upkappa}
   &
   \text{Def of \pattContF}
   \notag
   \\
   &
   \upkappa' := \pattArgsF(\seq{p}')(\exApp{\uplambda}{\upsigma_{rest}'}),\;\upsigma' := \elimRec{\seq{x}}{\upkappa'}
   &
   \text{Def of $\pattContF$}
   \notag
   \\
   &
   \upkappa \disjjoin \upkappa' = \upkappa^{\dag}
   &
   \text{(IH)}
   \notag
   \\
   &
   \qedLocal
   \upsigma \disjjoin \upsigma' = \elimRec{\seq{x}}{\upkappa^{\dag}}
   &
   \text{$\disjjoin$-record}
   \notag
\end{flalign}
\end{proof}
