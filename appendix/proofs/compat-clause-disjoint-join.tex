Start with the following assumptions:
\begin{equation}
   \compatPatt{c}{\upmu}
\end{equation}
\begin{equation}
   \locallabel{desugar-left}
   c \desugarFwdS \upsigma
\end{equation}
\begin{equation}
   \locallabel{desugar-right}
   \upmu \desugarFwdS \upsigma'
\end{equation}
We proceed by induction on the derivation of $\compatPattSym$
\begin{proof}
\small
\begin{flalign}
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[]
         {
            \exists i.\; (\compatPatt{p_i}{p_i'})
            \opAnd
            \forall j < i.\; p_j = p'_j
         }
         {
            \compatPatt{(\clause{\seq{p}}{s})}{(\clause{\seq{p}'}{s'})} 
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \text{Have three cases for $\compatPatt{p_i}{p'_i}$}
   &
   \notag
   \\
   \subcase{constr-neq}
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPatt{}{}$-constr-neq}}
         ]
         {
            \exVar{c} \neq \exVar{c'}
         }
         {
            \compatPatt{\exConstr{c}{\seq{p}}}{\exConstr{c'}{\seq{p}'}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \clause{\seq{p}}{s} \desugarFwdS \upsigma
   &
   \text{Assumption \localref{desugar-left}}
   \notag
   \\
   &
   \clause{\seq{p}'}{s'} \desugarFwdS \upsigma'
   &
   \text{Assumption \localref{desugar-right}}
   \notag
   \\
   &
   \qedLocal
   \upsigma \disjjoin \upsigma' = \upsigma \cup \upsigma'
   &
   \text{(IH) and $\disjjoin$-constr-neq}
   \notag
   \\
   \subcase{constr-eq}
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPatt{}{}$-constr-eq}}
         ]
         {
            \compatPatt{\seq{p}}{\seq{p}'}
         }
         {
            \compatPatt{\exConstr{c}{\seq{p}}}{\exConstr{c}{\seq{p}'}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   (\clause{\exConstr{c}{\seq{p}}\cdot\seq{p}'}{s} )\desugarFwdS \elimBind{c}{(\varepsilon \disjjoin (\clause{\seq{p}\cdot\seq{p}'}{s}))}
   &
   \text{$\desugarFwdS$ and assumption \localref{desugar-left}}
   \notag
   \\
   &
   \text{Let $\upsigma = (\varepsilon \disjjoin (\clause{\seq{p}\cdot\seq{p}'}{s}))$}
   &
   \notag
   \\
   &
   (\elimBind{c}{\upsigma}) \disjjoin (\clause{\exConstr{c}{\seq{p}''}\cdot\seq{p}'''}{s'}) = \elimBind{c}{(\upsigma \disjjoin (\clause{\seq{p}''\cdot\seq{p}'''}{s'}) )}
   &
   \text{$\disjjoin$-constr-eq}
   \notag
   \\
   &
   (\clause{\exConstr{c}{\seq{p}''}\cdot\seq{p}'''}{s'} )\desugarFwdS \elimBind{c}{(\varepsilon \disjjoin (\clause{\seq{p}''\cdot\seq{p}'''}{s'}))}
   &
   \text{$\desugarFwdS$ and assumption \localref{desugar-right}}
   \notag
   \\
   &
   \upkappa \disjjoin \upkappa' = \upkappa^{\dag}
   &
   \text{(IH) and $\disjjoin$-lambda}
   \notag
   \\
   &
   \qedLocal
   \upsigma \disjjoin \upsigma' = c \mapsto \upkappa^{\dag}
   &
   \text{$\disjjoin$-constr-eq}
   \notag
   \\
   \subcase{record}
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPatt{}{}$-record}}
         ]
         {
            \compatPatt{\seq{p}}{\seq{p}'}
         }
         {
            \compatPatt{\exRecord{\seq{\bind{\exVar{x}}{p}}}}{\exRecord{\seq{\bind{\exVar{x}}{p}}'}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \upkappa := \elimArgsF(\seq{p})(\exApp{\uplambda}{\upsigma_{\dag}}),\;\upsigma := \elimRec{\seq{x}}{\upkappa}
   &
   \text{Def of \mkElimF}
   \notag
   \\
   &
   \upkappa' := \elimArgsF(\seq{p}')(\exApp{\uplambda}{\upsigma_{\dag}'}),\;\upsigma' := \elimRec{\seq{x}}{\upkappa'}
   &
   \text{Def of $\mkElimF$}
   \notag
   \\
   &
   \upkappa \disjjoin \upkappa' = \upkappa^{\dag}
   &
   \text{(IH) and $\disjjoin$-lambda}
   \notag
   \\
   &
   \qedLocal
   \upsigma \disjjoin \upsigma' = \elimRec{\seq{x}}{\upkappa^{\dag}}
   &
   \text{$\disjjoin$-record}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            right={$\length{\seq{p}} = \length{\seq{p}'} \ge 2$}
         ]
         {
            \exists i. \compatPatt{p_i}{p_i'}
            \opAnd
            \forall j < i.\; p_j = p'_j
         }
         {
            \compatPatt{\clause{\seq{p}}{s}}{\clause{\seq{p}'}{s'}} 
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[lab={\ruleName{$\desugarFwdS$-clause-curried}}, right={$\seq{p} \neq \seqEmpty$}]
         {
            \clause{\seq{p}}{s} \desugarFwdS \upsigma_1
         }
         {
            \clause{p_1 \cons \seq{p}}{s}
            \desugarFwdS
            \mkElimF(p_1)(\exFun{\upsigma})
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \upsigma = \seqRangeOp{p_1}{p_{i-1}}{\elimmapsto} \elimmapsto c_1 \elimmapsto \upsigma_{\dag},\quad p = p_1
   &
   \text{$\compatPattSym$-tail and $c \desugarFwdS \upsigma$}
   \notag
   \\
   &
   \upsigma' = \seqRangeOp{p'_1}{p'_{i-1}}{\elimmapsto} \elimmapsto c_2 \elimmapsto \upsigma'_{\dag},\quad p' = p'_1
   &
   \text{$\compatPattSym$-tail and $c \desugarFwdS \upsigma$}
   \notag
   \\
   &
   \qedLocal
   \upsigma \disjjoin \upsigma' = \seqRangeOp{p_1}{p_{i-1}}{\elimmapsto} \elimmapsto (\set{\elimBind{c_1}{\upsigma_{\dag}}} \cup \set{\elimBind{c_2}{\upsigma'_{\dag}}})
   &
   \disjjoin\text{-no-overlap}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            right={$\length{\upmu}\ge2$}
         ]
         {
            \forall c_i \in \upmu. \; \compatPatt{c}{c_i}
         }
         {
            \compatPatt{c}{\upmu}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   c \desugarFwdS \upsigma
   &
   \text{assumption \localref{desugar-left}}
   \notag
   \\
   &
   \forall c_i \in \upmu.\; \exists \uptau_i.\; \upsigma \disjjoin \upsigma_i = \uptau_i
   &
   \text{premise}
   \\
   &
   \forall c_i,c_j \in \upmu.\; c_i \desugarFwdS \upsigma_i,\; c_k \desugarFwdS \upsigma_j,\; \exists \uptau_{i,j}.\;\upsigma_i \disjjoin \upsigma_j = \uptau_{i,j}
   &
   \text{premise}
   \\
   &
   \upmu \desugarFwdS \upsigma'
   &
   \text{assumption \localref{desugar-right}}
   \notag
   \\
   &
   \qedLocal
   \exists \uptau.\;\upsigma \disjjoin \upsigma' = \uptau
   &
   \thmref{join-preserves-join}
   \notag
\end{flalign}
\end{proof}
