By case analysis and induction on derivations of $\compatPattSym$. For our first pass, we will only consider pairs of clauses. Then, we will generalize this to arbitrary numbers of clauses.
% We define the following partial order on eliminators:
% \begin{equation}
%    \upsigma \elimOrd \upsigma' := 
%    \hfill
%    \quad \upsigma  = \upsigma_{pre} \elimmapsto \upsigma_{rest}
%    \opAnd
%    \upsigma' = \upsigma_{pre} \elimmapsto \upsigma'_{rest}
% \end{equation}
% And we use $\upsigma \sqcap \upsigma'$ to be the greatest lowerbound of $\upsigma,\;\upsigma'$ with respect to this ordering.
\begin{proof}
\small
\begin{flalign}
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPattSym$-head}}
         ]
         {
            \compatPatt{p}{p'}
         }
         {
            \compatPatt{\clause{p\cdot\seq{p_{rest}}}{s}}{p'\cdot\clause{\seq{p_{rest}}'}{s'}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \clause{p\cdot\seq{p_{rest}}}{s}\desugarFwdS \upsigma
   &
   \\
   &
   \clause{p'\cdot\seq{p_{rest}}}{s'}\desugarFwdS \upsigma'
   &
   \\
   &
   \text{$\clause{\seq{p_{rest}}}{s}\desugarFwdS \upsigma_{rest}$}
   &
   \text{(1) and $\desugarFwdS$-clause-curried}
   \\
   &
   \text{$\clause{\seq{p_{rest}}'}{s'}\desugarFwdS \upsigma_{rest}'$}
   &
   \text{(2) and $\desugarFwdS$-clause-curried}
   \\
   &
   \text{Have three cases for $\compatPatt{p}{p'}$}
   &
   \notag
   \\
   \subcase{Constr-neq}
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPatt{}{}$-constr-neq}}
         ]
         {
            \exVar{c} \neq \exVar{c'}
         }
         {
            \compatPatt{\exConstr{c}{\seq{p}}}{\exConstr{c'}{\seq{p}'}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \qedLocal
   \upsigma \disjjoin \upsigma' = \upsigma \cup \upsigma'
   &
   \text{(IH) and $\disjjoin$-constr-neq}
   \notag
   \\
   \subcase{constr-eq}
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPatt{}{}$-constr-eq}}
         ]
         {
            \compatPatt{\seq{p}}{\seq{p}'}
         }
         {
            \compatPatt{\exConstr{c}{\seq{p}}}{\exConstr{c}{\seq{p}'}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \upkappa := \elimArgsF(\seq{p})(\exApp{\uplambda}{\upsigma_{rest}}),\;\upsigma := c \mapsto \upkappa
   &
   \text{Def of $\mkElimF$}
   \notag
   \\
   &
   \upkappa' := \elimArgsF(\seq{p}')(\exApp{\uplambda}{\upsigma_{rest}'}),\;\upsigma' := c \mapsto \upkappa'
   &
   \text{Def of $\mkElimF$}
   \notag
   \\
   &
   \upkappa \disjjoin \upkappa' = \upkappa^{\dag}
   &
   \text{(IH) and $\disjjoin$-lambda}
   \notag
   \\
   &
   \qedLocal
   \upsigma \disjjoin \upsigma' = c \mapsto \upkappa^{\dag}
   &
   \text{$\disjjoin$-constr-eq}
   \notag
   \\
   \subcase{record}
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\compatPatt{}{}$-record}}
         ]
         {
            \compatPatt{\seq{p}}{\seq{p}'}
         }
         {
            \compatPatt{\exRecord{\seq{\bind{\exVar{x}}{p}}}}{\exRecord{\seq{\bind{\exVar{x}}{p}}'}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \upkappa := \elimArgsF(\seq{p})(\exApp{\uplambda}{\upsigma_{rest}}),\;\upsigma := \elimRec{\seq{x}}{\upkappa}
   &
   \text{Def of \mkElimF}
   \notag
   \\
   &
   \upkappa' := \elimArgsF(\seq{p}')(\exApp{\uplambda}{\upsigma_{rest}'}),\;\upsigma' := \elimRec{\seq{x}}{\upkappa'}
   &
   \text{Def of $\mkElimF$}
   \notag
   \\
   &
   \upkappa \disjjoin \upkappa' = \upkappa^{\dag}
   &
   \text{(IH) and $\disjjoin$-lambda}
   \notag
   \\
   &
   \qedLocal
   \upsigma \disjjoin \upsigma' = \elimRec{\seq{x}}{\upkappa^{\dag}}
   &
   \text{$\disjjoin$-record}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            right={$\length{\seq{p}} = \length{\seq{p}'} \ge 2$}
         ]
         {
            \exists i.\; \compatPatt{p_i}{p_i'}
            \;
            \forall j < i.\; p_j = p'_j
         }
         {
            \compatPatt{\clause{\seq{p}}{s}}{\clause{\seq{p}'}{s'}} 
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \upsigma = \seqRangeOp{p_1}{p_{i-1}}{\elimmapsto} \elimmapsto c_1 \elimmapsto \upsigma_{rest},\quad p = p_1
   &
   \text{$\compatPattSym$-tail and $c \desugarFwdS \upsigma$}
   \notag
   \\
   &
   \upsigma' = \seqRangeOp{p'_1}{p'_{i-1}}{\elimmapsto} \elimmapsto c_2 \elimmapsto \upsigma'_{rest},\quad p' = p'_1
   &
   \text{$\compatPattSym$-tail and $c \desugarFwdS \upsigma$}
   \notag
   \\
   &
   \qedLocal
   \upsigma \disjjoin \upsigma' = \seqRangeOp{p_1}{p_{i-1}}{\elimmapsto} \elimmapsto (\set{\elimBind{c_1}{\upsigma_{rest}}} \cup \set{\elimBind{c_2}{\upsigma'_{rest}}})
   &
   \disjjoin\text{-no-overlap}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            right={$\length{\upmu}\ge2$}
         ]
         {
            \forall c_i \in \upmu. \; \compatPatt{c}{c_i}
         }
         {
            \compatPatt{c}{\upmu}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   c := \clause{\seq{p}}{s}
   \quad
   c_i := \clause{\seq{p_i}}{s_i}
   \quad
   \forall i.\; \exists j.\;\forall k < j.\;p_k = p_{i,k} \opAnd \compatPatt{p_j}{p_{i,j} \opAnd i_j = j}  
   &
   \notag
   \\
   &
   \text{Let $j' = \opName{min}_i(i_j)$}
   &
   \\
   &
   \upsigma := \seqRangeOp{p_1}{p_{j'-1}}{\elimmapsto} \elimmapsto \upsigma_{rest}
   &
   \\
   &
   \upsigma' := \seqRangeOp{p_1}{p_{j'-1}}{\elimmapsto} \elimmapsto \upsigma'_{rest}
   &
   \notag
   \\
   &
   \upsigma \disjjoin \upsigma' = \seqRangeOp{p_1}{p_{j'-1}}{\elimmapsto} \elimmapsto (\upsigma \cup \upsigma')
   &
   \disjjoin\text{-no-overlap}
   \notag
\end{flalign}
\end{proof}
