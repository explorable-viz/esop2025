      \begin{smathpar}

            lab={\ruleName{$\matchFwdS$-uncurried}}

            v, \seqRange{\clauseUncurried{p_1}{s_1}}{\clauseUncurried{p_n}{s_n}} \matchFwdR{p_i} \upgamma, s_i
         }
      \end{smathpar}
   }

\end{proof}
