By case analysis on $\matchFwdS$ derivation.
\begin{proof}
\small
\begin{flalign}
   &
   u \desugar v
   &\text{suppose}
   \\
   &
   \upmu \desugar \upsigma
   &\text{suppose}
   \\
   &
   u, \upmu \match \upgamma
   &\text{suppose}
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\matchFwdS$-uncurried}}
         ]
         {
            v, p_i \matchFwdS \upgamma
            \\
            \forall i' \ne i.\nexists \upgamma. \, v, p_i \matchFwdS \upgamma
         }
         {
            v, \seqRange{\clauseUncurried{p_1}{s_1}}{\clauseUncurried{p_j}{s_j}} \matchFwdR{p_i} \upgamma, s_i
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\desugarFwdR$-clauses}}]
         {
            \inferrule*[left={\ruleName{$\desugarFwdS$-clause-uncurried}}, right={$(\forall i \numleq j)$}]
            {
               s_i \desugar e_i
               \\
               \upsigma_i \eqdef \mkElimF(p_i)(e_i)
            }
            {
               (\clause{p_i}{s_i}) \desugar \upsigma_i
            }
         }
         {
            \upmu = (\seqRange{\clause{p_1}{s_1}}{\clause{p_j}{s_j}})
            \desugar
            \seqRangeOp{\upsigma_1}{\upsigma_j}{\disjjoin} = \upsigma
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
\end{flalign}
\end{proof}
