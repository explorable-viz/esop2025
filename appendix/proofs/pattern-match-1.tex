By case analysis on $\matchFwdS$ derivation.
\begin{proof}
\small
\begin{flalign}
   &
   u \desugar v
   &\text{suppose}
   \\
   &
   \upmu \desugar \upsigma
   &\text{suppose}
   \\
   &
   u, \upmu \match \upgamma
   &\text{suppose}
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\matchFwdS$-uncurried}}
         ]
         {
            u, p_i \matchFwdS \upgamma
            \\
            \forall i' \ne i.\nexists \upgamma. \, v, p_i \matchFwdS \upgamma
         }
         {
            u, \seqRange{\clauseUncurried{p_1}{s_1}}{\clauseUncurried{p_j}{s_j}} \matchFwdR{p_i} \upgamma, s_i
         }
      \end{smathpar}
   }
   &
   \locallabel{match}
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\desugar$-clauses}}]
         {
            \inferrule*[left={\ruleName{$\desugar$-clause-uncurried}}, right={$(\forall i \numleq j)$}]
            {
               s_i \desugar e_i
               \\
               \upsigma_i \eqdef \mkElimF(p_i)(e_i)
            }
            {
               (\clause{p_i}{s_i}) \desugar \upsigma_i
            }
         }
         {
            \upmu = (\seqRange{\clause{p_1}{s_1}}{\clause{p_j}{s_j}})
            \desugar
            \seqRangeOp{\upsigma_1}{\upsigma_j}{\disjjoin} = \upsigma
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \locallabel{desugar}
   \\
   &
   v, \upsigma_i \match \upgamma', e_i \wedge \upgamma \desugar \upgamma'
   &
   \text{Def.~$\upsigma_i$, \localref{match}, \lemref{match-domains-agree}}
   \locallabel{gamma}
   \\
   &
   \forall i' \neq i.v, \upsigma_i'\;\matchNeg
   &
   \text{\localref{desugar}, \lemref{match-domains-agree}}
   \locallabel{no-match}
   \\
   &
   \qedLocal v, \upsigma \match \upgamma', e_i \wedge \upgamma \desugar \upgamma'
   &
   \text{\localref{gamma}, \localref{no-match}, \lemref{match-increasing}}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\match$-curried}},
            right={$\upmu' \neq \seqEmpty$}
         ]
         {
            u, p \match \upgamma
            \\
            (\forall i \numleq j.\text{if }p_i = p\text{ then }(\clause{\seq{p_i}'}{s_i}) \in \upmu'\text{ else }v, p_i\;\matchNeg)
         }
         {
            u, \upmu = \seqRange{((p_1 \cons \clause{\seq{p_1}')}{s_1})}{((p_j \cons \clause{\seq{p_j}')}{s_j})}
            \match
            \upgamma, \exLambda{\upmu'} = \upsigma
         }
      \end{smathpar}
   }
   &
   \notag
\end{flalign}
\end{proof}
