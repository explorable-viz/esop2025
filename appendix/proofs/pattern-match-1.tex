By case analysis on $\matchFwdS$ derivation.
\begin{proof}
\small
\begin{flalign}
   &
   u \desugar v
   &\text{suppose}
   \\
   &
   \upmu \desugar \upsigma
   &\text{suppose}
   \\
   &
   u, \upmu \match \upgamma
   &\text{suppose}
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\matchFwdS$-uncurried}}
         ]
         {
            v, p_i \matchFwdS \upgamma
            \\
            \forall i' \ne i.\nexists \upgamma. \, v, p_i \matchFwdS \upgamma
         }
         {
            v, \seqRange{\clauseUncurried{p_1}{s_1}}{\clauseUncurried{p_j}{s_j}} \matchFwdR{p_i} \upgamma, s_i
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[lab={\ruleName{$\desugarFwdR$-clauses}}]
         {
            c_i \desugarFwdS \upsigma_i
            \quad
            (\forall i \numleq j)
         }
         {
            \upmu = \seqRange{c_1}{c_j} \desugarFwdS \seqRangeOp{\upsigma_1}{\upsigma_j}{\disjjoin} = \upsigma
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
\end{flalign}
\end{proof}
