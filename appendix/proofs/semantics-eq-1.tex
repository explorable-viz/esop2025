By induction on derivations of $\evalSugS$. We use the term \textit{inversion on} a rule, to mean that we have established the conclusion to a judgement,
and we are therefore able to establish the premises of that rule. This is generally possible as we have fixed our inputs to the non-deterministic judgements, such as desugaring. \newline
Assume: 
\begin{equation}
   \upGamma \vdash \bind{s}{A}
\end{equation}
\begin{equation}
   \upgamma : \upGamma
\end{equation}   
\begin{equation}
   s \desugarFwdS e
\end{equation}
\begin{equation}
   \locallabel{desug:env}
   \upgamma \desugarFwdS \upgamma'
\end{equation}
\begin{proof}
\small
\begin{flalign}
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%   
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-var}}
         ]
         {
            \strut
         }
         {
            {\upgamma^{\dag}} \cons (\bind{x}{v}), \exVar{x} \evalSugS v
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugarFwdS$-var}}
         ]
         {
            \strut
         }
         {
            \exVar{x} \desugarFwdS \exVar{x}
         }
      \end{smathpar}
   }
   &
   \text{Inversion on $\desugarFwdS$}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-var}}
         ]
         {
            \strut
         }
         {
            {{\upgamma}^{\dag}}' \cdot (\bind{\exVar{x}}{v'}), \exVar{x} \evalFwdS v'
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\evalFwdS$}
   \notag
   \\
   &
   \qedLocal
   \text{$v \desugarFwdS v'$.}
   &
   \text{\lemref{lemma:varlookup} and assumption \localref{desug:env}}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-constr}}
         ]
         {
            \upgamma, \seq{s} \evalSugS \seq{v}
         }
         {
            \upgamma, \exConstr{c}{\seq{s}} \evalSugS \exConstr{c}{\seq{v}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugarFwdS$-constr}}
         ]
         {
            \seq{s} \desugarFwdS \seq{e}
         }
         {
            \exConstr{c}{\seq{s}} \desugarFwdS \exConstr{c}{\seq{e}}
         }
      \end{smathpar}
   }
   &
   \text{Inversion of $\desugarFwdS$}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-constr}}
         ]
         {
            \upgamma', \seq{e} \evalFwdS \seq{v}'
         }
         {
            \upgamma', \exConstr{c}{\seq{e}} \evalFwdS \exConstr{c}{\seq{v}'}
         }
      \end{smathpar}
   }
   &
   \text{Inversion on $\evalFwdS$}
   \notag
   \\
   &
   \text{$\seq{v} \desugarFwdS \seq{v}'$}
   &
   \text{IH}
   \notag
   \\
   &
   \qedLocal
   \text{$\exConstr{c}{\seq{v}} \desugarFwdS \exConstr{c}{\seq{v}'}$}
   &
   \notag
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-app}},
            width=3in,
         ]
         {
            \upgamma, (s, s') \evalSugR{(S,S')} (\exClosure{\upgamma_1}{g}{\upmu}, v)
            \\
            \upgamma_1, g \closeDefsS \upgamma_2
            \\
            v, \upmu \matchFwdR{p} \upgamma_3, s^\dag
            \\
            \upgamma_1 \concat \upgamma_2 \concat \upgamma_3, s^\dag \evalSugR{S^\dag} v^{\dag}
         }
         {
            \upgamma, \exApp{s}{s'} \evalSugR{\trApp{S}{S'}{p}{S^{\dag}}} v^{\dag}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugarFwdS$-app}}
         ]
         {
            s \desugarFwdS e
            \\
            s' \desugarFwdS e'
         }
         {
            \exApp{s}{s'} \desugarFwdS \exApp{e}{e'}
         }
      \end{smathpar}
   }
   &
   \text{Inversion on $\desugarFwdS$}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-apply}},
            width=3.3in,
         ]
         {
            \upgamma, (e, e') \evalS (\exClosure{\upgamma_1}{\uprho}{\upsigma}, v')
            \\
            \upgamma'_1, \uprho \closeDefsFwdR \upgamma'_2
            \\
            v', \upsigma \matchFwdS \upgamma'_3, e^{\dag}
            \\
            \upgamma'_1 \concat \upgamma'_2 \concat \upgamma'_3, e^{\dag} \evalFwdS u
         }
         {
            \upgamma, \exApp{e}{e'}
            \evalFwdS
            u
         }
      \end{smathpar}
   }
   &
   \text{Inversion on $\evalFwdS$-app}
   \notag
   \\
   &
   \exClosure{\upgamma_1}{g}{\upmu} \desugarFwdS \exClosure{\upgamma'_1}{\uprho}{\upsigma}
   &
   \text{IH}
   \\
   &
   v \desugarFwdS v'
   &
   \text{IH}
   \notag
   \\
   &
   \upgamma_2 \desugarFwdS \upgamma'_2
   &
   \text{\lemref{closedefs-eq}}
   \notag
   \\
   &
   \upgamma_3 \desugarFwdS \upgamma'_3 \opAnd s^{\dag} \desugarFwdS e^{\dag}
   &
   \text{$\upmu \desugarFwdS \upsigma$ and \lemref{matching-bi}}
   \notag
   \\
   &
   v^{\dag} \desugarFwdS u
   &
   \text{IH}
   \notag
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-let-rec}}
         ]
         {
            \upgamma, g \closeDefsS \upgamma^{\dag}
            \\
            \upgamma \concat \upgamma^{\dag}, s \evalSugS v
         }
         {
            \upgamma, \exLetRecPiecewise{g}{s} \evalSugS v
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[lab={\ruleName{$\desugarFwdS$-let-rec}}]
         {
            \upmu_i \desugarFwdR \sigma_i
            \quad
            (\forall i \numleq \length{\seq{x}})
            \\
            s \desugarFwdR e
         }
         {
            \exLetRecPiecewise{\seq{\bind{x}{\upmu}}}{s}
            \desugarFwdR
            \exLetRecPiecewise{\seq{\bind{x}{\sigma}}}{e}
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\desugarFwdS$-let-rec}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[lab={\ruleName{$\evalFwdS$-let-rec}}]
         {
            \upgamma', \seq{\bind{\exVar{x}}{\upsigma}} \closeDefsS {\upgamma^{\dag}}'
            \\
            \upgamma' \concat {\upgamma^{\dag}}', e \evalFwdS v'
         }
         {
            \upgamma', \exLetRecPiecewise{\seq{\bind{\exVar{x}}{\upsigma}}}{e} \evalFwdS v'
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\evalFwdS$-let-rec}
   \notag
   \\
   &
   \text{Then $\upgamma \concat \upgamma^{\dag} \desugarFwdS \upgamma' \concat {\upgamma^{\dag}}'$}
   &
   \text{assumption \localref{desug:env} and \lemref{closedefs-eq}}
   \notag
   \\
   &
   \qedLocal
   \text{Use these environments to establish that $v \desugarFwdS v'$}
   &
   \text{IH}
   \notag
\end{flalign}
\end{proof}
