By induction on derivations of $\evalSugS$. \newline
Assume: 
\begin{equation}
   \Gamma \vdash \bind{s}{A}
\end{equation}
\begin{equation}
   \upgamma : \Gamma
\end{equation}   
\begin{equation}
   s \desugarFwdS e
\end{equation}
\begin{equation}
   \locallabel{desug:env}
   \upgamma \desugarFwdS \upgamma'
\end{equation}
\begin{proof}
\small
\begin{flalign}
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%   
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-var}}
         ]
         {
            \strut
         }
         {
            {\upgamma^{\dag}} \cons (\bind{x}{v}), \exVar{x} \evalSugS v
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugarFwdS$-var}}
         ]
         {
            \strut
         }
         {
            \exVar{x} \desugarFwdS \exVar{x}
         }
      \end{smathpar}
   }
   &
   \text{Inversion on $\desugarFwdS$}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-var}}
         ]
         {
            \strut
         }
         {
            {{\upgamma}^{\dag}}' \cdot (\bind{\exVar{x}}{v'}), \exVar{x} \evalFwdS v'
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\evalFwdS$}
   \notag
   \\
   &
   \qedLocal
   \text{$v \desugarFwdS v'$.}
   &
   \text{\lemref{lemma:varlookup} and assumption \localref{desug:env}}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseName{\thmref{semantics-eq}-constr}
   &
   \notag
   \\
   &
   \defProofVar{s}{\exConstr{c}{\seq{s}'}}, \quad
   \defProofVar{e}{\exConstr{c}{\seq{e}'}}\text{, so $s \desugarFwdS e$ if $\seq{s}' \desugarFwdS \seq{e}'$}, \quad
   \upgamma \desugarFwdS \upgamma'
   &
   \notag
   \\
   &
   \subcaseName{surface-then-core}
   &
   \notag
   \\
   &
   \text{$\upgamma, s \evalSugS v \; \implies \; \defProofVar{v}{\exConstr{c}{\seq{v}}}$. }
   &
   \text{inversion of $\evalFwdS$-constr}
   \notag
   \\
   &
   \text{Then, we get that $\upgamma', e \evalFwdS \exConstr{c}{\seq{v}'}$}
   &
   \text{$\desugarFwdS$-constr and $\evalFwdS$-constr}
   \notag
   \\
   &
   \text{We already have that $\seq{v} \desugarFwdS \seq{v}'$}
   &
   \text{IH}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseName{\thmref{semantics-eq}-app}
   &
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseName{\thmref{semantics-eq}-let-rec}
   &
   \notag
   \\
   &
   \text{$\defProofVar{s}{\exLetRecPiecewise{\seq{g}}{s'}}$, which means $\defProofVar{e}{\exLetRec{\seq{\bind{x}{\upsigma}}}{e'}}$. For this case, fix $\upgamma \desugarFwdS \upgamma^{\dag}$} 
   &
   \text{$\desugarFwdS$-let-rec}
   \notag
   \\
   &
   \subcaseName{surface-then-core}
   &
   \notag
   \\
   &
   \text{$\upgamma, s \evalSugS v$ means $\upgamma, \seq{g} \closeDefsS \upgamma'$ and $\upgamma \concat \upgamma', s' \evalSugS v$}
   &
   \text{inversion of $\evalSugS$-let-rec}
   \notag
   \\
   &
   \text{${\upgamma^{\dag}}, \exLetRec{\seq{\bind{\exVar{x}}{\upsigma}}}{e'} \evalFwdS v'$ means $\upgamma', \seq{\bind{\exVar{x}}{\upsigma}} \closeDefsS {\upgamma^{\dag}}'$ and ${\upgamma^{\dag}} \concat {\upgamma^{\dag}}', e' \evalFwdS v'$}
   &
   \text{inversion of $\evalFwdS$-let-rec}
   \notag
   \\
   &
   \text{We then have that $\upgamma' \desugarFwdS {\upgamma^{\dag}}'$ and thus that $\upgamma \concat \upgamma' \desugarFwdS {\upgamma^{\dag}} \concat {\upgamma^{\dag}}'$}
   &
   \text{\lemref{closedefs-eq}}
   \notag
   \\
   &
   \text{We use the concatenated environments and $s'$, $e'$, to get that $v \desugarFwdS v'$}
   &
   \text{IH}
   \notag
\end{flalign}
\end{proof}
