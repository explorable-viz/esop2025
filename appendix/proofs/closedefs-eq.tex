Assume the following:
\begin{equation}
   \upgamma_1 \desugarFwdS \upgamma_2
\end{equation}
\begin{equation}
   g \desugarFwdS \uprho
\end{equation}
\begin{proof}
   \begin{flalign}
      \intertext{\crossrule}
      %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
      &
      \subcaseName{$\closeDefsS$-then-$\desugarFwdS$}
      &
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*
            {
               g = \seq{\bind{x}{\mu}}
               \\
               v_i = \exClosure{\upgamma_1}{g}{\upmu_i}
               \quad
               \forall i \le \length{\seq{x}}
            }
            {
               \upgamma_1, g \closeDefsS \set{\seq{\bind{x}{v}}}
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[]
            {
               \uprho = \set{\seq{\bind{x}{\upsigma}}}
               \\
               v'_i = \exClosureNew{\upgamma_2}{\uprho}{\upsigma_i}
               \quad
               (\forall i \numleq \length{\seq{x}})
            }
            {
               \upgamma_2, \uprho
               \closeDefsFwdR
               \set{\seq{\bind{x}{v'}}}
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \forall i \numleq \length{\seq{x}} \; v_i \desugarFwdS v'_i
      &
      \text{$g \desugarFwdS \uprho \opAnd \upgamma_1 \desugarFwdS \upgamma_2$}
      \notag
      \\
      &
      \qedLocal
      \upgamma_1' \desugarFwdS \upgamma_2'
      &
      \text{$\desugarFwdS$-env}
      \notag
      \\
      &
      \subcaseName{$\desugarFwdS$-then-$\closeDefsS$}
      &
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[]
            {
               \uprho = \set{\seq{\bind{x}{\upsigma}}}
               \\
               v'_i = \exClosureNew{\upgamma_2}{\uprho}{\upsigma_i}
               \quad
               (\forall i \numleq \length{\seq{x}})
            }
            {
               \upgamma_2, \uprho
               \closeDefsFwdR
               \set{\seq{\bind{x}{v'}}}
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*
            {
               g = \seq{\bind{x}{\mu}}
               \\
               v_i = \exClosure{\upgamma_1}{g}{\upmu_i}
               \quad
               \forall i \le \length{\seq{x}}
            }
            {
               \upgamma_1, g \closeDefsS \set{\seq{\bind{x}{v}}}
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \forall i\numleq \length{\seq{x}}\; v_i \desugarFwdS v'_i
      &
      \text{$g \desugarFwdS \uprho \opAnd \upgamma_1 \desugarFwdS \upgamma_2$}
      \notag
      \\
      &
      \qedLocal
      \upgamma_1' \desugarFwdS \upgamma_2'
      &
      \text{$\desugarFwdS$-env}
      \notag
   \end{flalign}
\end{proof}
