By induction on derivations of $\evalFwdS$\newline
Assume: 
\begin{equation}
   \Gamma \vdash \bind{s}{A}
\end{equation}
\begin{equation}
   \upgamma : \Gamma
\end{equation}   
\begin{equation}
   s \desugarFwdS e
\end{equation}
\begin{equation}
   \locallabel{desug:env}
   \upgamma \desugarFwdS \upgamma'
\end{equation}
\begin{proof}
\small
\begin{flalign}
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%   
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-var}}
         ]
         {
            \strut
         }
         {
            {\upgamma^{\dag}}' \cdot (\bind{x}{v'}), x \evalFwdS v'
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugarFwdS$-var}}
         ]
         {
            \strut
         }
         {
            \exVar{x} \desugarFwdS \exVar{x}
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\desugarFwdS$-var}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-var}}
         ]
         {
            \strut
         }
         {
            {\upgamma^{\dag}} \cons (\bind{x}{v}), \exVar{x} \evalSugS v
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\evalSugS$-var}
   \notag
   \\
   &
   \qedLocal
   v \desugarFwdS v'
   &
   \text{assumption \localref{desug:env} and \lemref{lemma:varlookup}}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-constr}}
         ]
         {
            \upgamma', \seq{e} \evalFwdS \seq{v}'
         }
         {
            \upgamma', \exConstr{c}{\seq{e}} \evalFwdS \exConstr{c}{\seq{v}'}
         }         
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugarFwdS$-constr}}
         ]
         {
            \seq{s} \desugarFwdS \seq{e}
         }
         {
            \exConstr{c}{\seq{s}} \desugarFwdS \exConstr{c}{\seq{e}}
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\desugarFwdS$}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-constr}}
         ]
         {
            \upgamma, \seq{s} \evalSugS \seq{v}
         }
         {
            \upgamma, \exConstr{c}{\seq{s}} \evalSugS \exConstr{c}{\seq{v}}
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\evalSugS$-constr}
   \notag
   \\
   &
   \qedLocal
   \text{$\upgamma, \seq{s}' \evalSugS \seq{v}^{\dag}$, and $\upgamma', \seq{e}' \evalFwdS \seq{v}'$, with $\seq{v}^{\dag} \desugarFwdS \seq{v}'$}
   &
   \text{IH}
   \notag
   \\
   &
   \text{This gives us $\upgamma, s \evalSugS \exConstr{c}{\seq{v}^{\dag}}$}
   &
   \text{$\evalSugS$-constr}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseName{\thmref{semantics-eq}-app}
   &
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseName{\thmref{semantics-eq}-let-rec}
   &
   \notag
   \\
   &
   \text{$\defProofVar{s}{\exLetRecPiecewise{\seq{g}}{s'}}$, which means $\defProofVar{e}{\exLetRec{\seq{\bind{x}{\upsigma}}}{e'}}$. For this case, fix $\upgamma \desugarFwdS \upgamma^{\dag}$} 
   &
   \text{$\desugarFwdS$-let-rec}
   \notag
   \\
   &
   \subcaseName{core-then-surface}
   &
   \notag
   \\
   &
   \text{${\upgamma^{\dag}}, \exLetRec{\seq{\bind{\exVar{x}}{\upsigma}}}{e'} \evalFwdS v'$, then $\upgamma^{\dag}, \seq{\bind{\exVar{x}}{\upsigma}} \closeDefsS {{\upgamma}^\dag}'$ and ${\upgamma^{\dag}} \concat {{\upgamma}^\dag}', e' \evalFwdS v'$}
   &
   \text{inversion of $\evalFwdS$-let-rec}
   \notag
   \\
   &
   \text{$\upgamma, \exLetRec{\seq{g}}{s'} \evalSugS v$, with $\upgamma, \seq{g} \closeDefsS \upgamma'$ and $\upgamma \concat \upgamma', s' \evalSugS v$}
   &
   \text{inversion of $\evalSugS$-let-rec}
   \notag
   \\
   &
   \text{We can conlude that $\upgamma' \desugarFwdS {{\upgamma}^\dag}'$}
   &
   \text{\lemref{closedefs-eq}}
   \notag
   \\
   &
   \text{We can thus conclude that $\upgamma \concat \upgamma' \desugarFwdS {\upgamma^{\dag}} \concat {{\upgamma}^\dag}'$}
   &
   \notag
   \\
   &
   \text{Then we use these concatenated environments to conclude that $v \desugarFwdS v'$}
   &
   \text{IH}
   \notag
\end{flalign}
\end{proof}
