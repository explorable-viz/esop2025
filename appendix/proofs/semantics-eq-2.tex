By induction on derivations of $\evalFwdS$ We use the term \textit{inversion on} a rule, to mean that we have established the conclusion to a judgement,
and we are therefore able to establish the premises of that rule. This is generally possible as we have fixed our inputs to the non-deterministic judgements, such as desugaring. \newline
Assume:
\begin{equation}
   \upGamma \vdash \bind{s}{A}
\end{equation}
\begin{equation}
   \upgamma : \upGamma
\end{equation}
\begin{equation}
   s \desugarFwdS e
\end{equation}
\begin{equation}
   \locallabel{desug:env}
   \upgamma \desugarFwdS \upgamma'
\end{equation}
\begin{proof}
\small
\begin{flalign}
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-var}}
         ]
         {
            \strut
         }
         {
            {\upgamma^{\dag}}' \cdot (\bind{x}{v'}), x \evalFwdS v'
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugarFwdS$-var}}
         ]
         {
            \strut
         }
         {
            \exVar{x} \desugarFwdS \exVar{x}
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\desugarFwdS$-var}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-var}}
         ]
         {
            \strut
         }
         {
            {\upgamma^{\dag}} \cons (\bind{x}{v}), \exVar{x} \evalSugS v
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\evalSugS$-var}
   \notag
   \\
   &
   \qedLocal
   v \desugarFwdS v'
   &
   \text{assumption \localref{desug:env} and \lemref{lemma:varlookup}}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-constr}}
         ]
         {
            \upgamma', \seq{e} \evalFwdS \seq{v}'
         }
         {
            \upgamma', \exConstr{c}{\seq{e}} \evalFwdS \exConstr{c}{\seq{v}'}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugarFwdS$-constr}}
         ]
         {
            \seq{s} \desugarFwdS \seq{e}
         }
         {
            \exConstr{c}{\seq{s}} \desugarFwdS \exConstr{c}{\seq{e}}
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\desugarFwdS$}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-constr}}
         ]
         {
            \upgamma, \seq{s} \evalSugS \seq{v}
         }
         {
            \upgamma, \exConstr{c}{\seq{s}} \evalSugS \exConstr{c}{\seq{v}}
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\evalSugS$-constr}
   \notag
   \\
   &
   \qedLocal
   \text{$\upgamma, \seq{s}' \evalSugS \seq{v}^{\dag}$, and $\upgamma', \seq{e}' \evalFwdS \seq{v}'$, with $\seq{v}^{\dag} \desugarFwdS \seq{v}'$}
   &
   \text{IH}
   \notag
   \\
   &
   \text{This gives us $\upgamma, s \evalSugS \exConstr{c}{\seq{v}^{\dag}}$}
   &
   \text{$\evalSugS$-constr}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-apply}},
            width=3.3in,
         ]
         {
            \upgamma, (e, e') \evalS (\exClosure{\upgamma'_1}{\uprho}{\upsigma}, v')
            \\
            \upgamma'_1, \uprho \closeDefsFwdR \upgamma'_2
            \\
            v', \upsigma \matchFwdS \upgamma_3, e^{\dag}
            \\
            \upgamma'_1 \concat \upgamma'_2 \concat \upgamma'_3, e^{\dag} \evalFwdS u
         }
         {
            \upgamma', \exApp{e}{e'}
            \evalFwdS
            u
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugarFwdS$-app}}
         ]
         {
            s \desugarFwdS e
            \\
            s' \desugarFwdS e'
         }
         {
            \exApp{s}{s'} \desugarFwdS \exApp{e'}{}
         }
      \end{smathpar}
   }
   &
   \text{Inversion on $\desugarFwdS$}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-app}},
            width=3in,
         ]
         {
            \upgamma, (s, s') \evalSugR{(S,S')} (\exClosure{\upgamma_1}{g}{\upmu}, v)
            \\
            \upgamma_1, g \closeDefsS \upgamma_2
            \\
            v, \upmu \matchFwdR{p} \upgamma_3, s^\dag
            \\
            \upgamma_1 \concat \upgamma_2 \concat \upgamma_3, s^\dag \evalSugR{S^\dag} v^{\dag}
         }
         {
            \upgamma, \exApp{s}{s'} \evalSugR{\trApp{S}{S'}{p}{S^{\dag}}} v^{\dag}
         }
      \end{smathpar}
   }
   &
   \text{Inversion on $\evalSugS$-app}
   \notag
   \\
   &
   \exClosure{\upgamma_1}{g}{\upmu} \desugarFwdS \exClosure{\upgamma'_1}{\uprho}{\upsigma}
   &
   \text{IH}
   \notag
   \\
   &
   v \desugarFwdS v'
   &
   \text{IH}
   \notag
   \\
   &
   \upgamma_2 \desugarFwdS  \upgamma'_2
   &
   \text{\lemref{closedefs-eq}}
   \notag
   \\
   &
   \upgamma_3 \desugarFwdS \upgamma'_3 \desugarFwdS \opAnd s^{\dag} \desugarFwdS e^{\dag}
   &
   \text{$\upmu \desugarFwdS \upsigma$ and \lemref{matching-bi}}
   \notag
   \\
   &
   \qedLocal
   v^{\dag} \desugarFwdS u
   &
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-let-rec}}
         ]
         {
            \upgamma, g \closeDefsS \upgamma'
            \\
            \upgamma \concat \upgamma', s \evalSugR{S} v
         }
         {
            \upgamma, \exLetRecPiecewise{g}{s} \evalSugR{\trLetRec{g}{S}} v
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[lab={\ruleName{$\desugarFwdR$-let-rec}}]
         {
            \upmu_i \desugarFwdR \upsigma_i
            \quad
            (\forall i \numleq \length{\seq{x}})
            \\
            s \desugarFwdR e
         }
         {
            \exLetRecPiecewise{\seq{\bind{x}{\upmu}}}{s}
            \desugarFwdR
            \exLetRecPiecewise{\seq{\bind{x}{\upsigma}}}{e}
         }
      \end{smathpar}
   }
   &
   \text{$\desugarFwdS$-let-rec}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-let-rec}}
         ]
         {
            {\upgamma^{\dag}}, \set{\seq{\bind{x}{\upsigma}}} \closeDefsFwdR {\upgamma^{\dag}}'
            \\
            {\upgamma^{\dag}} \concat {\upgamma^{\dag}}', e \evalFwdS v
         }
         {
            {\upgamma^{\dag}}, \exLetRec{\seq{\bind{x}{\upsigma}}}{e}
            \evalFwdS
            v
         }
      \end{smathpar}
   }
   &
   \text{inversion of $\evalFwdS$-let-rec}
   \notag
   \\
   &
   \text{Then we conclude that $\upgamma \concat \upgamma' \desugarFwdS {\upgamma^{\dag}} \concat {\upgamma^{\dag}}'$}
   &
   \text{assumption \localref{desug:env} and \lemref{closedefs-eq}}
   \notag
   \\
   &
   \qedLocal
   v \desugarFwdS v'
   &
   \text{IH}
   \notag
\end{flalign}
\end{proof}
