      }

                  {(\seqEmpty, \clause{p'_{j'} \cons \pi'_{j'}}{s'_{j'}})}
         \desugarBwd{
            \seqRange{(\seqEmpty, \clause{p_1 \cons \pi_1}{s_1})}

         \exFun{\sigma}

      }
      %

         lab={\ruleName{$\desugarBwd{}$-clauses-record}}

            (\seqRange{\pattRecord{\seq{\bind{x}{p_1}}}}{\pattRecord{\seq{\bind{x}{p_j}}}})
            \clauseWith{\cons}
            \seq{k}
         }
