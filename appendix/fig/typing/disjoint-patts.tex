         lab={\ruleName{$\compatPatt{}{}$-record}}

         \forall i \le \length{\seq{\exVar{x}}}
      }
      {
         \compatPatt{\exRecord{\seq{\bind{\exVar{x}}{p}}}}{\exRecord{\seq{\bind{\exVar{x}}{p}}'}}
      }
   \end{smathpar}
