      %========================================
      \PattSeqOpt &:: \Cxt \to \Set{Set}
      \\
      \PattSeqOpt\;\Gamma &\eqdef 1 + \exists A.\PattSeq\;A\;\Gamma
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\mkElimF :: \Patt\;A\;\Gamma' \to (\Cont\;(\Gamma \concat \Gamma')\;B \to \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \mkElimF
      &::
      \Patt\;A\;\Gamma' \to \Cont\;(\Gamma \concat \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \mkElimF(\pattVar{x})(\kappa)
      &=
      \elimVar{x}{\kappa}
      \\
      \mkElimF(\pattConstr{c}{\seq{p}})(\kappa)
      &=
      \elimConstr{\elimBind{c}{\elimArgsF(\seq{p})(\kappa)}}
      \\
      \mkElimF(\pattList{p}{o})(\kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\elimArgsF(p \cons o)(\kappa)}}
      \\
      \mkElimF(\pattRecord{\seq{\bind{x}{p}}})(\kappa)
      &=
      \elimRecord{\seq{x}}{\elimArgsF(\seq{p})(\kappa)}
      \\
      %========================================
      \intertext{
      \flushleft\shadebox{$\mkElimF :: \ListRestPatt\;A\;\Gamma' \to (\Cont\;(\Gamma \concat \Gamma')\;B \to \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \mkElimF
      &::
      \ListRestPatt\;A\;\Gamma' \to \Cont\;(\Gamma \concat \Gamma')\;B \to \Elim\;\Gamma\;A\;B
      \\
      %========================================
      \mkElimF(\pattListEnd)(\kappa)
      &=
      \elimConstr{\elimBind{\cNil}{\kappa}}
      \\
      \mkElimF(\pattListNext{p}{o})(\kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\elimArgsF(p \cons o)(\kappa)}}
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\elimArgsF :: \PattSeqOpt\;\Gamma' \to (\Cont\;(\Gamma \concat \Gamma')\;A \to \Cont\;\Gamma\;A)$}
      }
      %========================================
      \elimArgsF
      &::
      \PattSeqOpt\;\Gamma' \to \Cont\;(\Gamma \concat \Gamma')\;A \to \Cont\;\Gamma\;A
      \\
      %========================================
      \elimArgsF(\seqEmpty)(\kappa)
      &=
      \kappa
      \\
      \elimArgsF(\pi \cons \seq{\pi})(\sigma)
      &=
      \mkElimF(\pi)(\elimArgsF(\seq{\pi})(\sigma))
      %========================================
      \intertext{
      \flushleft\shadebox{$\orElseF :: \PattSeqOpt\;\Gamma' \to (\Cont\;(\Gamma \concat \Gamma')\;A \times \Expr\;\cxtEmpty\;A \to \Cont\;(\Gamma \concat \Gamma')\;A)$}
      }
      %========================================
      \begin{split}
      \orElseF
      &::
      \PattSeqOpt\;\Gamma' \to\\
      &\phantom{aj.}
      \Cont\;(\Gamma \concat \Gamma')\;A \times \Expr\;\cxtEmpty\;A \to \Cont\;(\Gamma \concat \Gamma')\;A
      \end{split}
      \\
      %========================================
      \orElseF_{\seqEmpty}(e, e')
      &=
      e
      \\
      \orElseF_{\pattVar{x} \cons \seq{\pi}}(\elimVar{x}{\kappa}, e)
      &=
      \elimVar{x}{\orElseF(\seq{\pi})(\kappa, e)}
      \\
      \orElseF_{\pattRecord{\seq{\bind{x}{p}}} \cons \seq{\pi}}(\elimRecord{\seq{x}}{\kappa}, e)
      &=
      \elimRecord{\seq{x}}{\orElseF_{\seq{p} \concat \seq{\pi}}(\kappa, e)}
      \\
      \orElseF_{\pattConstr{c}{\seq{p}} \cons \seq{\pi}}(\elimConstr{\elimBind{c}{\kappa}}, e)
      &=
      \elimConstr{\elimBind{c}{\orElseF_{\seq{p} \concat \seq{\pi}}(\kappa, e)}} \disjjoin \unlessF_{c}(e)
      \\
      \orElseF_{\pattList{p}{o} \cons \seq{\pi}}(\elimConstr{\elimBind{\cCons}{\sigma}}, e)
      &=
      \elimConstr{\elimBind{\cCons}{\orElseF_{p \cons o \cons \seq{\pi}}(\sigma)}} \disjjoin \unlessF_{\cCons}(e)
      \\
      \orElseF_{\pattListNext{p}{o} \cons \seq{\pi}}(\elimConstr{\elimBind{\cCons}{\sigma}}, e)
      &=
      \elimConstr{\elimBind{\cCons}{\orElseF_{p \cons o \cons \seq{\pi}}(\sigma)}} \disjjoin \unlessF_{\cCons}(e)
      \\
      \orElseF_{\pattListEnd \cons \seq{\pi}}(\elimConstr{\elimBind{\cNil}{\kappa}}, e)
      &=
      \elimConstr{\elimBind{\cNil}{\orElseF_{\seq{\pi}}(\kappa)}} \disjjoin \unlessF_{\cNil}(e)
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\unlessF :: \Pi(c: \Ctr).\Expr\;\seqEmpty\;A \to \Elim\;\Gamma\;\datatype{\Sigma}{c}\;A$}
      }
      %========================================
      \unlessF
      &:: \Pi(c: \Ctr).\Expr\;\seqEmpty\;A \to \Elim\;\Gamma\;\datatype{\Sigma}{c}\;A
      \\
      %========================================
      \unlessF_{c}(e)
      &=
      \elimConstr{\elimBind{c'}{\iterateF(\length{\seq{B}}, \lambda\kappa.\elimVar{x}{\kappa})(e)} \mid \bind{c'}{\seq{B}} \in \datatype{\Sigma}{c} \setminus \set{c}}
      \\
