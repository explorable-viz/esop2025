      %========================================
      \PattSeqOpt &:: \Cxt \to \Set{Set}
      \\
      \PattSeqOpt\;\Gamma &\eqdef 1 + \exists A.\PattSeq\;A\;\Gamma
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\elimF :: \Patt\;A\;\Gamma' \times (\Cont\;(\Gamma \concat \Gamma')\;B \to \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \elimF(\pattVar{x})(\kappa)
      &=
      \elimVar{x}{\kappa}
      \\
      \elimF(\pattConstr{c}{\seq{p}}, \kappa)
      &=
      \elimConstr{\elimBind{c}{\elimArgsF(\seq{p}, \kappa)}}
      \\
      \elimF(\pattList{p}{o}, \kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\elimArgsF(p \cons o, \kappa)}}
      \\
      \elimF(\pattRecord{\seq{\bind{x}{p}}}, \kappa)
      &=
      \elimRecord{\seq{x}}{\elimArgsF(\seq{p}, \kappa)}
      \\
      %========================================
      \intertext{
      \flushleft\shadebox{$\elimF :: \ListRestPatt\;A\;\Gamma' \times (\Cont\;(\Gamma \concat \Gamma')\;B \to \Elim\;\Gamma\;A\;B)$}
      }
      %========================================
      \elimF(\pattListEnd, \kappa)
      &=
      \elimConstr{\elimBind{\cNil}{\kappa}}
      \\
      \elimF((\pattListNext{p}{o}), \kappa)
      &=
      \elimConstr{\elimBind{\cCons}{\elimArgsF(p \cons o, \kappa)}}
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\elimArgsF :: \PattSeqOpt\;\Gamma' \times (\Cont\;(\Gamma \concat \Gamma')\;A \to \Cont\;\Gamma\;A)$}
      }
      %========================================
      \elimArgsF(\seqEmpty)(\kappa)
      &=
      \kappa
      \\
      \elimArgsF(\pi \cons \seq{\pi}, \sigma)
      &=
      \elimF(\pi, \elimArgsF(\seq{\pi}, \sigma))
      %========================================
      \intertext{
      \flushleft\shadebox{$\orElseF :: \PattSeqOpt\;\Gamma' \times \Cont\;(\Gamma \concat \Gamma')\;A \times \Expr\;\cxtEmpty\;A \to \Cont\;(\Gamma \concat \Gamma')\;A$}
      }
      %========================================
      \orElseF(\seqEmpty, e, e')
      &=
      e
      \\
      \orElseF(\pattVar{x} \cons \seq{\pi}, \elimVar{x}{\kappa}, e)
      &=
      \elimVar{x}{\orElseF(\seq{\pi}, \kappa, e)}
      \\
      \orElseF(\pattRecord{\seq{\bind{x}{p}}} \cons \seq{\pi}, \elimRecord{\seq{x}}{\kappa}, e)
      &=
      \elimRecord{\seq{x}}{\orElseF(\seq{p} \concat \seq{\pi}, \kappa, e)}
      \\
      \orElseF(\pattConstr{c}{\seq{p}} \cons \seq{\pi}, \elimConstr{\elimBind{c}{\kappa}}, e)
      &=
      \elimConstr{\elimBind{c}{\orElseF(\seq{p} \concat \seq{\pi}, \kappa, e)}} \disjjoin \unlessF(c, e)
      \\
      \orElseF(\pattList{p}{o} \cons \seq{\pi}, \elimConstr{\elimBind{\cCons}{\sigma}}, e)
      &=
      \elimConstr{\elimBind{\cCons}{\orElseF(p \cons o \cons \seq{\pi}, \sigma)}} \disjjoin \unlessF(\cCons, e)
      \\
      \orElseF((\pattListNext{p}{o}) \cons \seq{\pi}, \elimConstr{\elimBind{\cCons}{\sigma}}, e)
      &=
      \elimConstr{\elimBind{\cCons}{\orElseF(p \cons o \cons \seq{\pi}, \sigma)}} \disjjoin \unlessF(\cCons, e)
      \\
      \orElseF(\pattListEnd \cons \seq{\pi}, \elimConstr{\elimBind{\cNil}{\kappa}}, e)
      &=
      \elimConstr{\elimBind{\cNil}{\orElseF(\seq{\pi}, \kappa)}} \disjjoin \unlessF(\cNil, e)
      \\[2mm]
      %========================================
      \intertext{
      \flushleft\shadebox{$\unlessF :: \Pi(c: \Ctr).\Expr\;\seqEmpty\;A \to \Elim\;\Gamma\;D\;A$ \emph{where} $\datatype{\Sigma}{c} = \tyForeign{\seq{A}}{D}$}
      }
      %========================================
      \unlessF(c, e)
      &=
      \elimConstr{\elimBind{c'}{\iterateF(\length{\seq{B}}, \lambda\kappa.\elimVar{x}{\kappa})(e)} \mid \datatype{\Sigma}{c'} = \tyForeign{\seq{B}}{D} \wedge c \neq c'}
      \\
