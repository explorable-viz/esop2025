\begin{figure}
   {\small \flushleft \shadebox{$\seq{p},\upsigma \elimmapsto \uptau$}
   \begin{smathpar}
      \inferrule*
      {
         \strut
      }
      {
         \seq{p},\exVar{e} \elimmapsto \exVar{e}
      }
      %
      \and
      %
      \inferrule*
      {
         \strut
      }
      {
         \seq{p},\emptyset \elimmapsto \emptyset
      }
      %
      \and
      %
      \inferrule*
      {
         \seq{p},\upsigma \elimmapsto \uptau
      }
      {
         p\cdot\seq{p},\elimPhi{\exPair{p}{\upsigma}\cdot\seq{\exPair{p}{\upsigma}}'} \elimmapsto \uptau
      }
      %
      \and
      %
      \inferrule*
      {
         \exVar{p'}\notin\phiKeys{\elimPhi{\seq{\exPair{p}{\upsigma}}}}
      }
      {
         p'\cdot\seq{p}', \elimPhi{\seq{\exPair{p}{\upsigma}}} \elimmapsto \emptyset  
      }
   \end{smathpar}

      }
      %
      \and
      %
      \inferrule*
      {
         \upsigma \disjjoin \exPair{\seq{p}}{\exVar{e}} = \uptau
      }
      {
         \elimPhi{\exPair{p}{\upsigma} \cdot \seq{\exPair{p}{\upsigma}}' } \disjjoin \exPair{p\cdot\seq{p}}{\exVar{e}} = \elimPhi{\exPair{p}{\uptau} \cdot \seq{\exPair{p}{\upsigma}}'}
      }
      %
      \and
      %
      \inferrule*
      {

      }
      {
         \elimPhi{\seq{\exPair{p}{\upsigma}}} \disjjoin \exPair{p'\cdot\seq{p}}{\exVar{e}} = \elimPhi{\seq{\exPair{p}{\upsigma}} \cdot \exPair{p'}{\uptau}}
      }
   \end{smathpar}}
