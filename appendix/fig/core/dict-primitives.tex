\begin{align*}
   % =============================================================
   \dictMap &: \Val{(\tyFun{A}{B})} \times \Val{(\tyDict{A})} \to \Val{(\tyDict{B})}
   % =============================================================
   \\
   \dictMap(v, d_{\upalpha})
   &=
   (\annDict{\seq{\bind{s}{(\upalpha', v')}}}{\upbeta}, \bigcup\set{\seq{E}} \cup \set{(\upalpha, \upbeta)})
   \\
   &\quad\textit{where }\upbeta\textit{ fresh and }
   \set{\seq{\bind{s}{(\upalpha', (v',E))}}} =
   \set{\bind{s}{(\upalpha', \apply(v, u))} \mid \bind{s}{(\upalpha', u)} \in d}
   \\[2mm]
   % =============================================================
   \dictIntersectionWith
   &:
   \Val{(\tyFun{A}{\tyFun{B}{A'}})} \times \Val{(\tyDict{A})} \times \Val{(\tyDict{B})} \to \Val{(\tyDict{A'})}
   % =============================================================
   \\
   \dictIntersectionWith(v, d_{\upalpha}, d'_{\upalpha'})
   &=
    (\annDict{\seq{\bind{s}{(\upbeta', v'}}}{\upbeta},
    \bigcup\set{\seq{E}} \cup \set{(\upalpha, \upbeta), (\upalpha', \upbeta)}
   \\
   &\quad\textit{ where }\upbeta\textit{ fresh and}
   \\
   &\quad
   \;\set{\seq{\bind{s}{(\upbeta', (v',E))}}} =
   \intersectionWith(\lambda((\upbeta', u), (\_, u')).(\upbeta', \apply(v, (u, u'))), d, d')
   \\[2mm]
   % =============================================================
   \dictDisjUnion &: \Val{(\tyDict{A})}^2 \to \Val{(\tyDict{A})}
   % =============================================================
   \\
   \dictDisjUnion(d_{\upalpha}, d'_{\upalpha'})
   &=
   ((d \disjunion d')_{\upbeta}, \set{(\upalpha, \upbeta), (\upalpha', \upbeta)})
   \quad\;\textit{iff }\upbeta\textit{ fresh and }\dom{d} \cap \dom{d'} = \varnothing
   \\[2mm]
   % =============================================================
   \dictDifference &: \Val{(\tyDict{A})} \times \Val{(\tyDict{B})} \to \Val{(\tyDict{A})}
   % =============================================================
   \\
   \dictDifference(d_{\upalpha}, d'_{\upbeta})
   &=
   ((d \setminus d')_{\upbeta}, \set{(\upalpha, \upbeta), (\upalpha', \upbeta)})
   \quad\;\textit{where }\upbeta\textit{ fresh}
   \\[2mm]
   % =============================================================
   \dictGet &: \Val{\tyStr} \times \Val{(\tyDict{A})} \to \Val{A}
   % =============================================================
   \\
   \dictGet(\annStr{s}{\upalpha}, d_{\upalpha'}) &= (v, \varnothing)
   \quad\textit{iff }\bind{s}{(\upbeta, v)} \in d
   \\[2mm]
   % =============================================================
   \dictFoldl &: \Val{(\tyFun{B}{\tyFun{A}{B}})} \times \Val{B} \times \Val{(\tyDict{A})} \to \Val{B}
   % =============================================================
   \\
   \dictFoldl(v, u, d_\upalpha) &= \foldlWithIndex(f, (u, \varnothing), d)
   \\
   &\quad\textit{where }
%   f(s, (\seq{t}, v'), (\upbeta, u')) = \opFwd{\apply}(v, (v', u')) \map \First(\uplambda (S, S').(s, (S, S')) \cons \seq{t})
   \\[2mm]
   % =============================================================
   \dictFromRecord &: \Val{(\tyRec{\seqRange{\bind{x_1}{A}}{\bind{x_n}{A}}})} \to \Val{(\tyDict{A})}
   % =============================================================
   \\
   \dictFromRecord(\annRec{\seq{\bind{x}{v}}}{\upalpha})
   &=
   (\annDict{\seq{\bind{s}{(\upbeta, v)}}}{\upalpha'},
    \set{(\upalpha, \upalpha')} \cup (\upalpha, \set{\seq{\upbeta}})
   \quad\textit{where }\upalpha', \seq{\upbeta}\textit{ fresh and }\str(x_i) = s_i
\end{align*}
