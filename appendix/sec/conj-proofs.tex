\subsection{\lemref{im-preim-conjugate}}
For any relation $R \subseteq X \times Y$ the functions $\demandedByR_{R}$ and $\demandsR_{R}$ are conjugate.
\setcounter{equation}{0}
\begin{proof}
   Let $X' \in \powerset{X}$, $Y' \in \powerset{Y}$.
   \small
   \newline
   Case: $\demandedByR_{R}(X') \cap Y' = \emptyset \implies X' \cap \demandsR_{R}(Y') = \emptyset$
   \begin{flalign}
      &
      \demandedByR_{R}(X') \cap Y' = \emptyset
      &
      \text{suppose}
      \notag
      \\
      &
      \set{y \in Y \mid \exists x \in X'.(x,y) \in R} \cap Y' = \emptyset
      &
      \text{def of $\demandedByR_{R}$}
      \notag
      \\
      &
      \set{y \in Y' \mid \exists x \in X'.(x,y) \in R} = \emptyset
      &
      \text{def of $\cap$}
      \locallabel{preim-intersection}
      \\
      &
      \forall y \in Y' .\; \forall x \in X'. (x,y) \not\in R
      &
      \text{equivalent to (\localref{preim-intersection})}
      \locallabel{unintersect}
      \\
      &
      \set{x \in X' \mid \exists y \in Y'.(x,y) \in R} = \emptyset
      &
      \text{equivalent to (\localref{unintersect})}
      \notag
      \\
      &
      \qedLocal
      X' \cap \set{x \in X \mid \exists y \in Y'.(x,y) \in R} = \emptyset
      &
      \text{def of $\cap$}
      \notag
      \\
      &
      X' \cap \demandsR_{R}(Y') = \emptyset 
      &
      \text{def of $\demandsR_{R}$}
      \notag
   \end{flalign}
   \newline
   Case: $ X' \cap \demandsR_{R}(Y') = \emptyset \implies \demandedByR_{R}(X') \cap Y' = \emptyset$
   \begin{flalign}
      &
      X' \cap \demandsR_{R}(Y') = \emptyset
      &
      \text{suppose}
      \notag
      \\
      &
      X' \cap \set{x \in X \mid \exists y \in Y'.(x,y) \in R} = \emptyset
      &
      \text{def of $\demandsR_{R}$}
      \notag
      \\
      &
      \set{x \in X' \mid \exists y \in Y'.(x,y) \in R}
      &
      \text{def of $\cap$}
      \locallabel{preim-intersect-2}
      \\
      &
      \forall y \in Y' .\; \forall x \in X'. (x,y) \not\in R
      &
      \text{equivalent to (\localref{preim-intersect-2})}
      \locallabel{preim-unintersect-2}
      \\
      &
      \qedLocal
      \set{y \in Y \mid \exists x \in X'.(x,y) \in R} \cap Y' = \emptyset
      &
      \text{equivalent to (\localref{preim-unintersect-2})}
      \notag
      \\
      &
      \demandedByR_{R}(X') \cap Y' = \emptyset
      &
      \text{def of $\demandedByR_{R}$}
      \notag
   \end{flalign}
\end{proof}

\subsection{\propref{conj-galois-correspond}}
Suppose functions $f: A \to B$ and $g: B \to A$ between Boolean algebras. The following statements are
equivalent:
\begin{enumerate}
   \item $f$ and $g$ form a conjugate pair
   \item $f$ and $\dual{g}$ form a Galois connection
\end{enumerate}
\setcounter{equation}{0}
\begin{proof}
   Fix $x \in A$ and $y \in B$.
   \newline
   Case: $f, g$ conjugate $\implies$ $f, \dual{g}$ form Galois connection.
   \begin{flalign}
      &
      f(x) \leq_{B} y
      &
      \text{(suppose)}
      \notag
      \\
      &
      f(x) \wedge_{B} \neg y = \bot
      &
      \notag
      \\
      &
      x \wedge_{A} g(\neg y) = \bot
      &
      \text{(conjugacy)}
      \notag
      \\
      &
      x \leq_{A} \neg (g(\neg y))
      &
      \notag
      \\
      &
      \qedLocal
      x \leq_{A} \dual{g}(y)
      &
      \text{(def of $\dual{\cdot}$)}
      \notag
   \end{flalign}
   \newline
   Case: $f, \dual{g}$ Galois connection $\implies$ $f, g$ conjugate
   \begin{flalign}
      &
      f(x) \wedge_{B} y = \bot
      &
      \text{suppose}
      \notag
      \\
      &
      f(x) \leq_{B} \neg y 
      &
      \notag
      \\
      &
      x \leq_{A} \dual{g}(\neg y)
      &
      \text{(GC)}
      \notag
      \\
      &
      x \leq_{A} \neg (g (y))
      &
      \text{(def of $\dual{\cdot}$)}
      \notag
      \\
      &
      x \wedge_{A} g(y) = \bot
      &
      \notag
   \end{flalign}
\end{proof}

\subsection{\lemref{conjugate:image-preimage-duality}}

\begin{proof}

   \begin{flalign}

      \text{def of $\demandedByR_R$}
      \notag
      \\
      &
      = \compl{\set{y \in Y \mid \exists x \in \compl{(X')}.(x,y) \in R}}

\subsection{Correctness of $\demandedByAlg{}$}
To prove \propref{demandedByAlg-computes-demandedBy}, we first need to show that
the procedure $\demandedByAlg{}$ always terminates, and then we need to show that
when it terminates it does so with the correct result.
\subsubsection{Termination}
We can prove termination by defining a ranking function for $\demandedByV{}$ in terms of
$G$ and $X$. The ranking function $\opName{Rank}(G, X) \eqdef \exPair{|\E(G)|}{|X|}$. 
We then use the lexicographic ordering on these pairs of integers. 

If we consider the rule \ruleName{$\demandedByAlg{}$-step}, we can conclude that the
ranking function decreases on consecutive states of the algorithm:
For an $\alpha \in X$, we remove $\outE{G}(\alpha)$ from $G$. If $|\outE{G}(\alpha)| \geq 1$
then $|\E(G \setminus_{E} \outE{G}(|\alpha))| < |\E(G)|$. If $|\outE{G}(\alpha)| = 0$ then
$G$ stays the same size, but $|X \cup \outN{G}(\alpha)| = |X| < |\alpha \cdot X|$. In either
case, the ranking function of the consecutive state strictly decreases every step.

\subsubsection{Proof of \propref{demandedByAlg-computes-demandedBy}}
To prove \propref{demandedByAlg-computes-demandedBy}, we begin by noting that
any graph $H$ computed by $\demandedByV{}$ is a finite directed acyclic graph (DAG).
We will use the following lemma's.
\begin{lemma}[Node reachable from a source]
   Given a finite DAG $H$ we have that $\forall \beta \in \V(H)$ $\exists \alpha \in \sources{H}$ with $R_{H}(\alpha, \beta)$.
\end{lemma}
\begin{proof}
   For a sketch of this fact, fix a $\beta$. If $\beta \in \sources{H}$,
   in which case $R_{H}(\beta, \beta)$ trivially. If $\beta \notin \sources{H}$, then $\exists \alpha$
   with $(\alpha,\beta) \in \E(H)$. Then we can repeeat this check with $\alpha$. Since $H$ is finite
   we eventually reach a source.
\end{proof}


\begin{lemma}
   \label{lem:partial-slice-sub-slice}
   Given graphs $G,H,H'$, and nodes $X \subseteq \sources{G}$ if
   $H, X, G \demandedByV{} H'$ then $H$ is a subgraph of $H'$.
\end{lemma}
\begin{proof}
   Can proceed by induction on rules. For \ruleName{$\demandedByV{}$-done}, $H = H'$.
   For \ruleName{$\demandedByV{}$-step}, we have that $H \subseteq H \cup \outE{G}(\alpha)$.
   Then it holds by induction hypothesis
\end{proof}

      R_{H'}(\alpha', \beta') \wedge R_{H'}(\beta', \beta)
      &
      \text{(\localref{alpha-reaches-beta'}, \localref{beta'-reaches-beta}, IH)}
      \locallabel{h'-conjunct}
      \\
      &
      R_{H'}(\alpha, \beta)
      &
      \text{(\localref{alpha-beta-in-slice}, \localref{h'-conjunct})}
      \notag
      \\
      &
      R_{H'}(\alpha', \alpha)
      &
      \text{(\ref{lem:partial-slice-sub-slice})}
      \notag
      \\
      \qedLocal
      &
      R_{H'}(\alpha', \alpha) \wedge R_{H'}(\alpha, \beta)
      &
      \notag
      % \\
      % &
      % \caseName{Case $\impliedby$}
      % &
      % \notag
      % \\
      % &
      % R_{H'}(\alpha', \alpha) \wedge R_{H'}(\alpha, \beta)
      % &
      % \text{(assume)}
      % \\
      % &
      % ((\alpha, \beta') \in \E(H')) \wedge R_{H'}(\beta',\beta)
      % \quad
      % (\exists \beta' \in \outN{G}(\alpha))
      % &
      % \\
      % &
      % R_{H'}(\alpha', \beta') \wedge R_{H'}(\beta', \beta)
      % &
      % \\
   \end{flalign}
\end{proof}

\begin{theorem}
   \ref{prop:dembyv-preserves-paths} $\implies$ \ref{prop:reachability-in-g''}
\end{theorem}
\setcounter{equation}{0}
\begin{proof}
   % Set $H \eqdef \exPair{X}{\emptyset}$. Then $\forall \alpha \in X$, if for some $\beta \in \sinks{G}$ we have $R_{G}(\alpha, \alpha) \wedge R_{G}(\alpha, \beta)$ then by \ref{prop:dembyv-preserves-paths} this is equivalent to $R_{H}(\alpha, \alpha) \wedge R_{H'}(\alpha, \beta)$, and thus we have \ref{prop:reachability-in-g''} by taking the second conjunct in each statement.
   \begin{flalign}
      &
      H, X, G \demandedByAlg{} H'
      &
      \text{assume}
      \notag
      \\
      &
         R_{H}(\alpha', \alpha) \wedge R_{G}(\alpha, \beta) \iff R_{H'}(\alpha', \alpha) \wedge R_{H'}(\alpha, \beta)
      &
      \text{(\ref{prop:dembyv-preserves-paths})}
      \notag
      \\
      &
      H = \exPair{X}{\emptyset}
      &
      \text{assume}
      \notag
      \\
      &
      \caseName{Case $\implies$}
      &
      \notag
      \\
      &
      \exists \beta \in \sinks{G}\text{ with } R_{G}(\alpha, \beta)
      &
      \text{assume}
      \locallabel{beta-in-sinks}
      \\
      &
      R_{H}(\alpha, \alpha) \wedge R_{G}(\alpha, \beta)
      &
      \text{($R_{H}$ reflexive, \localref{beta-in-sinks})}
      \locallabel{h-g-conjunct}
      \\
      &
      R_{H'}(\alpha, \alpha) \wedge R_{H'}(\alpha, \beta)
      &
      \text{(\ref{prop:dembyv-preserves-paths}, \localref{h-g-conjunct})}
      \locallabel{h-conjunct}
      \\
      &
      R_{H'}(\alpha, \beta)
      &
      \text{(\localref{h-conjunct}, $\wedge$)}
      \notag
      \\
      \qedLocal
      &
      R_{G}(\alpha, \beta) \implies  R_{H'}(\alpha, \beta)
      &
      \notag
      \\
      &
      \caseName{Case $\impliedby$}
      &
      \notag
      \\
      &
      \exists \beta \in \sinks{H'}\text{ with } R_{H'}(\alpha, \beta)
      &
      \text{assume}
      \locallabel{beta-in-h'}
      \\
      &
      R_{H'}(\alpha, \alpha) \wedge R_{H'}(\alpha, \beta)
      &
      \text{(\localref{beta-in-h'}, $R_{H'}$ reflexive)}
      \locallabel{h'-conjunct}
      \\
      &
      R_{H}(\alpha, \alpha) \wedge R_{G}(\alpha, \beta)
      &
      \text{(\ref{prop:dembyv-preserves-paths}, \localref{h'-conjunct})}
      \locallabel{h-g-conjunct-2}
      \\
      &
      R_{G}(\alpha, \beta)
      &
      \text{(\localref{h-g-conjunct-2}, def of $\wedge$)}
      \notag
      \\
      \qedLocal
      &
      R_{H'}(\alpha, \beta) \implies R_{G}(\alpha, \beta)
      &
      \notag
   \end{flalign}
\end{proof}
