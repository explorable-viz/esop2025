\subsubsection{Proof of \propref{demandedByAlg-computes-demandedBy}}
To prove \propref{demandedByAlg-computes-demandedBy}, we begin by noting that
any graph $H$ computed by $\demandedByV{}$ is a finite directed acyclic graph (DAG).

\begin{proposition}[Reachability in $G^{\twoPrime}$]
   \label{prop:reachability-in-g''}
   For graphs $G$, $H$ with $X \subseteq \sources{G}$, if $\exPair{X}{\emptyset}, X, G \demandedByV{} H$ then the following holds:
   $$
    \forall \alpha \in X \quad R_{G}(\alpha,\beta) \iff R_{H}(\alpha, \beta)
   $$
\end{proposition}

\setcounter{equation}{0}
\begin{theorem}
   \ref{prop:reachability-in-g''} $\implies$ \ref{prop:demandedByAlg-computes-demandedBy}
\end{theorem}
\begin{proof}
   Attempt to prove that \ref{prop:reachability-in-g''} implies each direction of the equivalence in \ref{prop:demandedByAlg-computes-demandedBy}.
   \begin{flalign}
      &
      \forall \alpha \in X \; R_{G}(\alpha, \beta) \iff R_{H}(\alpha, \beta)
      &
      \text{(\ref{prop:reachability-in-g''})}
      \notag
      \\
      &
      \caseName{WTS: } R_{G}(\alpha, \beta) \implies R_{H}(\alpha, \beta)
      &
      \notag
      \\
      &
      X \demandedByAlg{} Y
      \quad
      (\exists Y)
      &
      \text{(suppose)}
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*
            {
               \exPair{X}{\emptyset}, X, G \demandedByV{} H
            }
            {
               X \demandedByAlg{} \V(H) \cap \sinks{G} \Lowlight{(= Y)}
            }
         \end{smathpar}
      }
      &
      \text{(def of $\demandedByAlg{}$)}
      \notag
      \\
      &
      \text{Want to show: }\demandedByR_{G}(X) = \V(H) \cap \sinks{G}
      &
      \notag
      \\
      &
      \caseName{WTS: } \demandedByR_{G}(X) \supseteq (\V(H) \cap \sinks{G})
      &
      \notag
      \\
      &
      \beta \in \demandedByR_{G}(X)
      \quad
      (\forall \beta)
      &
      \text{(assume)}
      \locallabel{beta-in-demanded}
      \\
      &
      \beta \in \sinks{G}
      &
      \text{(\localref{beta-in-demanded})}
      \locallabel{beta-in-sinks-g}
      \\
      &
      R_{G}(\alpha, \beta)
      \quad
      (\exists \alpha \in X)
      &
      \text{(def of $\demandedByR_{G}$)}
      \locallabel{alpha-reaches-beta-1}
      \\
      &
      R_{H}(\alpha, \beta)
      &
      \text{(\localref{alpha-reaches-beta-1}, \ref{prop:reachability-in-g''})}
      \\
      &
      \beta \in \V(H)
      &
      \text{(Def of $R_{H}$)}
      \locallabel{beta-in-h-1}
      \\
      \qedLocal
      & 
      \beta \in \V(H) \cap \sinks{G}
      &
      \text{(\localref{beta-in-sinks-g}, \localref{beta-in-h-1})}
      \locallabel{}
      \\
      &
      \caseName{WTS: }\demandedByR_{G}(X)  \subseteq (\V(H) \cap \sinks{G})
      &
      \notag
      \\
      &
      \beta \in \V(H) \cap \sinks{G}
      &
      \text{(assume)}
      \notag
      \\
      &
      R_{H}(\alpha, \beta)
      \quad
      (\exists \alpha \in \sources{H} \Lowlight{= X})
      &
      \notag
      \\
      &
      R_{G}(\alpha, \beta)
      &
      \text{\ref{prop:reachability-in-g''}}
      \\
      &
      \beta \in \demandedByR_{G}(X)
      &
      \text{(def of $\demandedByR_{G}$)}
      \\
      &
      \caseName{WTS: } X \demandedByAlg{G} Y \impliedby \demandedByR_{G}(X) = Y
      &
      \notag
      \\
      &
      \demandedByR_{G}(X) = Y
      &
      \text{(assume)}
      \\
      &
      Y = \set{\beta \in \sinks{G} \mid \exists \alpha \in X \; R_{G}(\alpha, \beta)}
      &
      \text{(def of $\demandedByR_{G}$)}
      \notag
      \\
      &
      (X, \emptyset), X, G \demandedByV{} H
      \quad
      (\exists H)
      &
      \text{termination}
      \locallabel{demandedby-terminates}
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule
            {
               \exPair{X}{\emptyset}, X, G \demandedByV{} H
            }
            {
               X \demandedByAlg{} \V(H) \cap \sinks{G}
            }
         \end{smathpar}
      }
      &
      \text{(\localref{demandedby-terminates})}
      \notag
      \\
      &
      \text{Want to show that } \V(H) \cap \sinks{G} = \demandedByR_{G}(X)
      &
      \notag
      \\
      &
      \text{Analagous to above.}
      &
   \end{flalign}
\end{proof}

We can prove \propref{reachability-in-g''} with the following:
\begin{proposition}
   \label{prop:dembyv-preserves-paths}
   For graphs $G, H, H'$ and $X \subseteq \V(G)$, if $H, X, G \demandedByV{} H'$ then
   $\forall \alpha \in \sources{H}, \beta \in \V{H}$:
   $$
      R_{H\cup G}(\alpha, \beta) \iff R_{H'}(\alpha, \beta)
   $$
\end{proposition}
\setcounter{equation}{0}
\begin{proof}
   By induction
   \begin{flalign}
      &
      X \subseteq \V(G)
      &
      \text{assume}
      \notag
      \\
      &
      H, X, G \demandedByV{} H'
      &
      \text{assume}
      \\
      &
      \caseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*
            {
               \strut
            }
            {
               H, \emptyset, G \demandedByV{} H
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      H = H \subseteq H \cup G
      &
      \locallabel{H-subgraph-H-cup-G}
      \\
      &
      \caseName{WTS: } R_{G}(\alpha, \beta) \impliedby R_{H'}(\alpha, \beta)
      &
      \notag
      \\
      &
      R_{H}(\alpha, \beta)
      &
      \text{assume}
      \\
      \qedLocal
      &
      R_{H\cup G}(\alpha, \beta)
      &
      \text{(\localref{H-subgraph-H-cup-G})}
      \notag
      \\
      &
      \caseName{WTS: } R_{G}(\alpha, \beta) \implies R_{H'}(\alpha, \beta)
      &
      \notag
      \\
      &
      R_{H \cup G}(\alpha, \beta)
      \quad
      (\forall \beta \in \V(H). \exists \alpha \in \sources{H})
      &
      \text{assume}
      \\
      \qedLocal
      &
      R_{H}(\alpha, \beta)
      &
      \notag
      \\
      &
      \caseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*
            {
               H \cup \outE{G}(\alpha), X \cup \outN{G}(\alpha), G \setminus_{E} \outE{G}(\alpha)
               \demandedByV{}
               H'
            }
            {
               H, \alpha \cdot X, G
               \demandedByV{}
               H'
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      R_{(H \cup \outE{G}) \cup (G \setminus_{E} \outE{G}(\alpha))}(\alpha, \beta)
      \iff
      R_{H'}(\alpha, \beta)
      &
      \text{assume IH}
      \locallabel{reachable-IH}
      \\
      &
      \caseName{WTS: } R_{H\cup G}(\alpha, \beta) \implies R_{H'}(\alpha, \beta)
      &
      \notag
      \\
      &
      R_{H \cup G}(\alpha, \beta)
      &
      \text{assume}
      \locallabel{h-cup-g-assumption}
      \\
      &
      (H \cup \outE{G}) \cup (G \setminus_{E} \outE{G}(\alpha)) = H \cup G
      &
      \locallabel{union-equal}
      \\
      &
      R_{H \cup G} = R_{(H \cup \outE{G}) \cup (G \setminus_{E} \outE{G}(\alpha))}
      &
      \text{(\localref{union-equal}, \localref{h-cup-g-assumption})}
      \notag
      \\
      &
      R_{(H \cup \outE{G}) \cup (G \setminus_{E} \outE{G}(\alpha))}(\alpha, \beta)
      &
      \\
      \qedLocal
      &
      R_{H'}(\alpha, \beta)
      &
      \text{(\localref{reachable-IH})}
      \notag
      \\
      &
      \caseName{WTS: } R_{H\cup G}(\alpha, \beta) \impliedby R_{H'}(\alpha, \beta)
      &
      \notag
      \\
      &
      R_{H'}(\alpha, \beta)
      &
      \text{assume}
      \notag
      \\
      \qedLocal
      &
      R_{H \cup G}(\alpha, \beta)
      &
      \text{(\ref{lem:full-slice-sub-partial-unexplored})}
      \notag
   \end{flalign}
\end{proof}

\begin{theorem}
   \ref{prop:dembyv-preserves-paths} $\implies$ \ref{prop:reachability-in-g''}
\end{theorem}
\setcounter{equation}{0}
\begin{proof}
   \begin{flalign}
      &
      H, X, G \demandedByAlg{} H'
      &
      \text{assume}
      \notag
      \\
      &
         R_{H \cup G}(\alpha, \beta) \iff R_{H'}(\alpha, \beta)
      &
      \text{(\ref{prop:dembyv-preserves-paths})}
      \locallabel{assume-iff}
      \\
      &
      H = \exPair{X}{\emptyset}
      &
      \text{assume}
      \notag
      \\
      &
      \caseName{WTS: } R_{H\cup G}(\alpha, \beta) \implies  R_{H'}(\alpha, \beta)
      &
      \notag
      \\
      &
      R_{G}(\alpha, \beta)
      &
      \text{assume}
      \locallabel{assume-g}
      \\
      &
      R_{H \cup G}(\alpha, \beta)
      &
      \text{($G \subseteq H \cup G$)}
      \notag
      \\
      \qedLocal
      &
      R_{H'}(\alpha, \beta)
      &
      \text{(\localref{assume-iff})}
      \notag
      \\
      &
      \caseName{WTS: } R_{H\cup G}(\alpha, \beta) \impliedby R_{H'}(\alpha, \beta)
      &
      \notag
      \\
      &
      R_{H'}(\alpha, \beta)
      &
      \text{assume}
      \locallabel{assume-h-prime}
      \\
      &
      R_{H \cup G}(\alpha, \beta)
      &
      \text{(\localref{assume-iff})}
      \notag
      \\
      \qedLocal
      &
      R_{G}(\alpha, \beta)
      &
      \text{$H \cup G = G$}
      \notag
   \end{flalign}
\end{proof}
