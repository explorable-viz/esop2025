By induction on derivations of $\evalFwdS$\newline
Assume:
\begin{equation}
   \Gamma \vdash \bind{s}{A}
\end{equation}
\begin{equation}
   \gamma : \Gamma
\end{equation}
\begin{equation}
   s \desugarFwdS e
\end{equation}
\begin{equation}
   \locallabel{desug:env}
   \gamma \desugarFwdS \gamma'
\end{equation}
\begin{proof}
\small
\begin{flalign}
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-var}}
         ]
         {
            \strut
         }
         {
            {\gamma^{\dag}}' \cdot (\bind{x}{v'}), x \evalFwdS v'
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugarFwdS$-var}}
         ]
         {
            \strut
         }
         {
            \exVar{x} \desugarFwdS \exVar{x}
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\desugarFwdS$-var}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-var}}
         ]
         {
            \strut
         }
         {
            {\gamma^{\dag}} \cons (\bind{x}{v}), \exVar{x} \evalSugS v
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\evalSugS$-var}
   \notag
   \\
   &
   \qedLocal
   v \desugarFwdS v'
   &
   \text{(\localref{desug:env}) and \lemref{lemma:varlookup}}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-constr}}
         ]
         {
            \gamma', \seq{e} \evalFwdS \seq{v}'
         }
         {
            \gamma', \exConstr{c}{\seq{e}} \evalFwdS \exConstr{c}{\seq{v}'}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugarFwdS$-constr}}
         ]
         {
            \seq{s} \desugarFwdS \seq{e}
         }
         {
            \exConstr{c}{\seq{s}} \desugarFwdS \exConstr{c}{\seq{e}}
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\desugarFwdS$}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-constr}}
         ]
         {
            \gamma, \seq{s} \evalSugS \seq{v}
         }
         {
            \gamma, \exConstr{c}{\seq{s}} \evalSugS \exConstr{c}{\seq{v}}
         }
      \end{smathpar}
   }
   &
   \text{inversion on $\evalSugS$-constr}
   \notag
   \\
   &
   \seq{v} \desugarFwdS \seq{v}'
   &
   \text{IH}
   \notag
   \\
   &
   \qedLocal
   \exConstr{c}{\seq{v}} \desugarFwdS \exConstr{c}{\seq{v}'}
   &
   \text{$\evalSugS$-constr}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-apply}},
            width=3.3in,
         ]
         {
            \gamma, (e, e') \evalS (\exClosure{\gamma'_1}{\rho}{\sigma}, v')
            \\
            \gamma'_1, \rho \closeDefsFwdR \gamma'_2
            \\
            v', \sigma \matchFwdS \gamma_3, e^{\dag}
            \\
            \gamma'_1 \concat \gamma'_2 \concat \gamma'_3, e^{\dag} \evalFwdS u
         }
         {
            \gamma', \exApp{e}{e'}
            \evalFwdS
            u
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugarFwdS$-app}}
         ]
         {
            s \desugarFwdS e
            \\
            s' \desugarFwdS e'
         }
         {
            \exApp{s}{s'} \desugarFwdS \exApp{e'}{}
         }
      \end{smathpar}
   }
   &
   \text{Inversion on $\desugarFwdS$}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-app}},
            width=3in,
         ]
         {
            \gamma, (s, s') \evalSugR{(S,S')} (\exClosure{\gamma_1}{g}{\mu}, v)
            \\
            \gamma_1, g \closeDefsS \gamma_2
            \\
            v, \mu \matchFwdR{p} \gamma_3, s^\dag
            \\
            \gamma_1 \concat \gamma_2 \concat \gamma_3, s^\dag \evalSugR{S^\dag} v^{\dag}
         }
         {
            \gamma, \exApp{s}{s'} \evalSugR{\trApp{S}{S'}{p}{S^{\dag}}} v^{\dag}
         }
      \end{smathpar}
   }
   &
   \text{Inversion on $\evalSugS$-app}
   \notag
   \\
   &
   \exClosure{\gamma_1}{g}{\mu} \desugarFwdS \exClosure{\gamma'_1}{\rho}{\sigma}
   &
   \text{IH}
   \notag
   \\
   &
   v \desugarFwdS v'
   &
   \text{IH}
   \notag
   \\
   &
   \gamma_2 \desugarFwdS  \gamma'_2
   &
   \text{\lemref{closedefs-eq}}
   \notag
   \\
   &
   \gamma_3 \desugarFwdS \gamma'_3 \desugarFwdS \opAnd s^{\dag} \desugarFwdS e^{\dag}
   &
   \text{$\mu \desugarFwdS \sigma$ and \lemref{matching-bi}}
   \notag
   \\
   &
   \qedLocal
   v^{\dag} \desugarFwdS u
   &
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-let-rec}}
         ]
         {
            \gamma, g \closeDefsS \gamma'
            \\
            \gamma \concat \gamma', s \evalSugR{S} v
         }
         {
            \gamma, \exLetRecPiecewise{g}{s} \evalSugR{\trLetRec{g}{S}} v
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[lab={\ruleName{$\desugarFwdR$-let-rec}}]
         {
            \mu_i \desugarFwdR \sigma_i
            \quad
            (\forall i \numleq \length{\seq{x}})
            \\
            s \desugarFwdR e
         }
         {
            \exLetRecPiecewise{\seq{\bind{x}{\mu}}}{s}
            \desugarFwdR
            \exLetRecPiecewise{\seq{\bind{x}{\sigma}}}{e}
         }
      \end{smathpar}
   }
   &
   \text{$\desugarFwdS$-let-rec}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-let-rec}}
         ]
         {
            {\gamma^{\dag}}, \set{\seq{\bind{x}{\sigma}}} \closeDefsFwdR {\gamma^{\dag}}'
            \\
            {\gamma^{\dag}} \concat {\gamma^{\dag}}', e \evalFwdS v
         }
         {
            {\gamma^{\dag}}, \exLetRec{\seq{\bind{x}{\sigma}}}{e}
            \evalFwdS
            v
         }
      \end{smathpar}
   }
   &
   \text{inversion of $\evalFwdS$-let-rec}
   \notag
   \\
   &
   \text{Then we conclude that $\gamma \concat \gamma' \desugarFwdS {\gamma^{\dag}} \concat {\gamma^{\dag}}'$}
   &
   \text{(\localref{desug:env}) and \lemref{closedefs-eq}}
   \notag
   \\
   &
   \qedLocal
   v \desugarFwdS v'
   &
   \text{IH}
   \notag
\end{flalign}
\end{proof}
