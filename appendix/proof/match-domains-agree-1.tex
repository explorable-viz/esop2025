Induction on $\match$ derivation.
\begin{proof}
\small
\begin{flalign}
   &
   u, p \match \gamma
   &
   \text{suppose}
   \notag
   \\
   &
   u \desugar v
   &
   \text{suppose}
   \notag
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-var}}
         ]
         {
            \strut
         }
         {
            u, x \match \set{\bind{\exVar{x}}{u}} = \gamma
         }
      \end{smathpar}
   }
   \\
   &
   \mkElimF(x, \kappa) = \elimVar{x}{\kappa}
   &
   \text{Def.~\mkElimF}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\match$-var}}]
         {
            \strut
         }
         {
            v, \elimVar{x}{\kappa} \match \set{\bind{x}{v}}, \kappa
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\desugar$-env}}]
         {
             u \desugar v
         }
         {
             \gamma = \set{\bind{x}{u}} \desugar \set{\bind{x}{v}}
         }
      \end{smathpar}
    }
   &
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-constr}}
         ]
         {
            \seq{u},\seq{p} \match \gamma
         }
         {
            \exConstr{c}{\seq{u}},\exConstr{c}{\seq{p}} \match \gamma
         }
      \end{smathpar}
   }
\end{flalign}
\end{proof}
