We assume the following:
\begin{equation}
    \gamma \desugarFwdS \gamma'
\end{equation}
\begin{equation}
    v \desugarFwdS v'
\end{equation}
\begin{proof}
    \begin{flalign}
    \intertext{\crossrule}
    %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
    \Case{\bind{x}{v} \in \gamma}
    &
    \derivation{\derivationWidth}{
        \begin{smathpar}
            \inferrule*[]
            {
                v_i \desugarFwdS v'_i \quad (\forall i \le \length{\seq{x}})
            }
            {
                \set{\seq{\bind{x}{v}}} \desugarFwdS \set{\seq{\bind{x}{v}}'}
            }
        \end{smathpar}
    }
    &
    \notag
    \\
    &
    \exists i.\; v = v_i
    &
    \text{Assumption (1)}
    \notag
    \\
    &
    \qedLocal
    \bind{x}{v'} \in \gamma'
    &
    \text{Assumption (2), and $\desugarFwdS$}
    \notag
    \\
    \Case{\bind{x}{v'} \in \gamma'}
    &
    \derivation{\derivationWidth}{
        \begin{smathpar}
            \inferrule*[]
            {
                v_i \desugarFwdS v'_i \quad (\forall i \le \length{\seq{x}})
            }
            {
                \set{\seq{\bind{x}{v}}} \desugarFwdS \set{\seq{\bind{x}{v}}'}
            }
        \end{smathpar}
    }
    &
    \\
    &
    \exists i.\; v' = v'_i
    &
    \text{Assumption (1)}
    \notag
    \\
    &
    \qedLocal
    \bind{x}{v} \in \gamma
    &
    \text{Assumption (2) and inversion on $\desugarFwdS$}
    \notag
    \end{flalign}
 \end{proof}
