\begin{proof}
   \small
    \begin{flalign}
      &
      v \desugarFwdS v'
      &
      \text{suppose}
      \locallabel{v-desugar}
      \\
      &
      \gamma \desugarFwdS \gamma'
      &
      \text{suppose}
      \locallabel{gamma-desugar}
      \\
    %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
    \Case{\bind{x}{v} \in \gamma}
    &
    \derivation{\derivationWidth}{
        \begin{smathpar}
            \inferrule*[]
            {
                v_i \desugarFwdS v'_i \quad (\forall i \le \length{\seq{x}})
            }
            {
                \set{\seq{\bind{x}{v}}} \desugarFwdS \set{\seq{\bind{x}{v}}'}
            }
        \end{smathpar}
    }
    &
    \notag
    \\
    &
    \exists i.\; v = v_i
    &
    \text{(1)}
    \notag
    \\
    &
    \qedLocal
    \bind{x}{v'} \in \gamma'
    &
    \text{(2), and $\desugarFwdS$}
    \notag
    \\
    \Case{\bind{x}{v'} \in \gamma'}
    &
    \derivation{\derivationWidth}{
        \begin{smathpar}
            \inferrule*[]
            {
                v_i \desugarFwdS v'_i \quad (\forall i \le \length{\seq{x}})
            }
            {
                \set{\seq{\bind{x}{v}}} \desugarFwdS \set{\seq{\bind{x}{v}}'}
            }
        \end{smathpar}
    }
    &
    \\
    &
    \exists i.\; v' = v'_i
    &
    \text{(\localref{v-desugar})}
    \notag
    \\
    &
    \qedLocal
    \bind{x}{v} \in \gamma
    &
    \text{(\localref{gamma-desugar}) and inversion on $\desugarFwdS$}
    \notag
    \end{flalign}
 \end{proof}
