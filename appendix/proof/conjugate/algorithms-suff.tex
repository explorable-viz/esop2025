\subsection{Correctness of $\sufficesE{}$}

\begin{definition}[Partial Slice]
   \label{def:partial-slice}
   Graphs $H,P,G$ constitute a \textit{partial slice} iff:
   \begin{enumerate}
      \item $H,P,G$ are edge-disjoint
      \item $\forall (\alpha,\beta) \in (\sources{G \cup P \cup H} \setminus \sources{H}) \times \V(H).\;(\alpha,\beta) \notin R_{G \cup P \cup H}$
   \end{enumerate}
\end{definition}
\begin{corollary}
   $H,P,G$ a partial slice $\iff$ $H,P,G$ edge disjoint and $\forall (\alpha,\beta) \in \E(G) \cup \E(P).\;\beta \notin \V(H)$ 
\end{corollary}
\begin{definition}[Input Relation]
   For a graph $G$ with reachability relation $R$, define it's \text{input relation}
   $R_{I} \eqdef (\sources{G} \times \V(G)) \cap R$
\end{definition}

\begin{definition}[$\sufficesR^*$ for a dependence graph]
   For a graph $G$, with an input relation $R_{I}$, and a set of input vertices $X \subseteq \sources{G}$:
   \[\sufficesR^*_{G}(X) \eqdef \sufficesR_{R_{I}}(X) = \set{y \in \V(G) \mid \nexists x \in \sources{G} \setminus X.\; (x,y) \in R_{I}}\]
\end{definition}

\begin{lemma}
   \label{lem:appendix:proof:conjugate:algorithms-suff:star-intersect-sinks}
   $\sufficesR_{G}(X) = \sinks{G} \cap \sufficesR^*_{G}(X)$
\end{lemma}
\begin{lemma}
   \label{lem:appendix:proof:conjugate:algorithms-suff:suffices-sources}
   If $H,P,G$ a partial slice with $\sources{G} \cap \V(P) \subseteq \sources{P}$ and $\sinks{H} \subseteq \sinks{G}$:
   \[ \sufficesR^*_{G \cup P \cup H}(\sources{H}) = \V(H) \]
\end{lemma}
\begin{proof}
   \small
   \begin{flalign}
      &
      H,P,G \text{ a partial slice}
      &
      \text{suppose}
      \notag
      \\
      &
      \sources{G} \cap \V(P) \subseteq \sources{P}
      &
      \text{suppose}
      \locallabel{sources-p-subset-sources-g}
      \\
      &
      \sinks{H} \subseteq \sinks{G}
      &
      \text{suppose}
      \locallabel{sinks-h-subset-sinks-g}
      \\
      &
      \caseName{$\subseteq$ direction}
      \notag
      \\
      &
      y \in \sufficesR^*_{G \cup P \cup H}(\sources{H})
      &
      \text{suppose}
      \notag
      \\
      &
      \forall x \in \sources{G \cup P \cup H} \setminus \sources{H}.\; (x,y) \notin R_{G \cup P \cup H}
      &
      \text{def. $\sufficesR^*$}
      \locallabel{x-notin-h-cant-reach-h}
      \\
      &
      \text{Remains to show: } y \in \V(H)
      &
      \notag
      \\
      &
      \neg(\alpha \in \sources{G} \wedge E = \inE{P}(\alpha) \neq \emptyset)
      &
      \text{\localref{sources-p-subset-sources-g}}
      \notag
      \\
      &
      \neg(\alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset)
      &
      \text{\localref{sinks-h-subset-sinks-g}}
      \notag
      \\
      &
      \caseName{$\supseteq$ direction}
      \notag
      \\
      &
      y \in \V(H)
      &
      \text{suppose}
      \notag
      \\
      &
      y \in \V(G \cup P \cup H)
      &
      \notag
      \\
      &
      \sources{G \cup P \cup H} \setminus \sources{H} \subseteq (\V(G \cup P) \setminus \V(H))
      &
      \notag
      \\
      &
      \forall \alpha \in \sources{G \cup P \cup H} \setminus \sources{H}.\; (\alpha,y) \notin R_{G \cup P \cup H}
      &
      \text{def. partial slice}
      \notag
      \\
      &
      \qedLocal
      y \in \sufficesR^*_{G \cup P \cup H}(\sources{H})
      &
      \notag
   \end{flalign}
\end{proof}
\begin{lemma}
   \label{lem:appendix:proof:conjugate:algorithms-suff:partial-slice-preserved}
   \item
   \begin{enumerate}
      \item $H,P,G$ a partial slice $ \iff H,P \cup E,G \edgeMinus E$ a partial slice
      \item If $\alpha \in \sources{G}$ then: $H,P,G$ a partial slice $\iff H \cup E, P \edgeMinus E, G$ a partial slice
   \end{enumerate}
\end{lemma}
\proofContext{partial-slice-lem}
\begin{proof}
   \small
   \begin{flalign}
      &
      \caseName{(1)}
      \notag
      \\
      &
      \subcase{\Rightarrow}
      \notag
      \\
      &
      H,P,G
      \text{ a partial slice}
      &
      \text{suppose}
      \notag
      \\
      &
      \forall (\alpha, \beta) \in \E(G) \cup \E(P).\; \beta \notin \V(H)
      &
      \text{def. partial slice}
      \notag
      \\
      &
      \E(G) \cup \E(P) = \E(G \edgeMinus E) \cup \E(P \cup E)
      &
      \notag
      \\
      &
      \qedLocal
      \forall (\alpha, \beta) \in \E(G \edgeMinus E) \cup \E(P \cup E).\; \beta \notin \V(H)
      &
      \notag
      \\
      &
      \subcase{\Leftarrow}
      \notag
      \\
      &
      H,P \cup E, G \edgeMinus E \text{ a partial slice}
      &
      \text{suppose}
      \notag
      \\
      &
      \forall (\alpha, \beta) \in \E(G \edgeMinus E) \cup \E(P \cup E).\; \beta \notin \V(H)
      &
      \text{def. partial slice}
      \notag
      \\
      &
      \E(G) \cup \E(P) = \E(G \edgeMinus E) \cup \E(P \cup E)
      &
      \notag
      \\
      &
      \qedLocal
      \forall (\alpha, \beta) \in \E(G) \cup \E(P).\; \beta \notin \V(H)
      &
      \notag
      \\
      &
      \caseName{(2)}
      \notag
      \\
      &
      \subcase{\Rightarrow}
      \notag
      \\
      &
      H,P,G
      \text{ a partial slice}
      &
      \text{suppose}
      \notag
      \\
      &
      \forall \beta \in \V(G).\;\nexists (\beta, \alpha) \in \E(G)
      &
      \text{$\alpha \in \sources{G}$}
      \locallabel{alpha-source}
      \\
      &
      \forall (x,y) \in \E(G). y \notin \V(H \cup E)
      &
      \text{def. partial slice, \localref{alpha-source}}
      \notag
      \\
      &
      \V(H \cup E) = \V(H) \cup \set{\alpha}
      &
      \text{def. $E$}
      \notag
      \\
      &
      (x, y) \in \E(P)
      &
      \text{suppose}
      \notag
      \\
      &
      \subcase{(x,y) \in \E(P) \edgeMinus E}
      \notag
      \\
      &
      y \notin \V(H)
      &
      \notag
      \\
      &
      y \neq \alpha
      &
      \text{$(x,y) \notin E$}
      \notag
      \\
      &
      \qedLocal
      y \notin \V(H \cup E)
      &
      \notag
      \\
      &
      \subcase{(x,y) \in E}
      \notag
      \\
      &
      \qedLocal
      (x,y) \notin \E(P) \edgeMinus E
      &
      \notag
      \\
      &
      \subcase{\Leftarrow}
      \notag
      \\
      &
      H \cup E, P \edgeMinus E, G \text{ a partial slice}
      &
      \text{suppose}
      \notag
      \\
      &
      \forall (\alpha, \beta) \in \E(G) \cup \E(P \edgeMinus E).\; \beta \notin \V(H \cup E)
      &
      \text{def. partial slice}
      \notag
      \\
      &
      \forall (\alpha, \beta) \in \E(G) \cup \E(P \edgeMinus E).\; \beta \notin \V(H)
      &
      \text{immediate}
      \notag
      \\
      &
      (\beta, \alpha) \in E \subset \E(P \edgeMinus E) \cup E = \E(P)
      &
      \text{suppose}
      \notag
      \\
      &
      \alpha \notin \V(H)
      &
      \text{def. $E$}
      \notag
      \\
      &
      \qedLocal
      \forall (\alpha', \beta') \in \E(P)
      &
      \notag
      \\
      &
      \qedLocal
      \forall (\alpha', \beta') \in \E(G) \cup \E(P).\; \beta' \notin \V(H)
      &
      \notag
   \end{flalign}
\end{proof}

\begin{lemma}
   \label{lem:appendix:proof:conjugate:algorithms-suff:aux}
   Suppose $H,P,G$ a partial slice, then:
   \begin{enumerate}
      \item $\sufficesR^*_{G \cup P \cup H}(\sources{H}) = \sufficesR^*_{(G \edgeMinus E)\cup (P \cup E) \cup H}(\sources{H})$
      \item $\sufficesR^*_{G \cup P \cup H}(\sources{H}) = \sufficesR^*_{G \cup (P \edgeMinus E) \cup (H \cup E)}(\sources{H \cup E})$
   \end{enumerate}
\end{lemma}
\proofContext{auxiliary-lemma}
\begin{proof}
   \begin{enumerate}
      \item $G \cup P \cup H = (G \edgeMinus E) \cup (P \cup E) \cup H$
      \item $G \cup P \cup H = G \cup (P \edgeMinus E) \cup (H \cup E)$, and $\sources{H} = \sources{H \cup E}$
   \end{enumerate} 
\end{proof}
\begin{proposition}
   \label{prop:appendix:proof:conjugate:algorithms-suff:correctness-sufficesE}
   Suppose $H,P,G$ a partial slice:
   \[ (\exists H'. \; H,P,G \sufficesE{} H' \text{ with }\V(H') = Y) \iff \sufficesR^*_{G \cup P \cup H}(\sources{H}) = Y\]
\end{proposition}

\proofContext{correctness-sufficesE}
\begin{proof}
   \small
   \begin{flalign}
      &
      H, P, G
      \text{ a partial slice}
      &
      \text{suppose}
      \\
      &
      \caseName{$\Rightarrow$ direction:}
      \notag
      \\
      &
      H, P, G \sufficesE{} H' \text{ with } \sinks{H'}=Y
      &
      \text{suppose, $\exists H'$}
      \notag
      \\
      &
      \text{By induction on derivation of $\sufficesE{}$.}
      &
      \notag
      \\
      &
      \subcaseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{done}},
               right={$\sources{G} \cap \V(P) \subseteq \sources{P} \wedge \sinks{H} \subseteq \sinks{G}$}
            ]
            {
               \strut
            }
            {
               H, P, G \sufficesE{} H
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \qedLocal
      \sufficesR^*_{G \cup P \cup H}(\sources{H}) = \V(H)
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:suffices-sources}}
      \notag
      \\
      &
      \subcaseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{pending}},
               right={$\alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset$}
            ]
            {
               H, P \cup E, G \edgeMinus E
               \sufficesE{}
               H'
            }
            {
               H, P, G
               \sufficesE{}
               H'
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      H, P \cup E, G \edgeMinus E
      \text{ a partial slice}
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:partial-slice-preserved}}
      \notag
      \\
      &
      \sufficesR^*_{(G \edgeMinus E) \cup (P \cup E) \cup H}(\sources{H}) = Y
      &
      \text{IH}
      \notag
      \\
      &
      \qedLocal
      \sufficesR^*_{G \cup P \cup H}(\sources{H}) = Y
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:aux}(i)}
      \notag
      \\
      &
      \subcaseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{extend}},
               right={$\alpha \in \sources{G} \wedge E = \inE{P}(\alpha) \neq \emptyset$}
            ]
            {
               H \cup E,
               P \edgeMinus E,
               G
               \sufficesE{}
               H'
            }
            {
               H, P, G
               \sufficesE{}
               H'
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      H \cup E, P \edgeMinus E, G
      \text{ a partial slice}
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:partial-slice-preserved}}
      \notag
      \\
      &
      \sufficesR^*_{G \cup (P \edgeMinus E) \cup (H \cup E)}(\sources{H \cup E})
      &
      \text{IH}
      \notag
      \\
      &
      \qedLocal
      \sufficesR^*_{G \cup P \cup H}(\sources{H})
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:aux}(ii)}
      \notag
      \\
      &
      \caseName{$\Leftarrow$ direction}
      \notag
      \\
      &
      \text{By strong lexicographical induction over $(\length{\E(G)},\length{\E(P)})$}
      &
      \notag
      \\
      &
      \sufficesR^*_{G \cup P \cup H}(\sources{H}) = Y
      &
      \text{suppose}
      \notag
      \\
      &
      \subcase{\sources{G} \cap \V(P) \subseteq \sources{P}\wedge \sinks{H} \subseteq \sinks{G}}
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{done}},
               right={$\sources{G} \cap \V(P) \subseteq \sources{P}\wedge \sinks{H} \subseteq \sinks{G}$}
            ]
            {
               \strut
            }
            {
               H, P, G \sufficesE{} H
            }
         \end{smathpar}
      }
      &
      \text{def. $\sufficesE{}$}
      \notag
      \\
      &
      \qedLocal

      \notag
      \\

      &

      \notag
      \\

      (\length{\E(G \edgeMinus E)}, \length{\E(P)} )< (\length{\E(G)}, \length{\E(P \cup E)})

      \notag
      \\

            {
               H \cup E,
               P \edgeMinus E,
               G
               \sufficesE{}
               H'
            }
            {
               H, P, G
               \sufficesE{}
               H'
            }
         \end{smathpar}
      }
      &
      \text{def. $\sufficesE{}$}
      \notag
      \\
      &
      \sufficesR^*_{G \cup P \cup H}(\sources{H}) = \sufficesR^*_{G \cup (P \edgeMinus E) \cup (H \cup E)}(\sources{H \cup E})
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:aux}(ii)}
      \\
      &
      \qedLocal
      \sufficesR^*_{G \cup P \cup H}(\sources{H}) = \V(H')\;\Lowlight{ = Y}

      \notag
   \end{flalign}
\end{proof}
