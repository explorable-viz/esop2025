\proofContext{sufficesAlg-suffices}
\begin{proof}
   \small
   \begin{flalign}
      &
      \caseName{$\Rightarrow$ direction:}
      &
      \notag
      \\
      &
      X \subseteq \sources{G}
      \text{ with }
      X \sufficesAlg{G} Y
      &
      \text{suppose}
      \locallabel{X-source-G}
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*
            {
               \exPair{X}{\emptyset}, \exPair{\emptyset}{\emptyset}, G \sufficesE{} H
            }
            {
               X \sufficesAlg{G} \sinks{H} \Lowlight{\;=Y}
            }
         \end{smathpar}
      }
      &
      \text{inversion; $\exists H$}
      \notag
      \\
      &
      \qedLocal
      \sufficesR_{G}(X) = \sufficesR^*_{G}(X) = Y
      &
      \text{(\localref{X-source-G}), \propref{appendix:proof:conjugate:algorithms-suff:correctness-sufficesE}}
      \notag
      \\
      &
      \caseName{$\Leftarrow$ direction: }
      \notag
      \\
      &
      X \subseteq \sources{G}
      \text{ with }
      \sufficesR_{G}(X) = \sufficesR^*_{G}(X) = Y
      &
      \text{suppose}
      \notag
      \\
      &
      (X, \emptyset), (\emptyset,\emptyset), G \sufficesE{} H
      &
      \text{\propref{appendix:proof:conjugate:algorithms-suff:correctness-sufficesE}; $\exists H$}
      \notag
      \\
      &
      \qedLocal
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*
            {
               \exPair{X}{\emptyset}, \exPair{\emptyset}{\emptyset}, G \sufficesE{} H
            }
            {
               X \sufficesAlg{G} \sinks{H} \Lowlight{\;=Y}
            }
         \end{smathpar}
      }
      &
      \text{def.~$\sufficesAlg{}$}
      \notag
   \end{flalign}
\end{proof}
\subsection{Correctness of $\sufficesE{}$}

\begin{definition}[$\sufficesR^*$ for a dependence graph]
   For a graph $G$ with output relation $R$, define:
   \[\sufficesR^*_G(X) \eqdef \sufficesR_{R}(X) = \set{y \in \sinks{G} \mid \nexists x \in \V(G)\setminus X.\; \exPair{x}{y} \in R}\]
\end{definition}

\begin{lemma}
   \label{lem:appendix:proof:conjugate:algorithms-suff:suff-subset-suffstar}
   $\sufficesR_G(X) \subseteq \sufficesR^*_{G}(X)$
\end{lemma}

\begin{lemma}
   \label{lem:appendix:proof:conjugate:algorithms-suff:suffices-sinks}
   If $X \subseteq \sinks{G}$ then $\sufficesR^*_{G}(X) = X$
\end{lemma}

\begin{lemma}
   \label{lem:appendix:proof:conjugate:algorithms-suff:aux}
   Suppose $\V(H), \V(P) \subseteq \V(G)$
   \begin{enumerate}
      \item If $\alpha \in \sinks{H}$ and $E = \outE{G}(\alpha) \neq \emptyset$ then $\sufficesR^*_{G \cup P}(\sinks{H}) = \sufficesR^*_{G \setminus_{\mathsf{E}} E \cup (P \cup E)}(\sinks{H})$
      \item If $\alpha \in \sources{G}$ and $E = \inE{P}(\alpha) \neq \emptyset$ then $\sufficesR^*_{G \cup P}(\sinks{H}) = \sufficesR^*_{G \cup (P \setminus_{\mathsf{E}} E)}(\sinks{H \cup E})$
   \end{enumerate}
\end{lemma}
\begin{proposition}
   \label{prop:appendix:proof:conjugate:algorithms-suff:correctness-sufficesE}
   For $\V(H), \V(P) \subseteq \V(G)$, and any $Y \subseteq \sinks{G}$:
   \[ (\exists H'. \; H,P,G \sufficesE{} H' \text{ with }\sinks{H'} = Y) \iff \sufficesR^*_{G \cup P}(\sinks{H}) = Y\]
\end{proposition}

\proofContext{correctness-sufficesE}
\begin{proof}
   \small
   \begin{flalign}
      &
      \V(H) \subseteq \V(G),\V(P) \subseteq \V(G)
      \text{ and }
      Y \subseteq \sinks{G}
      &
      \text{suppose}
      \notag
      \\
      &
      \caseName{$\Rightarrow$ direction:}
      \notag
      \\
      &
      H, P, G \sufficesE{} H' \text{ with } \sinks{H'}=Y
      &
      \text{suppose, $\exists H'$}
      \notag
      \\
      &
      \text{By induction on derivation of $\sufficesE{}$.}
      &
      \notag
      \\
      &
      \subcaseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{done}},
               right={$\sources{G} \subseteq \sources{P} \wedge \sinks{H} \subseteq \sinks{G}$}
            ]
            {
               \strut
            }
            {
               H, P, G \sufficesE{} H
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \qedLocal
      \sufficesR^*_{G \cup P}(\sinks{H}) = \sinks{H}
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:suffices-sinks}}
      \notag
      \\
      &
      \text{The above will need to be a little more fleshed out, since we need to establish}
      &
      \notag
      \\
      &
      \text{that in this situation unioning P and G will not create or lose sinks}
      &
      \notag
      \\
      &
      \subcaseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{pending}},
               right={$\alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset$}
            ]
            {
               H, P \cup E, G \setminus_{\mathsf{E}} E
               \sufficesE{}
               H'
            }
            {
               H, P, G
               \sufficesE{}
               H'
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \sufficesR^*_{(G \setminus_{\mathsf{E}} E) \cup (P \cup E)}(\sinks{H}) = Y
      &
      \text{IH}
      \notag
      \\
      &
      \qedLocal
      \sufficesR^*_{G \cup P}(\sinks{H}) = Y
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:aux}(i)}
      \notag
      \\
      &
      \subcaseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{extend}},
               right={$\alpha \in \sources{G} \wedge E = \inE{P}(\alpha) \neq \emptyset$}
            ]
            {
               H \cup E,
               P \setminus_{\mathsf{E}} E,
               G
               \sufficesE{}
               H'
            }
            {
               H, P, G
               \sufficesE{}
               H'
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \sufficesR^*_{G \cup (P \setminus_{\mathsf{E}} E)}(\sinks{H \cup E})
      &
      \text{IH}
      \notag
      \\
      &
      \qedLocal
      \sufficesR^*_{G \cup P}(\sinks{H})
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:aux}(ii)}
      \notag
      \\
      &
      \caseName{$\Leftarrow$ direction}
      \notag
      \\
      &
      \text{By strong lexicographical induction over $\exPair{\length{\E(G)}}{\length{\E(P)}}$}
      &
      \notag
      \\
      &
      \sufficesR^*_{G \cup P}(\sinks{H}) = Y
      &
      \text{suppose}
      \notag
      \\
      &
      \subcase{\sources{G} \subseteq \sources{G} \wedge \sinks{H} \subseteq \sinks{H}}
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{done}},
               right={$\sources{G} \subseteq \sources{P} \wedge \sinks{H} \subseteq \sinks{G}$}
            ]
            {
               \strut
            }
            {
               H, P, G \sufficesE{} H
            }
         \end{smathpar}
      }
      &
      \text{def. $\sufficesE{}$}
      \notag
      \\
      &
      \qedLocal
      \sinks{H} = \sufficesR^*_{G \cup P}(\sinks{H}) = Y
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:suff-subset-suffstar}}
      \notag
      \\
      &
      \subcase{\alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset}
      &
      \text{$\exists \alpha$, $\exists E$}
      \notag
      \\
      &
      \E(G \setminus_{\mathsf{E}} E) \subset \E(G)
      &
      \notag
      \\
      &
      H, P \cup E, G \setminus_{\mathsf{E}} E \sufficesE{} H'
      &
      \text{IH; $\exists H'$}
      \notag
      \\
      &
      \sinks{H'} = \sufficesR^*_{G \cup P}(\sinks{H})
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:aux}(i)}
      \notag
      \\
      &
      \qedLocal
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{pending}},
               right={$\alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset$}
            ]
            {
               H, P \cup E, G \setminus_{\mathsf{E}} E
               \sufficesE{}
               H'
            }
            {
               H, P, G
               \sufficesE{}
               H'
            }
         \end{smathpar}
      }
      &
      \text{def. $\sufficesE{}$}
      \notag
      \\
      &
      \subcase{\alpha \in \sources{G} \wedge E = \inE{P}(\alpha) \neq \emptyset}
      &
      \text{$\exists \alpha$, $\exists E$}
      \notag
      \\
   \end{flalign}
\end{proof}
