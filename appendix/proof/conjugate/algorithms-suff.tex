\subsection{Correctness of $\sufficesE{}$}

\begin{definition}[Partial Slice]
   Graphs $H,P,G$ contribute a \textit{partial slice} iff:
   \begin{itemize}
      \item $H,P,G$ are edge-disjoint
      \item $\forall (\alpha, \beta) \in \E(G) \cup \E(P)\; \beta \notin \V(H)$
   \end{itemize}
\end{definition}
\begin{definition}[Input Relation]
   For a graph $G$ with reachability relation $R$, define it's \text{input relation}
   $R_{I} \eqdef (\sources{G} \times \V{G}) \cap R$
\end{definition}

\begin{definition}[$\sufficesR^*$ for a dependence graph]
   For a graph $G$, with an input relation $R_{I}$, and a set of input vertices $X = \subseteq \sources{G}$:
   \[\sufficesR^*_{G}(X) \eqdef \sufficesR_{R_{I}}(X) = \set{y \in \V(G) \mid \nexists x \in \sources{G} \setminus X.\; (x,y) \in R_{I}}\]
\end{definition}

\begin{lemma}
   \label{lem:appendix:proof:conjugate:algorithms-suff:suff-subset-suffstar}
   $\sufficesR_G(X) = \sinks{G} \cap \sufficesR^*_{G}(X)$
\end{lemma}

\begin{lemma}
   \label{lem:appendix:proof:conjugate:algorithms-suff:suffices-sinks}
   If $X \subseteq \sinks{G}$ then $\sufficesR^*_{G}(X) = X$
\end{lemma}

\begin{lemma}
   If $H,P,G$ a partial slice with $\sources{G} \subseteq \sources{P}$ and $\sinks{H} \subseteq \sinks{G}$:
   \[ \sufficesR^*_{G \cup P \cup H}(\sources{H}) = \V(H) \]
\end{lemma}
\begin{lemma}
   \label{lem:appendix:proof:conjugate:algorithms-suff:aux}
   Suppose $\V(H), \V(P) \subseteq \V(G)$, with $H,G,P$ edge-disjoint, then:
   \begin{enumerate}
      \item If $\alpha \in \sinks{H}$ and $E = \outE{G}(\alpha) \neq \emptyset$ then \newline $\sufficesR^*_{G,P,H}(\V(H)) = \sufficesR^*_{G \edgeMinus E,P \cup E,H}(\sinks{H})$
      \item If $\alpha \in \sources{G}$ and $E = \inE{P}(\alpha) \neq \emptyset$, and $\inN{P}(\alpha) \subset \V(H)$, then \newline $\sufficesR^*_{G,P,H}(\V(H)) = \sufficesR^*_{G,P \edgeMinus E,H \cup E}(\V(H \cup E))$
   \end{enumerate}
\end{lemma}
\proofContext{auxiliary-lemma}
\begin{proof}
   By calculation
   \begin{flalign}
      &
      \subcase{\ref{lem:appendix:proof:conjugate:algorithms-suff:aux}(i)}
      \notag
      \\
      &
      G \cup P \cup H = (G \edgeMinus E) \cup (P \cup E) \cup H
      &
      \text{immediate}
      \notag
      \\
      &
      \sufficesR^*_{G,P,H}(\V(H)) = \set{y \in \demandedByR^*_{G \cup P \cup H}(\V(H)) \mid \nexists x \in \V(G)\setminus \demandedByR^*_{G \cup P \cup H}(\V(H)).\; (x,y) \in R}
      &
      \notag
      \\
      &
      = \set{y \in \demandedByR^*_{(G \edgeMinus) \cup (P \cup E) \cup H}(\V(H)) \mid \nexists x \in \V(G)\setminus \demandedByR^*_{(G \edgeMinus E) \cup (P \cup E) \cup H}(\V(H)).\; (x,y) \in R}
      &
      \notag
      \\
      &
      = \sufficesR^*_{G \edgeMinus E, P \cup E, H}(\V(H))
      &
      \notag
      \\
      &
      \subcase{\ref{lem:appendix:proof:conjugate:algorithms-suff:aux}(ii)}
      \notag
      \\
      &
      \demandedByR^*_{G \cup P \cup H}(\V(H)) = \demandedByR^*_{G \cup (P \edgeMinus E) \cup (H \cup E)}(\V(H \cup E))
      &
      \text{immediate}
      \notag
      \\
      &
      \sufficesR^*_{G,P,H}(\V(H)) = \set{y \in \demandedByR^*_{G \cup P \cup H}(\V(H)) \mid \nexists x \in \V(G)\setminus \demandedByR^*_{G \cup P \cup H}(\V(H)).\; (x,y) \in R}
      &
      \notag
      \\
      &
      = \set{y \in \demandedByR^*_{G \cup (P \edgeMinus E) \cup (H \cup E)}(\V(H \cup E)) \mid \nexists x \in \V(G)\setminus \demandedByR^*_{G \cup (P \edgeMinus E) \cup (H \cup E)}(\V(H \cup E)).\; (x,y) \in R}
      &
      \notag
      \\
      &
      = \sufficesR^*_{G,P \edgeMinus E, H \cup E}(\V(H \cup E))
      &
      \notag
   \end{flalign}
\end{proof}
\begin{proposition}
   \label{prop:appendix:proof:conjugate:algorithms-suff:correctness-sufficesE}
   Suppose $H,P,G$ a partial slice:
   \[ (\exists H'. \; H,P,G \sufficesE{} H' \text{ with }\V(H') = Y) \iff \sufficesR^*_{G \cup P \cup H}(\sources{H}) = Y\]
\end{proposition}

\proofContext{correctness-sufficesE}
\begin{proof}
   \small
   \begin{flalign}
      &
      \V(H) \subseteq \V(G),\V(P) \subseteq \V(G)
      \text{ and }
      Y \subseteq \sinks{G}
      &
      \text{suppose}
      \notag
      \\
      &
      \caseName{$\Rightarrow$ direction:}
      \notag
      \\
      &
      H, P, G \sufficesE{} H' \text{ with } \sinks{H'}=Y
      &
      \text{suppose, $\exists H'$}
      \notag
      \\
      &
      \text{By induction on derivation of $\sufficesE{}$.}
      &
      \notag
      \\
      &
      \subcaseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{done}},
               right={$\sources{G} \subseteq \sources{P} \wedge \sinks{H} \subseteq \sinks{G}$}
            ]
            {
               \strut
            }
            {
               H, P, G \sufficesE{} H
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \qedLocal
      \sufficesR^*_{G \cup P}(\sinks{H}) = \sinks{H}
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:suffices-sinks}}
      \notag
      \\
      &
      \text{The above will need to be a little more fleshed out, since we need to establish}
      &
      \notag
      \\
      &
      \text{that in this situation unioning P and G will not create or lose sinks}
      &
      \notag
      \\
      &
      \subcaseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{pending}},
               right={$\alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset$}
            ]
            {
               H, P \cup E, G \edgeMinus E
               \sufficesE{}
               H'
            }
            {
               H, P, G
               \sufficesE{}
               H'
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \sufficesR^*_{(G \edgeMinus E) \cup (P \cup E)}(\sinks{H}) = Y
      &
      \text{IH}
      \notag
      \\
      &
      \qedLocal
      \sufficesR^*_{G \cup P}(\sinks{H}) = Y
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:aux}(i)}
      \notag
      \\
      &
      \subcaseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{extend}},
               right={$\alpha \in \sources{G} \wedge E = \inE{P}(\alpha) \neq \emptyset$}
            ]
            {
               H \cup E,
               P \edgeMinus E,
               G
               \sufficesE{}
               H'
            }
            {
               H, P, G
               \sufficesE{}
               H'
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \sufficesR^*_{G \cup (P \edgeMinus E)}(\sinks{H \cup E})
      &
      \text{IH}
      \notag
      \\
      &
      \qedLocal
      \sufficesR^*_{G \cup P}(\sinks{H})
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:aux}(ii)}
      \notag
      \\
      &
      \caseName{$\Leftarrow$ direction}
      \notag
      \\
      &
      \text{By strong lexicographical induction over $(\length{\E(G)},\length{\E(P)})$}
      &
      \notag
      \\
      &
      \sufficesR^*_{G \cup P}(\sinks{H}) = Y
      &
      \text{suppose}
      \notag
      \\
      &
      \subcase{\sources{G} \subseteq \sources{G} \wedge \sinks{H} \subseteq \sinks{H}}
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{done}},
               right={$\sources{G} \subseteq \sources{P} \wedge \sinks{H} \subseteq \sinks{G}$}
            ]
            {
               \strut
            }
            {
               H, P, G \sufficesE{} H
            }
         \end{smathpar}
      }
      &
      \text{def. $\sufficesE{}$}
      \notag
      \\
      &
      \qedLocal
      \sinks{H} = \sufficesR^*_{G \cup P}(\sinks{H}) = Y
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:suff-subset-suffstar}}
      \notag
      \\
      &
      \subcase{\alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset}
      &
      \text{$\exists \alpha$, $\exists E$}
      \notag
      \\
      &
      (\length{\E(G \edgeMinus E)}, \length{\E(P)} )< (\length{\E(G)}, \length{\E(P \cup E)})
      &
      \notag
      \\
      &
      H, P \cup E, G \edgeMinus E \sufficesE{} H'
      &
      \text{IH; $\exists H'$}
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{pending}},
               right={$\alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset$}
            ]
            {
               H, P \cup E, G \edgeMinus E
               \sufficesE{}
               H'
            }
            {
               H, P, G
               \sufficesE{}
               H'
            }
         \end{smathpar}
      }
      &
      \text{def. $\sufficesE{}$}
      \notag
      \\
      &
      \sufficesR^*_{(G \edgeMinus E) \cup (P \cup E)}(\sinks{H}) = \sufficesR^*_{G \cup P}(\sinks{H})
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:aux}(i)}
      \notag
      \\
      &
      \qedLocal
      \sufficesR^*_{G \cup P}(\sinks{H}) = \sinks{H'}\;\Lowlight{ = Y}
      &
      \text{IH}
      \notag
      \\
      &
      \subcase{\alpha \in \sources{G} \wedge E = \inE{P}(\alpha) \neq \emptyset}
      &
      \text{$\exists \alpha$, $\exists E$}
      \notag
      \\
      &
      (\length{\E(G)},\length{\E(P \edgeMinus E)}) < (\length{\E(P)},\length{\E(P)})
      &
      \notag
      \\
      &
      H \cup E, P \edgeMinus E, G \sufficesE{} H'
      &
      \text{IH, $\exists H'$}
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{extend}},
               right={$\alpha \in \sources{G} \wedge E = \inE{P}(\alpha) \neq \emptyset$}
            ]
            {
               H \cup E,
               P \edgeMinus E,
               G
               \sufficesE{}
               H'
            }
            {
               H, P, G
               \sufficesE{}
               H'
            }
         \end{smathpar}
      }
      &
      \text{def. $\sufficesE{}$}
      \notag
      \\
      &
      \sufficesR^*_{G \cup P}(\sinks{H}) = \sufficesR^*_{G \cup (P \edgeMinus E)}(\sinks{H \cup E})
      &
      \text{\lemref{appendix:proof:conjugate:algorithms-suff:aux}(ii)}
      \\
      &
      \qedLocal
      \sufficesR^*_{G \cup P}(\sinks{H}) = \sinks{H'}\;\Lowlight{ = Y}
      &
      \text{IH}
      \notag
   \end{flalign}
\end{proof}
