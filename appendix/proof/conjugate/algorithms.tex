\subsection{Correctness of $\demandedByV{}$}

Reachability in $G$ induces a superset of the IO relation of $G$.

\begin{definition}
For any dependence graph $G$ with reachability relation $R$, define the \emph{output} relation of $G$ to be $R
\cap \V(G) \times \sinks{G}$.
\end{definition}

\begin{definition}[$\demandedByR^*$ for a dependence graph]
   For a graph $G$ with output relation $R$, define:
   \begin{itemize}
      \item $\demandedByR^*_G \eqdef \demandedByR_{R}: \powerset{\V(G)} \to \powerset{\sinks{G}}$ \hfill (demanded by$^*$)
   \end{itemize}
\end{definition}

\begin{lemma}
   \label{lem:appendix:proof:conjugate:algorithms:sinks-fixpoint}
   If $X \subseteq \sinks{G}$ then $\demandedByR_{G}(X) = X$.
\end{lemma}

\begin{proposition}
   \label{prop:appendix:proof:conjugate:algorithms:correctness-demByV}
   For any partial slice $H$ of $G$:
   \[(\exists H'.\;H, G \demandedByV{} H'\text{ with }\sinks{H'} = Y) \iff \demandedByR_G(\sinks{H}) = Y\]
\end{proposition}

\proofContext{correctness-demByV}
\begin{proof}
   \begin{flalign}
      &
      \caseName{$\Rightarrow$ direction:}
      \notag
      \\
      &
      \text{By induction on derivation of $\demandedByV{}$}
      &
      \notag
      \\
      &
      \subcaseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
            lab={\ruleName{done}},
            right={$\sinks{H} \subseteq \sinks{G}$}
            ]
            {
               \strut
            }
            {
               H, G \demandedByV{} H
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \qedLocal
      \demandedByR_G(\sinks{H}) = \sinks{H}
      &
      \text{$\sinks{H} \subseteq \sinks{G}$}
      \notag
      \\
      &
      \subcaseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{extend}},
               right={$\alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset$}
            ]
            {
               H \cup E, G \setminus_{\mathsf{E}} E
               \demandedByV{}
               H'
            }
            {
               H, G \demandedByV{} H'
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \demandedByR_{G \setminus_{\mathsf{E}} E}(\sinks{H \cup E}) = \sinks{H'}
      &
      \text{IH}
      \notag
      \\
      &
      (G \setminus_{\mathsf{E}} E) \cup (H \cup E) = G \cup H
      &
      \notag
      \\
      &
      \qedLocal
      \demandedByR_{G}(\sinks{H}) = \sinks{H'}
      &
      \text{\lemref{appendix:proof:conjugate:algorithms:sinks-fixpoint}}
      \notag
      \\
      &
      \caseName{$\Leftarrow$ direction:}
      \notag
      \\
      &
      \text{By strong lexicographical induction over $(\length{\E(G)}, \length{\sinks{H} \setminus \sinks{G}}) \subseteq \mathbb{N}^2$}
      &
      \notag
      \\
      &
      H\text{ a partial slice of }G\text{ with }\demandedByR_G(\sinks{H}) = Y
      &
      \text{suppose}
      \locallabel{assumption}
      \\
      \subcase{\length{\sinks{H} \setminus \sinks{G}} = 0}
      \notag
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
            lab={\ruleName{done}},
            right={$\sinks{H} \subseteq \sinks{G}$}
            ]
            {
               \strut
            }
            {
               H, G \demandedByV{} H
            }
         \end{smathpar}
      }
      &
      \text{def.~$\demandedByV{}$}
      \notag
      \\
      &
      \qedLocal
      \sinks{H} = \demandedByR_{G}(\sinks{H}) = Y
      &
      \text{\lemref{appendix:proof:conjugate:algorithms:sinks-fixpoint}; (\localref{assumption})}
      \notag
      \\
      \subcase{\length{\sinks{H} \setminus \sinks{G}} > 0}
      \notag
      &
      \alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset
      &
      \text{$\exists \alpha$, $\exists E$}
      \locallabel{exists-alpha}
      \\
      &
      \E(G \setminus_{\mathsf{E}} E) \subset \E(G)
      &
      \notag
      \\
      &
      H \cup E\text{ a partial slice of }G \setminus_{\mathsf{E}} E
      &
      \notag
      \\
      &
      H \cup E, G \setminus_{\mathsf{E}} E \demandedByV{} H'
      \text{ with }
      \sinks{H'} = \demandedByR_{G \setminus_{\mathsf{E}} E}(\sinks{H \cup E})
      &
      \text{IH; $\exists H'$}
      \notag
      \\
      &
      (G \setminus_{\mathsf{E}} E) \cup (H \cup E) = G \cup H
      &
      \notag
      \\
      &
      \sinks{H'} = \demandedByR_{G}(\sinks{H})
      &
      \notag
      \\
      &
      \qedLocal
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{extend}},
               right={$\alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset$}
            ]
            {
               H \cup E, G \setminus_{\mathsf{E}} E
               \demandedByV{}
               H'
            }
            {
               H, G \demandedByV{} H'
            }
         \end{smathpar}
      }
      &
      \text{def.~$\demandedByV{}$}
      \notag
   \end{flalign}
\end{proof}
