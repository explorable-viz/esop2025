\subsection{Correctness of $\demandedByV{}$}

The \emph{output} relation of a graph $G$ with reachability relation $R$, defined as $R \cap \V(G) \times
\sinks{G}$, gives rise to the following extension of $\demandedByR_{G}$:

\begin{definition}[$\demandedByR^*$ for a dependence graph]
   For a graph $G$ with output relation $R$, define:
   \begin{itemize}
      \item $\demandedByR^*_G \eqdef \demandedByR_{R}: \powerset{\V(G)} \to \powerset{\sinks{G}}$ \hfill (demanded by$^*$)
   \end{itemize}
\end{definition}

\begin{lemma}
   $\demandedByR_{G} \subseteq \demandedByR^*_{G}(X)$.
\end{lemma}

\begin{lemma}
   \label{lem:appendix:proof:conjugate:algorithms:sinks-fixpoint}
   If $X \subseteq \sinks{G}$ then $\demandedByR^*_{G}(X) = X$.
\end{lemma}

\begin{lemma}
   \label{lem:appendix:proof:conjugate:algorithms:edge-conservation}
   Suppose $\V(H) \subseteq \V(G)$ and $\E(H) \cap \E(G) = \emptyset$. For any $\alpha \in \sinks{H}$ and any
   $E = \outE{G}(\alpha) \neq \emptyset$ we have
   \[\demandedByR^*_G(\sinks{H}) = \demandedByR^*_{G \setminus_{\mathsf{E}} E} (\sinks{H \cup E})\]
\end{lemma}

\begin{proposition}
   \label{prop:appendix:proof:conjugate:algorithms:correctness-demByV}
   For any partial slice $H$ of $G$:
   \[(\exists H'.\;H, G \demandedByV{} H'\text{ with }\sinks{H'} = Y) \iff \demandedByR^*_G(\sinks{H}) = Y\]
\end{proposition}

\proofContext{correctness-demByV}
\begin{proof}
   \small
   \begin{flalign}
      &
      \caseName{$\Rightarrow$ direction:}
      \notag
      \\
      &
      \text{By induction on derivation of $\demandedByV{}$.}
      &
      \notag
      \\
      &
      \subcaseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
            lab={\ruleName{done}},
            right={$\sinks{H} \subseteq \sinks{G}$}
            ]
            {
               \strut
            }
            {
               H, G \demandedByV{} H
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \qedLocal
      \demandedByR^*_G(\sinks{H}) = \sinks{H}
      &
      \text{\lemref{appendix:proof:conjugate:algorithms:sinks-fixpoint}}
      \notag
      \\
      &
      \subcaseDerivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{extend}},
               right={$\alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset$}
            ]
            {
               H \cup E, G \setminus_{\mathsf{E}} E
               \demandedByV{}
               H'
            }
            {
               H, G \demandedByV{} H'
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \demandedByR^*_{G \setminus_{\mathsf{E}} E}(\sinks{H \cup E}) = \sinks{H'}
      &
      \text{IH}
      \notag
      \\
      &
      \qedLocal
      \demandedByR^*_{G}(\sinks{H}) = \sinks{H'}
      &
      \text{\lemref{appendix:proof:conjugate:algorithms:edge-conservation}}
      \notag
      \\
      &
      \caseName{$\Leftarrow$ direction:}
      \notag
      \\
      &
      \text{By strong induction (lexicographically) over $(\length{\E(G)}, \length{\sinks{H} \setminus \sinks{G}}) \subseteq \mathbb{N}^2$.}
      &
      \notag
      \\
      &
      H\text{ a partial slice of }G\text{ with }\demandedByR^*_G(\sinks{H}) = Y
      &
      \text{suppose}
      \locallabel{assumption}
      \\
      \subcase{\length{\sinks{H} \setminus \sinks{G}} = 0}
      \notag
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
            lab={\ruleName{done}},
            right={$\sinks{H} \subseteq \sinks{G}$}
            ]
            {
               \strut
            }
            {
               H, G \demandedByV{} H
            }
         \end{smathpar}
      }
      &
      \text{def.~$\demandedByV{}$}
      \notag
      \\
      &
      \qedLocal
      \sinks{H} = \demandedByR^*_{G}(\sinks{H}) = Y
      &
      \text{\lemref{appendix:proof:conjugate:algorithms:sinks-fixpoint}; (\localref{assumption})}
      \notag
      \\
      \subcase{\length{\sinks{H} \setminus \sinks{G}} > 0}
      \notag
      &
      \alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset
      &
      \text{$\exists \alpha$, $\exists E$}
      \locallabel{exists-alpha}
      \\
      &
      \E(G \setminus_{\mathsf{E}} E) \subset \E(G)
      &
      \notag
      \\
      &
      H \cup E\text{ a partial slice of }G \setminus_{\mathsf{E}} E
      &
      \notag
      \\
      &
      H \cup E, G \setminus_{\mathsf{E}} E \demandedByV{} H'
      \text{ with }
      \sinks{H'} = \demandedByR^*_{G \setminus_{\mathsf{E}} E}(\sinks{H \cup E})
      &
      \text{IH; $\exists H'$}
      \notag
      \\
      &
      (G \setminus_{\mathsf{E}} E) \cup (H \cup E) = G \cup H
      &
      \notag
      \\
      &
      \sinks{H'} = \demandedByR^*_{G}(\sinks{H})
      &
      \notag
      \\
      &
      \qedLocal
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               lab={\ruleName{extend}},
               right={$\alpha \in \sinks{H} \wedge E = \outE{G}(\alpha) \neq \emptyset$}
            ]
            {
               H \cup E, G \setminus_{\mathsf{E}} E
               \demandedByV{}
               H'
            }
            {
               H, G \demandedByV{} H'
            }
         \end{smathpar}
      }
      &
      \text{def.~$\demandedByV{}$}
      \notag
   \end{flalign}
\end{proof}
