Assume the following:
\begin{equation}
   \gamma_1 \desugarFwdS \gamma_2
\end{equation}
\begin{equation}
   g \desugarFwdS \rho
\end{equation}
\begin{proof}
   \begin{flalign}
      \intertext{\crossrule}
      %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
      &
      \subcaseName{$\closeDefsS$-then-$\desugarFwdS$}
      &
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*
            {
               g = \seq{\bind{x}{\mu}}
               \\
               v_i = \exClosure{\gamma_1}{g}{\mu_i}
               \quad
               \forall i \le \length{\seq{x}}
            }
            {
               \gamma_1, g \closeDefsS \set{\seq{\bind{x}{v}}}
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[]
            {
               \rho = \set{\seq{\bind{x}{\sigma}}}
               \\
               v'_i = \exClosureNew{\gamma_2}{\rho}{\sigma_i}
               \quad
               (\forall i \numleq \length{\seq{x}})
            }
            {
               \gamma_2, \rho
               \closeDefsFwdR
               \set{\seq{\bind{x}{v'}}}
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \forall i \numleq \length{\seq{x}} \; v_i \desugarFwdS v'_i
      &
      \text{$g \desugarFwdS \rho \opAnd \gamma_1 \desugarFwdS \gamma_2$}
      \notag
      \\
      &
      \qedLocal
      \gamma_1' \desugarFwdS \gamma_2'
      &
      \text{$\desugarFwdS$-env}
      \notag
      \\
      &
      \subcaseName{$\desugarFwdS$-then-$\closeDefsS$}
      &
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[]
            {
               \rho = \set{\seq{\bind{x}{\sigma}}}
               \\
               v'_i = \exClosureNew{\gamma_2}{\rho}{\sigma_i}
               \quad
               (\forall i \numleq \length{\seq{x}})
            }
            {
               \gamma_2, \rho
               \closeDefsFwdR
               \set{\seq{\bind{x}{v'}}}
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*
            {
               g = \seq{\bind{x}{\mu}}
               \\
               v_i = \exClosure{\gamma_1}{g}{\mu_i}
               \quad
               \forall i \le \length{\seq{x}}
            }
            {
               \gamma_1, g \closeDefsS \set{\seq{\bind{x}{v}}}
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \forall i\numleq \length{\seq{x}}\; v_i \desugarFwdS v'_i
      &
      \text{$g \desugarFwdS \rho \opAnd \gamma_1 \desugarFwdS \gamma_2$}
      \notag
      \\
      &
      \qedLocal
      \gamma_1' \desugarFwdS \gamma_2'
      &
      \text{$\desugarFwdS$-env}
      \notag
   \end{flalign}
\end{proof}
