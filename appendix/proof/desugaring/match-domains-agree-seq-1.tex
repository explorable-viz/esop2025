Induction on $\match$ derivation.
\begin{proof}
\small
\begin{flalign}
   &
   \seq{u}, \seq{p} \match \gamma
   &
   \text{suppose}
   \notag
   \\
   &
   \seq{u} \desugar \seq{v}
   &
   \text{suppose}
   \notag
   \\
   &
   \kappa' = \elimArgsF(\seq{p}, \kappa)
   &
   \text{suppose}
   \locallabel{suppose-kappa}
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-empty}}
          ]
          {
             \strut
          }
          {
             \seq{u} = \seqEmpty, \seq{p} = \seqEmpty \match \varnothing = \gamma
          }
       \end{smathpar}
   }
   &
   \locallabel{match-empty}
   \\
   &
   \kappa' = \elimArgsF(\seqEmpty,\kappa) = \kappa
   &
   \text{(\localref{suppose-kappa}), Def.~\elimArgsF}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-seq-empty}}
         ]
         {
            \strut
         }
         {
            \seqEmpty, \kappa \match \varnothing, \kappa
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\match$-seq}}
         ]
         {
            u_1, p_1 \match \gamma_1
            \\
            \seq{u}', \seq{p}' \match \gamma_2
         }
         {
            \seq{u} = u_1 \cons \seq{u}', \seq{p} = p_1 \cons \seq{p}' \match \gamma_1 \disjunion \gamma_2 = \gamma
         }
      \end{smathpar}
   }
   &
   \locallabel{match-constr}
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\desugar$-seq}}]
         {
            u_1 \desugar v_1
            \\
            \seq{u}' \desugar \seq{v}'
         }
         {
            u_1 \cons \seq{u}' \desugar v_1 \cons \seq{v}'
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \locallabel{desugar-constr}
   \\
   &
   \sigma \eqdef \elimArgsF(p \cons o,\kappa)
   &
   \notag
   \\
   &
   \mkElimF((\pattListNext{p}{o}), \kappa) = \elimConstr{\elimBind{\cCons}{\sigma}}
   &
   \text{Def.~\mkElimF}
   \notag
   \\
   &
   (v_1, v_2), \sigma \match \gamma', \kappa
   \textit{ and }
   \gamma \desugar \gamma'
   \quad
   (\exists \gamma')
   &
   \text{(\localref{match-constr}), (\localref{desugar-constr}), \lemref{match-domains-agree-seq} (1)}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\match$-constr}}]
         {
            (v_1, v_2), \sigma \match \gamma', \kappa
         }
         {
            \exConstr{\cCons}{v_1, v_2}, \elimConstr{\elimBind{\cCons}{\sigma}}
            \match
            \gamma', \kappa
         }
      \end{smathpar}
   }
   &
   \notag
\end{flalign}
\end{proof}
