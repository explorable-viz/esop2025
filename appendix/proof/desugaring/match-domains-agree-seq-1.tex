Induction on $\match$ derivation.
\begin{proof}
\small
\begin{flalign}
   &
   \seq{u}, \seq{p} \match \gamma
   &
   \text{suppose}
   \notag
   \\
   &
   \seq{u} \desugar \seq{v}
   &
   \text{suppose}
   \locallabel{suppose-seq-u}
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-seq-empty}}
          ]
          {
             \strut
          }
          {
             \seq{u} = \seqEmpty, \seq{p} = \seqEmpty \match \varnothing = \gamma
          }
       \end{smathpar}
   }
   &
   \locallabel{match-empty}
   \\
   &
   \elimArgsF(\seqEmpty,\kappa) = \kappa
   &
   \text{Def.~\elimArgsF}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-seq-empty}}
         ]
         {
            \strut
         }
         {
            \seqEmpty, \kappa \match \varnothing, \kappa
         }
      \end{smathpar}
   }
   &
   \notag
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-seq-non-empty}}
         ]
         {
            u, p \match \gamma_1
            \\
            \seq{u}', \seq{p}' \match \gamma_2
         }
         {
            \seq{u} = u \cons \seq{u}', \seq{p} = p \cons \seq{p}' \match \gamma_1 \disjunion \gamma_2 = \gamma
         }
      \end{smathpar}
   }
   &
   \locallabel{match-seq}
   \text{(\localref{suppose-seq-u})}
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\desugar$-seq-non-empty}}]
         {
            u \desugar v
            \\
            \seq{u}' \desugar \seq{v}'
         }
         {
            u \cons \seq{u}' \desugar v \cons \seq{v}' = \seq{v}
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \locallabel{desugar-seq}
   \\
   &
   \kappa' \eqdef \elimArgsF(\seq{p}',\kappa)
   &
   \notag
   \\
   &
   \seq{v}', \kappa' \match \gamma_2', \kappa
   \textit{ with }
   \gamma_2 \desugar \gamma_2'
   \quad (\exists \gamma_2')
   &
   \text{(\localref{match-seq}), (\localref{desugar-seq}), IH}
   \locallabel{match-v-seq}
   \\
   &
   \sigma \eqdef \mkElimF(p, \kappa')
   &
   \notag
   \\
   &
   v, \sigma \match \gamma_1', \kappa'
   \textit{ with }
   \gamma_1 \desugar \gamma_1'
   \quad (\exists \gamma_1')
   &
   \text{(\localref{match-seq}), (\localref{desugar-seq}), \lemref{match-domains-agree}}
   \locallabel{match-v}
   \\
   &
   \elimArgsF(p \cons \seq{p}', \kappa) = \sigma
   &
   \text{Def.~\mkElimF}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\match$-seq-non-empty}}
         ]
         {
            v, \sigma \match \gamma_1', \kappa'
            \\
            \seq{v}', \kappa' \match \gamma_2', \kappa
         }
         {
            v \cons \seq{v}',
            \sigma
            \match
            \gamma_1' \disjunion \gamma_2',
            \kappa
         }
        \end{smathpar}
   }
   &
   \text{(\localref{match-v}), (\localref{match-v-seq})}
   \notag
   \\
   &
   \qedLocal
   \gamma = \gamma_1 \disjunion \gamma_2 \desugar \gamma_1' \disjunion \gamma_2' = \gamma'
   &
   \text{(\localref{match-v}), (\localref{match-v-seq})}
   \notag
\end{flalign}
\end{proof}
