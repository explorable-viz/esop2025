Induction on surface-language $\match$ derivation.
\begin{proof}
\small
\begin{flalign}
   &
   \seq{u} \desugar \seq{v}
   &
   \text{suppose}
   \locallabel{us-desugar-vs}
   \\
   &
   \seq{(\pi, k)} \desugar \kappa
   &\text{suppose}
   \\
   &
   \seq{u}, \seq{(\pi, k)} \match \gamma, s
   &\text{suppose}
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-var}}
          ]
          {
             \seq{u}', (\seqRange{(\pi_1, k_1)}{(\pi_j, k_j)})
             \match \gamma^\dagger, s
          }
          {
             \Lowlight{\seq{u} =\;}
             u \cons \seq{u}', (\seqRange{(\pattVar{x} \cons \pi_1, k_1)}
                                         {(\pattVar{x} \cons \pi_j, k_j)})
             \match
             \set{\bind{x}{v}} \disjunion \gamma^\dagger\Lowlight{\;= \gamma}, s
          }
       \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\desugar$-clauses-var}}
         ]
         {
            \seqRange{(\pi_1, k_1)}{(\pi_j, k_j)} \desugar \kappa'
         }
         {
            \seqRange{(\pattVar{x} \cons \pi_1, k_1)}{(\pattVar{x} \cons \pi_j, k_j)}
            \desugar
            (\elimVar{x}{\kappa'})\Lowlight{\,= \kappa}
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
   \\
   &
   \seq{v} = v \cons \seq{v}'\text{ with }u \desugar v\text{ and }\seq{u}' \desugar \seq{v}'
   \quad
   (\exists v, \seq{v}')
   &
   \text{(\localref{us-desugar-vs})}
   \notag
   \\
   &
   \seq{v}', \kappa \match \gamma^\ddagger, e\textit{ with }\gamma^\dagger \desugar \gamma^\ddagger
   \quad
   (\exists\gamma^\ddagger, e)
   &
   \text{IH}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\match$-var}}]
         {
            \seq{v}', \kappa' \match \gamma^\ddagger, e
         }
         {
            v \cons \seq{v};, \elimVar{x}{\kappa'} \match \set{\bind{x}{v}} \disjunion \gamma^\dagger, e
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \qedLocal
   \set{\bind{x}{u}} \disjunion \gamma^\dagger
   \desugar
   \set{\bind{x}{v}} \disjunion \gamma^\ddagger
   &
   \text{($u \desugar v$)}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-done-curried}}
          ]
          {
             \strut
          }
          {
             \Lowlight{\seq{u} =\;}
             \seqEmpty, (\seqRange{(\seqEmpty, \clause{p_1 \cons \pi_1}{s_1})}
                                  {(\seqEmpty, \clause{p_j \cons \pi_j}{s_j})}) \\
             \match
             \envEmpty\Lowlight{\;=\gamma}, \exFun{(\seq{\clause{p \cons \pi}{s}})}\Lowlight{\;=s}
          }
       \end{smathpar}
   }
   &
   \notag
   \\
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\desugar$-clauses-curried}}
         ]
         {
            \seqRange{(p_1, \clause{\pi_1}{s_1})}{(p_j, \clause{\pi_j}{s_j})}
            \desugar
            \sigma
         }
         {
            \seqRange{(\seqEmpty, \clause{p_1 \cons \pi_1}{s_1})}
                     {(\seqEmpty, \clause{p_j \cons \pi_j}{s_j})}
            \desugar
            \exFun{\sigma}
         }
       \end{smathpar}
   }
   &
   \text{inversion}
   \notag
\end{flalign}
\end{proof}
