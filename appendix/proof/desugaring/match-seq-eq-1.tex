Induction on surface-language $\match$ derivation.
\begin{proof}
\small
\begin{flalign}
   &
   \seq{u} \desugar \seq{v}
   &
   \text{suppose}
   \locallabel{us-desugar-vs}
   \\
   &
   \seq{(\pi, k)} \desugar \kappa
   &
   \text{suppose}
   \notag
   \\
   &
   \seq{u}, \seq{(\pi, k)} \match \gamma, s
   &
   \text{suppose}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\match$-done-uncurried}}
          ]
          {
             \strut
          }
          {
             \Lowlight{\seq{u} =\;}
             \seqEmpty, (\seqEmpty, \clause{\seqEmpty}{s}) \match \envEmpty\Lowlight{\;= \gamma}, s
          }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugar$-clauses-done}}
         ]
         {
            s \desugar e
         }
         {
            (\seqEmpty, \clause{\seqEmpty}{s}) \desugar e
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \locallabel{done-uncurried-inversion}
   \\
   &
   \seq{v} = \seqEmpty
   &
   \text{(\localref{us-desugar-vs})}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\match$-done}}
         ]
         {
            \strut
         }
         {
            \Lowlight{\seq{v} =\;}
            \seqEmpty, e \match \envEmpty, e
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \qedLocal
   \envEmpty \desugar \envEmpty\text{ and }s \desugar e
   &
   \text{(\localref{done-uncurried-inversion})}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-done-curried}}
          ]
          {
             \strut
          }
          {
             \Lowlight{\seq{u} =\;}
             \seqEmpty, (\seqRange{(\seqEmpty, \clause{p_1 \cons \pi_1}{s_1})}
                                  {(\seqEmpty, \clause{p_j \cons \pi_j}{s_j})}) \\
             \match
             \envEmpty\Lowlight{\;=\gamma}, \exFun{(\seq{\clause{p \cons \pi}{s}})}\Lowlight{\;=s}
          }
       \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\desugar$-clauses-curried}}
         ]
         {
            \seqRange{(p_1, \clause{\pi_1}{s_1})}{(p_j, \clause{\pi_j}{s_j})}
            \desugar
            \sigma
         }
         {
            \seqRange{(\seqEmpty, \clause{p_1 \cons \pi_1}{s_1})}
                     {(\seqEmpty, \clause{p_j \cons \pi_j}{s_j})}
            \desugar
            \exFun{\sigma}\Lowlight{\;=\kappa}
         }
       \end{smathpar}
   }
   &
   \text{inversion}
   \locallabel{done-curried-inversion}
   \\
   &
   \seq{v} = \seqEmpty
   &
   \text{(\localref{us-desugar-vs})}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-done}}
         ]
         {
            \strut
         }
         {
            \Lowlight{\seq{v} =\;}
            \seqEmpty, \exFun{\sigma} \match \envEmpty, \exFun{\sigma}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \qedLocal
   \envEmpty \desugar \envEmpty
   &
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\desugar$-lambda}}]
         {
            \inferrule*[
               left={\ruleName{$\desugar$-fun}}
            ]
            {
               \seqRange{(p_1, \clause{\pi_1}{s_1})}
                        {(p_j, \clause{\pi_1}{s_j})}
               \desugar \sigma
            }
            {
               \seq{\clause{p \cons \pi}{s}}
               \desugar
               \sigma
            }
         }
         {
            \exFun{(\seq{\clause{p \cons \pi}{s}})} \desugar \exFun{\sigma}
         }
      \end{smathpar}
   }
   &
   \text{(\localref{done-curried-inversion})}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-var}}
          ]
          {
             \seq{u}', (\seqRange{(\pi_1, k_1)}{(\pi_j, k_j)})
             \match \gamma^\dagger, s
          }
          {
             \Lowlight{\seq{u} =\;}
             u \cons \seq{u}', (\seqRange{(\pattVar{x} \cons \pi_1, k_1)}
                                         {(\pattVar{x} \cons \pi_j, k_j)})
             \match
             \set{\bind{x}{v}} \disjunion \gamma^\dagger\Lowlight{\;= \gamma}, s
          }
       \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\desugar$-clauses-var}}
         ]
         {
            \seqRange{(\pi_1, k_1)}{(\pi_j, k_j)} \desugar \kappa'
         }
         {
            \seqRange{(\pattVar{x} \cons \pi_1, k_1)}{(\pattVar{x} \cons \pi_j, k_j)}
            \desugar
            (\elimVar{x}{\kappa'})\Lowlight{\,= \kappa}
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
   \\
   &
   \seq{v} = v \cons \seq{v}'\text{ with }u \desugar v\text{ and }\seq{u}' \desugar \seq{v}'
   \quad
   (\exists v, \seq{v}')
   &
   \text{(\localref{us-desugar-vs})}
   \notag
   \\
   &
   \seq{v}', \kappa' \match \gamma^\ddagger, e
   \textit{ with }\gamma^\dagger \desugar \gamma^\ddagger
   \textit{ and }s \desugar e
   \quad
   (\exists\gamma^\ddagger, e)
   &
   \text{IH}
   \locallabel{var-IH}
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\match$-var}}]
         {
            \seq{v}', \kappa' \match \gamma^\ddagger, e
         }
         {
            v \cons \seq{v}, \elimVar{x}{\kappa'} \match \set{\bind{x}{v}} \disjunion \gamma^\dagger, e
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \qedLocal
   \set{\bind{x}{u}} \disjunion \gamma^\dagger
   \desugar
   \set{\bind{x}{v}} \disjunion \gamma^\ddagger
   \textit{ and }
   s \desugar e
   &
   \text{($u \desugar v$), (\localref{var-IH})}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-record}}
         ]
         {
            \seq{u}^\ddagger \concat \seq{u}',
            (\seqRange{(\seq{p_1} \concat \pi_1, k_1)}{(\seq{p_j} \concat \pi_j, k_j)})
            \match \gamma, s
            \\
            \exRecord{\seq{\bind{y}{u}}^\ddagger} \subseteq \exRecord{\seq{\bind{x}{u}}^\dagger}
         }
         {
            \Lowlight{\seq{u} =\;}
            \exRecord{\seq{\bind{x}{u}}^\dagger} \cons \seq{u}',
            (\seqRange{(\pattRecord{\seq{\bind{y}{p_1}}} \cons \pi_1, k_1)}
                      {(\pattRecord{\seq{\bind{y}{p_j}}} \cons \pi_j, k_j)})
            \\
            \match
            \gamma, s
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\desugar$-clauses-record}}
         ]
         {
            \seqRange{(\seq{p_1} \concat \pi_1, k_1)}{(\seq{p_j} \concat \pi_j, k_j)} \desugar \kappa'
         }
         {
            \seqRange{(\pattRecord{\seq{\bind{x}{p_1}}} \cons \pi_1, k_1)}
                     {(\pattRecord{\seq{\bind{x}{p_j}}} \cons \pi_j, k_j)}
            \\
            \desugar
            (\elimRecord{\seq{x}}{\kappa'})
            \Lowlight{\;= \kappa}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \seq{v} = \exRec{\seq{\bind{x}{v}}^\dagger} \cons \seq{v}'
   \quad
   (\exists \seq{v}^\dagger, \seq{v}')
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-record}}
         ]
         {
            \seq{v}^\ddagger \concat \seq{v}', \kappa \match \gamma, e
            \\
            \exRec{\seq{\bind{y}{v}}^\ddagger} \subseteq \exRec{\seq{\bind{x}{v}}^\dagger}
         }
         {
            \Lowlight{\seq{v} =\;}\exRec{\seq{\bind{x}{v}}^\dagger} \cons \seq{v}',
            \elimRecord{\seq{y}}{\kappa}
            \match
            \gamma, e
         }
      \end{smathpar}
   }
   &
   \notag
\end{flalign}
\end{proof}
