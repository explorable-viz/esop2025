Induction on surface-language $\match$ derivation.
\begin{proof}
\small
\begin{flalign}
   &
   \seq{u} \desugar \seq{v}
   &\text{suppose}
   \\
   &
   \seq{(\pi, k)} \desugar \kappa
   &\text{suppose}
   \\
   &
   \seq{u}, \seq{(\pi, k)} \match \gamma, s
   &\text{suppose}
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\match$-var}}
          ]
          {
             \seq{v}, (\seqRange{(\pi_1, k_1)}{(\pi_j, k_j)})
             \match \gamma, s
          }
          {
             v \cons \seq{v}, (\seqRange{(\pattVar{x} \cons \pi_1, k_1)}
                                        {(\pattVar{x} \cons \pi_j, k_j)})
             \match
             \set{\bind{x}{v}} \disjunion \gamma, s
          }
       \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugar$-clauses-var}}
         ]
         {
            \seqRange{(\pi_1, k_1)}{(\pi_j, k_j)} \desugar \kappa
         }
         {
            \seqRange{(\pattVar{x} \cons \pi_1, k_1)}{(\pattVar{x} \cons \pi_j, k_j)}
            \desugar
            \elimVar{x}{\kappa}
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
   \\
   &
   \seq{v}', \kappa \match \gamma^\ddagger, e\textit{ with }\gamma^\dagger \desugar \gamma^\ddagger
   \quad
   (\exists\gamma^\ddagger, e)
   &
   \text{IH}
   \notag
\end{flalign}
\end{proof}
