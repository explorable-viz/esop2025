Case analysis on $\desugar$ derivation.
\begin{proof}
\small
\begin{flalign}
   &
   u \desugar v
   &\text{suppose}
   \\
   &
   \mu \desugar \sigma
   &\text{suppose}
   \\
   &
   v, \sigma \match \gamma, e
   &\text{suppose}
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugar$-clauses-uncurried}},
            right={$\forall i, i' \numleq j.p_i \neq p_{i'}$}
         ]
         {
            s_i \desugar e_i
            \\
            \sigma_i \eqdef \elimF(p_i, e_i)
            \quad
            (\forall i \numleq j)
         }
         {
            \mu = (\seqRange{\clause{p_1}{s_1}}{\clause{p_j}{s_j}})
            \desugar
            \seqRangeOp{\sigma_1}{\sigma_j}{\disjjoin} = \sigma
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   v, \sigma_i \match \gamma, e_i\text{ with }\forall i' \neq i.v, \sigma_i\;\matchNeg
   \quad
   (\exists i)
   &
   \text{\lemref{match-increasing-two}}
   \notag
\end{flalign}
\end{proof}
