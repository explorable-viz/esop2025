Induction on $\match$ derivation.
\begin{proof}
\small
\begin{flalign}
   &
   u, o \match \gamma
   &
   \text{suppose}
   \notag
   \\
   &
   u \desugar v
   &
   \text{suppose}
   \notag
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\match$-list-rest-nil}}
         ]
         {
            \strut
         }
         {
            \cNil, \pattListEnd \match \varnothing = \gamma
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\desugar$-constr}}]
         {
            \strut
         }
         {
            \cNil \desugar \cNil
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \locallabel{desugar-constr}
   \\
   &
   \elimArgsF(\seqEmpty,\kappa) = \kappa
   &
   \text{Def.~\elimArgsF}
   \notag
   \\
   &
   \mkElimF(\pattListEnd, \kappa) = \elimConstr{\elimBind{\cNil}{\kappa}}
   &
   \text{Def.~\mkElimF}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\match$-constr}}]
         {
            \inferrule*[left={\ruleName{$\match$-seq-empty}}]
            {
               \strut
            }
            {
               \seqEmpty, \kappa \match \varnothing, \kappa
            }
         }
         {
            \cNil, \elimConstr{\elimBind{\cNil}{\kappa}}
            \match
            \varnothing, \kappa
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-list-rest-cons}}
         ]
         {
            u_1, p, \match \gamma_1
            \\
            u_2, o \match \gamma_2
         }
         {
            u = \exConstr{\cCons}{u_1, u_2}, (\pattListNext{p}{o}) \match \gamma_1 \disjunion \gamma_2 = \gamma
         }
      \end{smathpar}
   }
   &
   \locallabel{match-constr}
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\desugar$-constr}}]
         {
            (u_1, u_2) \desugar (v_1, v_2)
         }
         {
            \exConstr{\cCons}{u_1, u_2} \desugar \exConstr{\cCons}{v, v'} = v
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \locallabel{desugar-constr}
   \\
   &
   \sigma \eqdef \elimArgsF(p \cons o,\kappa)
   &
   \notag
   \\
   &
   \mkElimF((\pattListNext{p}{o}), \kappa) = \elimConstr{\elimBind{\cCons}{\sigma}}
   &
   \text{Def.~\mkElimF}
   \notag
   \\
   &
   (v_1, v_2), \sigma \match \gamma', \kappa
   \textit{ and }
   \gamma \desugar \gamma'
   \quad
   (\exists \gamma')
   &
   \text{(\localref{match-constr}), (\localref{desugar-constr}), \lemref{match-domains-agree-seq} (1)}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\match$-constr}}]
         {
            (v_1, v_2), \sigma \match \gamma', \kappa
         }
         {
            \exConstr{\cCons}{v_1, v_2}, \elimConstr{\elimBind{\cCons}{\sigma}}
            \match
            \gamma', \kappa
         }
      \end{smathpar}
   }
   &
   \notag
\end{flalign}
\end{proof}
