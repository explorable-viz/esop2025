By induction on derivation of $\evalFwdS$
\begin{proof}
\small
\begin{flalign}
   &
   s \desugar e
   &
   \text{suppose}
   \locallabel{desug:expr}
   \\
   &
   \gamma \desugar \gamma'
   &
   \text{suppose}
   \locallabel{desug:env}
   \\
   &
   \gamma', e \evalFwdS v'
   &
   \text{suppose}
   \locallabel{eval:core}
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-var}}
         ]
         {
            \strut
         }
         {
            \gamma' = \gamma_2 \cdot (\bind{x}{v'}), x \evalFwdS v'
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugar$-var}}
         ]
         {
            \strut
         }
         {
            \exVar{x} \desugar \exVar{x}
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
   \\
   &
   \gamma = \gamma_1 \cons (\bind{x}{v}) \text{ with } \gamma_1 \desugar \gamma_2
   &
   \text{(\localref{desug:env})}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-var}}
         ]
         {
            \strut
         }
         {
            \gamma = \gamma_1 \cons (\bind{x}{v}), \exVar{x} \evalSugS v
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \qedLocal
   v \desugar v'
   &
   \text{(\localref{desug:env}), \lemref{varlookup}}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-constr}}
         ]
         {
            \gamma', \seq{e} \evalFwdS \seq{v}'
         }
         {
            \gamma', \exConstr{c}{\seq{e}} \evalFwdS \exConstr{c}{\seq{v}'}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugar$-constr}}
         ]
         {
            \seq{s} \desugar \seq{e}
         }
         {
            \exConstr{c}{\seq{s}} \desugar \exConstr{c}{\seq{e}}
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
   \\
   &
   \gamma, \seq{s} \evalSugS \seq{v}
   \textit{ with }
   \seq{v} \desugar \seq{v}'
   &
   \text{IH}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\evalSugS$-constr}}
         ]
         {
            \gamma, \seq{s} \evalSugS \seq{v}
         }
         {
            \gamma, \exConstr{c}{\seq{s}} \evalSugS \exConstr{c}{\seq{v}}
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\desugar$-constr}}
         ]
         {
            \seq{v} \desugar \seq{v}'
         }
         {
            \exConstr{c}{\seq{v}} \desugar \exConstr{c}{\seq{v}'}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-apply}},
            width=3.3in,
         ]
         {
            \gamma, (e, e') \evalS (\exClosure{\gamma'_1}{\rho}{\sigma}, v')
            \\
            \gamma'_1, \rho \closeDefsFwdR \gamma'_2
            \\
            v', \sigma \match \gamma_3, e^{\dag}
            \\
            \gamma'_1 \concat \gamma'_2 \concat \gamma'_3, e^{\dag} \evalFwdS u
         }
         {
            \gamma', \exApp{e}{e'}
            \evalFwdS
            u
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugar$-app}}
         ]
         {
            s \desugar e
            \\
            s' \desugar e'
         }
         {
            \exApp{s}{s'} \desugar \exApp{e'}{}
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
   \\
   &
   \gamma, s \evalSugS \exClosure{\gamma_1}{g}{\mu}
   \textit{ with }
   \gamma_1 \desugar \gamma'_1
   \opAnd
   g \desugar \rho
   \opAnd 
   \mu \desugar \sigma
   &
   \text{(IH)}
   \notag
   \\
   &
   \gamma, s' \evalSugS v
   \textit{ with }
   v \desugar v'
   &
   \text{(IH)}
   \notag
   \\
   &
   \exists \gamma_2.\; \gamma_1, g \closeDefsS \gamma_2
   \textit{ with }
   \gamma_2 \desugar \gamma'_2
   &
   \text{\lemref{closedefs-eq}}
   \notag
   \\
   &
   \exists \gamma_3, s^{\dag}.\; v, \mu \match \gamma_3, s^{\dag}
   \textit{ with }
   \gamma_3 \desugar \gamma'_3
   \opAnd
   s^{\dag} \desugar e^{\dag}
   &
   \text{\lemref{match-eq}}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-app}},
            width=3in,
         ]
         {
            \gamma, (s, s') \evalSugR{(S,S')} (\exClosure{\gamma_1}{g}{\mu}, v)
            \\
            \gamma_1, g \closeDefsS \gamma_2
            \\
            v, \mu \match \gamma_3, s^\dag
            \\
            \gamma_1 \concat \gamma_2 \concat \gamma_3, s^\dag \evalSugR{S^\dag} v^{\dag}
         }
         {
            \gamma, \exApp{s}{s'} \evalSugR{\trApp{S}{S'}{p}{S^{\dag}}} v^{\dag}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-let-rec}}
         ]
         {
            \gamma', \set{\seq{\bind{x}{\sigma}}} \closeDefsFwdR \gamma'
            \\
            \gamma' \concat {\gamma^{\ddagger}}, e \evalFwdS v
         }
         {
            \gamma', \exLetRec{\seq{\bind{x}{\sigma}}}{e}
            \evalFwdS
            v'
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[lab={\ruleName{$\desugar$-let-rec}}]
         {
            \seq{\mu} \desugar \seq{\sigma}
            \\
            s \desugar e
         }
         {
            \exLetRecPiecewise{\seq{\bind{x}{\mu}}}{s}
            \desugar
            \exLetRecPiecewise{\seq{\bind{x}{\sigma}}}{e}
         }
      \end{smathpar}
   }
   &
   \text{$\desugar$-let-rec}
   \notag
   \\
   &
   \exists \gamma^{\dag}.\; \gamma, \seq{\bind{x}{\mu}} \closeDefsS \gamma^{\dag}
   \textit{ with }
   \gamma^{\dag} \desugar \gamma^{\ddagger}
   &
   \lemref{closedefs-eq}
   \notag
   \\
   &
   \gamma \concat \gamma^{\dag} \desugar \gamma' \concat \gamma^{\ddagger}
   &
   (\localref{desug:env})
   \notag
   \\
   &
   \gamma \concat \gamma^{\dag}, s \evalSugS v
   \textit{ with }
   v \desugar v'
   &
   \text{(IH)}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-let-rec}}
         ]
         {
            \gamma, \seq{\bind{x}{\mu}} \closeDefsS \gamma^{\dag}
            \\
            \gamma \concat \gamma^{\dag}, s \evalSugR{S} v
         }
         {
            \gamma, \exLetRecPiecewise{\seq{\bind{x}{\mu}}}{s} \evalSugS v
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
\end{flalign}
\end{proof}
