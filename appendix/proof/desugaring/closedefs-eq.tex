\begin{proof}
   \small
   \begin{flalign}
      &
      \gamma_1 \desugar \gamma_2
      &
      \text{suppose}
      \locallabel{gamma-desugar}
      \\
      &
      g \desugar \rho
      &
      \text{suppose}
      \locallabel{g-desugar}
      \\
      \intertext{\crossrule}
      %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
      &
      \subcaseName{$\closeDefsS$-then-$\desugar$}
      &
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               left={\ruleName{$\closeDefsS$-surface}}
            ]
            {
               g = \seq{\bind{x}{\mu}}
               \\
               v_i = \exClosure{\gamma_1}{g}{\mu_i}
               \quad
               (\forall i \le \length{\seq{x}})
            }
            {
               \gamma_1, g \closeDefsS \set{\seq{\bind{x}{v}}}
            }
         \end{smathpar}
      }
      &
      \text{suppose}
      \locallabel{closed-surface}
      \\
      &
      \uprho = \set{\seq{\bind{x}{\upsigma}}} \text{ with } \seq{\upmu} \desugar \seq{\upsigma} \quad (\exists \seq{\upsigma})
      &
      \text{(\localref{g-desugar})}
      \locallabel{uprho-exists}
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               left={\ruleName{$\closeDefsS$-core}}
            ]
            {
               \rho = \set{\seq{\bind{x}{\sigma}}}
               \\
               v'_i = \exClosure{\gamma_2}{\rho}{\sigma_i}
               \quad
               (\forall i \numleq \length{\seq{x}})
            }
            {
               \gamma_2, \rho
               \closeDefsFwdR
               \set{\seq{\bind{x}{v'}}}
            }
         \end{smathpar}
      }
      &
      \text{(\localref{uprho-exists})}
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*
            [
               left={\ruleName{$\desugar$-env}}
            ]
            {
               \inferrule*
               [
                  left={\ruleName{$\desugar$-closure}},
                  right={$\forall i. \numleq \length{\seq{x}}$}
               ]
               {
                  \upgamma_1 \desugar \upgamma_2
                  \\
                  g \desugar \uprho
                  \\
                  \upmu_i \desugar \upsigma_i
               }
               {
                  v_i = \exClosure{\upgamma_1}{g}{\upmu_i} \desugar \exClosure{\upgamma_2}{\uprho}{\upsigma_i}
               }
            }
            {
               \set{\seq{\bind{\exVar{x}}{v}}} \desugar \set{\seq{\bind{\exVar{x}}{v'}}}
            }
         \end{smathpar}
      }
      &
      \notag
      \\
      &
      \subcaseName{$\desugar$-then-$\closeDefsS$}
      &
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*[
               left={\ruleName{$\closeDefsS$-core}}
            ]
            {
               \uprho = \set{\seq{\bind{x}{\upsigma}}}
               \\
               v'_i = \exClosure{\upgamma_2}{\uprho}{\upsigma_i}
               \quad
               (\forall i \numleq \length{\seq{x}})
            }
            {
               \upgamma_2, \uprho
               \closeDefsFwdR
               \set{\seq{\bind{x}{v'}}}
            }
         \end{smathpar}
      }
      &
      \text{suppose}
      \locallabel{close-rho}
      \\
      &
      g = \set{\seq{\bind{x}{\upmu}}} \text{ with } \seq{\upmu} \desugar \seq{\upsigma} \quad (\exists \seq{\upmu})
      &
      \text{(\localref{g-desugar})}
      \locallabel{g-exists}
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*
            [
               left={\ruleName{$\closeDefsS$-surface}}
            ]
            {
               g = \seq{\bind{x}{\mu}}
               \\
               v_i = \exClosure{\gamma_1}{g}{\mu_i}
               \quad
               (\forall i \le \length{\seq{x}})
            }
            {
               \gamma_1, g \closeDefsS \set{\seq{\bind{x}{v}}}
            }
         \end{smathpar}
      }
      &
      \text{(\localref{g-exists})}
      \notag
      \\
      &
      \derivation{\derivationWidth}{
         \begin{smathpar}
            \inferrule*
            [
               left={\ruleName{$\desugar$-env}}
            ]
            {
               \inferrule*
               [
                  left={\ruleName{$\desugar$-closure}},
                  right={$\forall i. \numleq \length{\seq{x}}$}
               ]
               {
                  \upgamma_1 \desugar \upgamma_2
                  \\
                  g \desugar \uprho
                  \\
                  \upmu_i \desugar \upsigma_i
               }
               {
                  v_i = \exClosure{\upgamma_1}{g}{\upmu_i} \desugar \exClosure{\upgamma_2}{\uprho}{\upsigma_i}
               }
            }
            {
               \set{\seq{\bind{\exVar{x}}{v}}} \desugar \set{\seq{\bind{\exVar{x}}{v'}}}
            }
         \end{smathpar}
      }
      &
      \notag
   \end{flalign}
\end{proof}
