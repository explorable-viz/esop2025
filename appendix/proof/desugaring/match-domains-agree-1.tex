Case analysis on $\match$ derivation.
\begin{proof}
\small
\begin{flalign}
   &
   u, p \match \gamma
   &
   \text{suppose}
   \notag
   \\
   &
   u \desugar v
   &
   \text{suppose}
   \notag
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-var}}
         ]
         {
            \strut
         }
         {
            u, x \match \set{\bind{\exVar{x}}{u}} = \gamma
         }
      \end{smathpar}
   }
   \\
   &
   \mkElimF(x, \kappa) = \elimVar{x}{\kappa}
   &
   \text{Def.~\mkElimF}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\match$-var}}]
         {
            \strut
         }
         {
            v, \elimVar{x}{\kappa} \match \set{\bind{x}{v}}, \kappa
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\desugar$-env}}]
         {
             u \desugar v
         }
         {
             \gamma = \set{\bind{x}{u}} \desugar \set{\bind{x}{v}}
         }
      \end{smathpar}
    }
   &
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-constr}}
         ]
         {
            \seq{u},\seq{p} \match \gamma
         }
         {
            u = \exConstr{c}{\seq{u}},\pattConstr{c}{\seq{p}} \match \gamma
         }
      \end{smathpar}
   }
   &
   \locallabel{match-constr}
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\desugar$-constr}}]
         {
            \seq{u} \desugar \seq{v}
         }
         {
             \exConstr{c}{\seq{u}} \desugar \exConstr{c}{\seq{v}} = v
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \locallabel{desugar-constr}
   \\
   &
   \kappa' \eqdef \elimArgsF(\seq{p},\kappa)
   &
   \notag
   \\
   &
   \mkElimF(\pattConstr{c}{\seq{p}}, \kappa) = \elimConstr{\elimBind{c}{\kappa'}}
   &
   \text{Def.~\mkElimF}
   \notag
   \\
   &
   \seq{v}, \kappa' \match \gamma', \kappa
   \textit{ and }
   \gamma \desugar \gamma'
   \quad
   (\exists \gamma')
   &
   \text{(\localref{match-constr}), (\localref{desugar-constr}), \lemref{match-domains-agree-seq} (1)}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\match$-constr}}]
         {
            \seq{v}, \kappa' \match \gamma', \kappa
         }
         {
            \exConstr{c}{\seq{v}}, \elimConstr{\elimBind{c}{\kappa'}}
            \match
            \gamma', \kappa
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-record}}
         ]
         {
            \seq{u}', \seq{p} \match \gamma
            \quad
            \exRecord{\seq{\bind{\exVar{y}}{u}}'} \subseteq \exRecord{\seq{\bind{\exVar{x}}{u}}}
         }
         {
            u = \exRecord{\seq{\bind{\exVar{x}}{u}}}, \exRecord{\seq{\bind{\exVar{y}}{p}}} \match \gamma
         }
      \end{smathpar}
   }
   &
   \locallabel{match-record}
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\desugar$-record}}]
         {
            \seq{u} \desugar \seq{v}
         }
         {
             \exRecord{\seq{\bind{x}{u}}} \desugar \exRecord{\seq{\bind{x}{v}}} = v
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \locallabel{desugar-record}
   \\
   &
   \kappa' \eqdef \elimArgsF(\seq{p},\kappa)
   &
   \notag
   \\
   &
   \mkElimF(\pattRecord{\seq{\bind{y}{p}}}, \kappa) = \elimRecord{\seq{y}}{\kappa'}
   &
   \text{Def.~\mkElimF}
   \notag
   \\
   &
   \seq{u}' \desugar \seq{v}'
   &
   (\localref{match-record}, \localref{desugar-record})
   \notag
   \\
   &
   \seq{v}', \kappa' \match \gamma', \kappa
   \textit{ and }
   \gamma \desugar \gamma'
   \quad
   (\exists \gamma')
   &
   \text{(\localref{match-record}), \lemref{match-domains-agree-seq} (1)}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-record}}
         ]
         {
            \seq{v}', \kappa' \match \gamma, \kappa
            \\
            \exRec{\seq{\bind{y}{v}}'} \subseteq \exRec{\seq{\bind{x}{v}}}
         }
         {
            \exRec{\seq{\bind{x}{v}}},
            \elimRecord{\seq{y}}{\kappa'}
            \match
            \gamma, \kappa
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            left={\ruleName{$\match$-nonempty-list}}
         ]
         {
            u_1, p \match \gamma_1
            \\
            u_2, o \match \gamma_2
         }
         {
            u = \exConstr{\cCons}{u_1, u_2}, \pattList{p}{o} \match \gamma_1 \disjunion \gamma_2 = \gamma
         }
      \end{smathpar}
   }
   &
   \locallabel{match-cons}
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\desugar$-constr}}]
         {
            (u_1, u_2) \desugar (v_1, v_2)
         }
         {
            \exConstr{\cCons}{u_1, u_2} \desugar \exConstr{\cCons}{v_1, v_2} = v
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \locallabel{desugar-cons}
   \\
   &
   \sigma \eqdef \elimArgsF(p \cons o,\kappa)
   &
   \notag
   \\
   &
   \mkElimF(\pattList{p}{o}, \kappa) = \elimConstr{\elimBind{\cCons}{\sigma}}
   &
   \text{Def.~\mkElimF}
   \notag
   \\
   &
   (v_1, v_2), \sigma \match \gamma', \kappa
   \textit{ and }
   \gamma \desugar \gamma'
   \quad
   (\exists \gamma')
   &
   \text{(\localref{match-cons}), (\localref{desugar-cons}), \lemref{match-domains-agree-seq} (1)}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[left={\ruleName{$\match$-constr}}]
         {
            (v_1, v_2), \sigma \match \gamma', \kappa
         }
         {
            \exConstr{\cCons}{v_1, v_2}, \elimConstr{\elimBind{\cCons}{\sigma}}
            \match
            \gamma', \kappa
         }
      \end{smathpar}
   }
   &
   \notag
\end{flalign}
\end{proof}
