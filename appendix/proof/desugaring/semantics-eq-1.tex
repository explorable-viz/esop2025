By induction on derivation of $\evalSugS$.
\begin{proof}
\small
\begin{flalign}
   &
   s_0 \desugar e_0
   &
   \text{suppose}
   \locallabel{desug:expr}
   \\
   &
   \gamma \desugar \gamma'
   &
   \text{suppose}
   \locallabel{desug:env}
   \\
   &
   \gamma, s_0 \evalSugS v
   &
   \text{suppose}
   \locallabel{eval:surf}
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-var}}
         ]
         {
            \strut
         }
         {
            {\gamma_1} \cons (\bind{x}{v}), \exVar{x} \evalSugS v
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugar$-var}}
         ]
         {
            \strut
         }
         {
            \exVar{x} \desugar \exVar{x}
         }
      \end{smathpar}
   }
   &
   \text{$\desugar$}
   \notag
   \\
   &
   \gamma_1 \desugar \gamma_2
   &
   (\localref{desug:env})
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-var}}
         ]
         {
            \strut
         }
         {
            \gamma_2 \cdot (\bind{\exVar{x}}{v'}), \exVar{x} \evalFwdS v'
         }
      \end{smathpar}
   }
   &
   \text{$\evalFwdS$}
   \notag
   \\
   &
   \qedLocal
   \text{$v \desugar v'$.}
   &
   \text{\lemref{varlookup}, (\localref{desug:env})}
   \notag
   \\
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-constr}}
         ]
         {
            \gamma, \seq{s} \evalSugS \seq{v}
         }
         {
            \gamma, \exConstr{c}{\seq{s}} \evalSugS \exConstr{c}{\seq{v}}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugar$-constr}}
         ]
         {
            \seq{s} \desugar \seq{e}
         }
         {
            \exConstr{c}{\seq{s}} \desugar \exConstr{c}{\seq{e}}
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
   \\
   &
   \gamma', e \evalFwdS \seq{v}'
   \textit{ with }
   \seq{v} \desugar \seq{v}'
   &
   \text{IH}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-constr}}
         ]
         {
            \gamma', \seq{e} \evalFwdS \seq{v}'
         }
         {
            \gamma', \exConstr{c}{\seq{e}} \evalFwdS \exConstr{c}{\seq{v}'}
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugar$-constr}}
         ]
         {
            \seq{v} \desugar \seq{v}'
         }
         {
            \exConstr{c}{\seq{v}} \desugar \exConstr{c}{\seq{v}'}
         }
      \end{smathpar}
   }
   &
   \notag
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-app}},
            width=3in,
         ]
         {
            \gamma, (s, s') \evalSugR{(S,S')} (\exClosure{\gamma_1}{g}{\mu}, v)
            \\
            \gamma_1, g \closeDefsS \gamma_2
            \\
            v, \mu \match \gamma_3, s^\dag
            \\
            \gamma_1 \concat \gamma_2 \concat \gamma_3, s^\dag \evalSugR{S^\dag} v^{\dag}
         }
         {
            \gamma, \exApp{s}{s'} \evalSugR{\trApp{S}{S'}{p}{S^{\dag}}} v^{\dag}
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\desugar$-app}}
         ]
         {
            s \desugar e
            \\
            s' \desugar e'
         }
         {
            \exApp{s}{s'} \desugar \exApp{e}{e'}
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
   \\
   &
   \gamma', e \evalFwdS \exClosure{\gamma'_1}{\rho}{\sigma}
   \textit{ with }
   \gamma_1 \desugar \gamma_2
   \opAnd
   g \desugar \rho
   \opAnd
   \mu \desugar \sigma
   &
   \text{(IH)}
   \notag
   \\
   &
   \gamma', e' \evalFwdS v'
   \textit{ with }
   v \desugar v'
   &
   \text{(IH)}
   \notag
   \\
   &
   \exists \gamma'_2.\; \gamma'_1, \rho \closeDefsS \gamma'_2
   \textit{ with }
   \gamma_2 \desugar \gamma'_2
   &
   \lemref{closedefs-eq}
   \notag
   \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalFwdS$-apply}},
            width=3.3in,
         ]
         {
            \gamma', (e, e') \evalS (\exClosure{\gamma_1}{\rho}{\sigma}, v')
            \\
            \gamma'_1, \rho \closeDefsFwdR \gamma'_2
            \\
            v', \sigma \match \gamma'_3, e^{\dag}
            \\
            \gamma'_1 \concat \gamma'_2 \concat \gamma'_3, e^{\dag} \evalFwdS u
         }
         {
            \gamma', \exApp{e}{e'}
            \evalFwdS
            u
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
   \\
   &
   \exClosure{\gamma_1}{g}{\mu} \desugar \exClosure{\gamma'_1}{\rho}{\sigma}
   &
   \text{IH}
   \notag
   \\
   &
   v \desugar v'
   &
   \text{IH}
   \notag
   \\
   &
   \gamma_2 \desugar \gamma'_2
   &
   \text{\lemref{closedefs-eq}}
   \notag
   \\
   &
   \gamma_3 \desugar \gamma'_3 \opAnd s^{\dag} \desugar e^{\dag}
   &
   \text{$\mu \desugar \sigma$, \lemref{match-eq}}
   \notag
   \\
   &
   v^{\dag} \desugar u
   &
   \text{IH}
   \notag
   \intertext{\crossrule}
   %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
   &
   \caseDerivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[
            lab={\ruleName{$\evalSugS$-let-rec}}
         ]
         {
            \gamma, \seq{\bind{x}{\mu}} \closeDefsS \gamma^{\dag}
            \\
            \gamma \concat \gamma^{\dag}, s \evalSugS v
         }
         {
            \gamma, \exLetRecPiecewise{\seq{\bind{x}{\mu}}}{s} \evalSugS v
         }
      \end{smathpar}
   }
   &
   \notag
   \\
   % & % Really not sure if these are needed for this to work
   % \exists e. s \desugar e
   % &
   % \text{(IH)}
   % \\
   % &
   % \mu_i \desugar \sigma_i \;(\forall i \numleq \length{\seq{x}})
   % &
   % \text{(IH)}
   % \\
   &
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[lab={\ruleName{$\desugar$-let-rec}}]
         {
            \seq{\mu} \desugar \seq{\sigma}
            \\
            s \desugar e
         }
         {
            \exLetRecPiecewise{\seq{\bind{x}{\mu}}}{s}
            \desugar
            \exLetRecPiecewise{\seq{\bind{x}{\sigma}}}{e}
         }
      \end{smathpar}
   }
   &
   \text{inversion}
   \notag
   \\
   &
   \exists \gamma'.\; \gamma', \seq{\bind{x}{\sigma}} \closeDefsS \gamma^{\ddagger}
   \textit{ with }
   \gamma^{\dag} \desugar \gamma^{\ddagger}
   &
   \text{(\lemref{closedefs-eq})}
   \notag
   \\
   &
   \gamma \concat \gamma^{\dag} \desugar \gamma' \concat \gamma^{\ddagger}
   &
   \text{(\localref{desug:env})}
   \notag
   \\
   &
   \gamma' \concat \gamma^{\ddagger}, e \evalFwdS v'
   \textit{ with }
   v \desugar v'
   &
   \text{(IH)}
   \notag
   \\
   &
   \qedLocal
   \derivation{\derivationWidth}{
      \begin{smathpar}
         \inferrule*[lab={\ruleName{$\evalFwdS$-let-rec}}]
         {
            \gamma', \seq{\bind{\exVar{x}}{\sigma}} \closeDefsS {\gamma^{\ddagger}}
            \\
            \gamma' \concat {\gamma^{\ddagger}}, e \evalFwdS v'
         }
         {
            \gamma', \exLetRecPiecewise{\seq{\bind{\exVar{x}}{\sigma}}}{e} \evalFwdS v'
         }
      \end{smathpar}
   }
   &
   \notag
\end{flalign}
\end{proof}
