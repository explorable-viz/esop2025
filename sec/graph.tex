\section{Dependence graph formulation}
\label{sec:graph}
We start by introducing dependence graphs and some standard graph-theoretic notions.
\subsection{Dependence graphs}

$G$ ranges over \emph{dependence graphs} $(V, E)$ with vertices $\alpha, \beta$. For the following definitions
fix $G = (V, E)$.

\begin{definition}[Opposite graph]
   Define $G^{-1} \eqdef (V, E^{-1})$ where $\param^{-1}$ denotes relational inverse.
\end{definition}
\begin{definition}[Complement]
   For a set $X$ and $Y \subseteq X$, we write $Y^c \subseteq X$ to denote the \textit{complement}
   of $Y$ in $X$. Note that the type of $\param^{c}$ depends on the contextual graph.
\end{definition}
\begin{definition}[Induced subgraph]
   For any $V' \subseteq V$, write $\induced{G}{V'}$ for the subgraph of $G$ induced by $V'$, that is
   $\induced{G}{V'} \eqdef (V', E')$ where $E' = \set{(\alpha,\beta) \in E \mid \alpha, \beta \in V'}$.
\end{definition}

\begin{definition}[Reachable]
   For any pair of vertices $\alpha, \beta \in V$, write $\reachable{G}(\alpha, \beta)$, to mean there exists
   $\seq{\alpha}$ such that $\forall i < \length{\seq{\beta}}.\;(\beta_i,\beta_{i+1}) \in E$ where $\seq{\beta}
   \eqdef \alpha \cons \seq{\alpha} \cons \beta$.
\end{definition}

\begin{definition}[Neighbours]
   Define the \emph{in neighbours} and \emph{out neighbours} of $G$ as the following functions of type $V \to
   \powerset V$:
   \begin{itemize}
      \item $\inN{G}(\alpha) \eqdef \set{\alpha' \in V \mid (\alpha', \alpha) \in E}$
      \item $\outN{G} \eqdef \inN{G^{-1}}$
   \end{itemize}
\end{definition}
We extend the idea of neighbours of a node to \textit{neighbourhoods} of a set of vertices in the graph.
\begin{definition}[Neighbourhoods]
   For any $V' \subseteq V$, define:
   \begin{itemize}
      \item $\inN{G}(V') \eqdef \set{\alpha \in V \mid \exists \beta \in V'.\; \alpha \in \inN{G}(\beta) \opAnd \alpha \notin V'}$
      \item $\outN{G}(V') \eqdef \inN{G^{-1}}(V')$
   \end{itemize}
\end{definition}

\begin{definition}[Sources and Sinks]
   Define the \emph{sources} and \emph{sinks} of $G$ as:
   \begin{itemize}
      \item $\sources{G} \eqdef \set{v \in V \mid \inN{G}(v) = \emptyset}$
      \item $\sinks{G} \eqdef \sources{G^{-1}}$
   \end{itemize}
\end{definition}

\begin{definition}[Subgraph]
   Write $G' \subseteq G$ to mean that $G'$ is a subgraph of $G$.
\end{definition}

\begin{definition}[Subgraph Extension]
   For any $G' = (V',E') \subseteq G$ and any $V^{\dag} \subseteq V$, define $G' \disjunion V^{\dag} \eqdef \induced{G}{V' \cup V^{\dag}}$.
\end{definition}

\subsection{Galois slicing over dependence graphs}

Fix a dependence graph $G$.

We characterize slicing over $G$ with 2 functions of the following types:
\begin{itemize}
   \item $\bwdSliceR{G} \;:\; \powerset{\sources{G}} \to \powerset{\sinks{G}}$
   \item $\fwdSliceR{G} \;:\; \powerset{\sinks{G}} \to \powerset{\sources{G}}$
\end{itemize}

For a set of outputs of interest, the backward slicing function $\bwdSliceR{G}$ computes just the inputs
needed for those outputs. More formally:

\begin{definition}[Backward slice]
   For any set of sources $V \subseteq \powerset{\sources{G}}$, define the \emph{backward slice}
   ${\bwdSliceR{G}}(V)$ of $V$ to be the set of sinks $\set{v \in \sinks{G} \mid \exists v' \in
   V.\;\reachable{G}(v', v)}$.
\end{definition}

For a set of inputs of interest, the forward slicing function $\fwdSliceR{G}$ computes the set of outputs for
which those inputs are sufficient. Formally:
\begin{definition}[Forward slice]
   For any set of sinks $V \subseteq \powerset{\sinks{G}}$, define the \emph{forward slice}
   ${\fwdSliceR{G}}(V)$ of $V$ to be the set of sources $\set{v \in \sources{G} \mid {\bwdSliceR{G}}(\set{v})
   \subseteq V}$.
\end{definition}

\begin{lemma}[Adjointness of ${\fwdSliceR{G}}$ and $\bwdSliceR{G}$]
   \item Suppose $U \subseteq \sources{G}$ and $V \subseteq \sinks{G}$. Then $U \subseteq {\fwdSliceR{G}}(V)
   \iff {\bwdSliceR{G}}(U) \subseteq V$.
\end{lemma}

\begin{definition}[Forward slicing algorithm]
   \figref{graph:slicing} defines the forward slicing relation $\fwdSliceR{G}$.
\end{definition}

\begin{theorem}[Correctness of forward slicing algorithm]
   \item For any $V \subseteq \powerset{\sinks{G}}$, suppose $G[V]\fwdSliceR{G} G'$. Then $\sources{G'} = {\fwdSliceR{G}}(V)$.
\end{theorem}

\begin{theorem}[De Morgan dual of forward slice]
   ${\fwdSliceR{G}}(V) = ({\bwdSliceR{G^{-1}}}(V^c))^{c}$
\end{theorem}

We use the notion of the neighbourhood of a subgraph to formulate a more efficient version of forward slicing.
Informally, we start with our subgraph, then we only consider its neighbourhood. If we can add a new node from
its neighbourhood to the current slice, then we add the new nodes neighbours to our current neighbourhood set.
We halt the algorithm only when we cannot add any new nodes from the current slices neighbourhood. This lets
us grow the slice without considering each node in the graph $\length{V}$ times.
