\subsection{Dependence graphs}

$G$ ranges over \emph{dependency graphs} $(V, E)$ where $V$ is a set of \emph{addresses} $\alpha, \beta
\in \Addr$ which serve as vertices.  Addresses include a distinguished address $\addrNull$ called the
\emph{null} address.

\begin{definition}[Opposite graph]
   For any $G = (V, E)$ define $G^{-1} \eqdef (V, E^{-1})$ where $\param^{-1}$ denotes relational inverse.
\end{definition}

\begin{definition}
   For any $G = (V, E)$, any $V' \subseteq V$ and any $\alpha \notin V$, define $\mkAddr{G}{V'}{\alpha} = (V \cup \set{\alpha}, E \cup (V' \times \set{\alpha}))$.
\end{definition}

\begin{definition}[Neighbours]
   For any $G = (V,E)$ define the \emph{in neighbours} and \emph{out neighbours} as the following functions of type $V \to \powerset V$:
   \begin{itemize}
      \item $\inN{G}(\alpha) = \set{\alpha' \in V \mid (\alpha', \alpha) \in E}$
      \item $\outN{G} = \inN{G^{-1}}$
   \end{itemize}
\end{definition}
We use $G' \subseteq G$ to mean that $G'$ is a subgraph of $G$.
\begin{definition}[Subgraph Extension]
   For any $G' = (V',E') \subseteq G = (V,E)$ and $V^{\dag} \subseteq V$, define $G' \disjunion V^{\dag} = G''$ where 
   $G''$ is the \textit{induced subgraph} $G[V' \cup V^{\dag}]$ formed by extending $G'$ with the vertices in $V^{\dag}$,
   and all edges with both endpoints in the set $V' \cup V^{\dag}$.
\end{definition}

\begin{definition}[Backwards Slicing (Graph)]
   We use $\bwdSliceS$ as the symbol for our backwards slicing operator. Intuitively, the relation 
   $\bwdSliceS$, when given a set of outputs of interest computes the subgraph of the computation that
   contains only the nodes necessary for the computation of those outputs. 
\end{definition}
