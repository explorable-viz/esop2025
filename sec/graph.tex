\section{Graphical Galois slicing}

\subsection{Preliminaries}

\begin{definition}[Complement]
   For any set $X$ and any $Y \subseteq X$, write $\compl{Y} \subseteq X$ to denote the \textit{complement} of $Y$
   with respect to $X$.
\end{definition}

\subsection{Galois connections for a relation}

The following Galois connection $\consumesF{D} \adjoint \producesF{D}: \powerset{X} \to \powerset{Y}$
generalises the usual image/preimage Galois connection for a function $f: X \to Y$ to a relation $D \subseteq
Y \times X$. (We flip the direction of the relation compared to $f$ so we can read $D$ as \emph{depends on}.)

\begin{definition}
   Define the following two functions:
   \begin{enumerate}
      \item $\consumesF{D}(Y') \eqdef \set{x \in X \mid \exists y \in Y'.\;D(y,x)}$ \hfill (consumes)
      \item $\producesF{D}(X') \eqdef \set{y \in Y \mid \forall x \in \compl{X'}.\;\neg D(y,x)}$ \hfill (produces)
   \end{enumerate}
\end{definition}

\begin{lemma}[Galois connection]
   $\consumesF{D}(Y') \subseteq X' \iff Y' \subseteq \producesF{D}(X')$
\end{lemma}

\noindent $\consumesF{D}$ collects just the inputs consumed by a set of outputs; $\producesF{D}$ collects just the outputs produced by a set of inputs.

\begin{definition}[De Morgan dual]
   Suppose $\mathcal{A}, \mathcal{B}$ are Boolean algebras.
   \begin{enumerate}
      \item For any $f: \mathcal{A} \to \mathcal{B}$, define $\dual{f}: \mathcal{A} \to \mathcal{B}$ to be $\neg_{\mathcal{B}} \after f \after \neg_{\mathcal{A}}$.
      \item For any $f \adjoint g: \mathcal{A} \to \mathcal{B}$, define $\dual{(f \adjoint g)}: \mathcal{B} \to \mathcal{A}$ to be $\dual{g} \adjoint \dual{f}$.
   \end{enumerate}
\end{definition}

\begin{definition} Define $\producesDualF{D} \adjoint \consumesDualF{D}: \powerset{Y} \to \powerset{X}$ as follows:
   \begin{enumerate}
      \item $\consumesDualF{D} \eqdef \dual{(\consumesF{D})}$ \hfill (produced by)
      \item $\producesDualF{D} \eqdef \dual{(\producesF{D})}$ \hfill (consumed by)
   \end{enumerate}
\end{definition}

\begin{lemma}[Relationship between dual and adjoint]
   $\consumesF{D} \adjoint \producesF{D} = \producesDualF{D^{-1}} \adjoint \consumesDualF{D^{-1}}$
\end{lemma}

\begin{definition}[Cooperating inputs and competing outputs]
   Define
   \begin{enumerate}
      \item $\relatedOutputsF{D} \eqdef \consumesF{DD^{-1}} = \consumesF{D}\consumesF{D^{-1}} = \consumesF{D}\producesDualF{D}$ \hfill (competes with)
      \item $\relatedInputsF{D} \eqdef \consumesF{D^{-1}D} = \consumesF{D^{-1}}\consumesF{D} = \producesDualF{D}\consumesF{D}$ \hfill (cooperates with)
   \end{enumerate}
\end{definition}

\subsection{Dependence graphs}

We now introduce \emph{dependence graphs} $G, H = (V, E)$ with vertices $\alpha, \beta \in V$ and edges
$(\alpha, \beta) \in E$ along with some standard graph-theoretic notions. For the following definitions fix a
dependence graph $G$.

\begin{definition}[Opposite graph]
   If $G = (V,E)$, define $G^{-1} \eqdef (V, E^{-1})$ where $\param^{-1}$ denotes relational inverse.
\end{definition}

\begin{definition}[Neighbours]
   Define the following functions of type $\V(G) \to \powerset{\V(G)}$:
   \begin{itemize}
      \item $\outN{G}(\alpha) \eqdef \set{\beta \in \V(G) \mid (\alpha, \beta) \in \E(G)}$ \hfill (out neighbours)
      \item $\inN{G} \eqdef \outN{G^{-1}}$ \hfill (in neighbours)
   \end{itemize}
\end{definition}

\begin{definition}[Edges]
   Define the following functions of type $\V(G) \to \powerset{\E(G)}$:
   \begin{itemize}
      \item $\outE{G}(\alpha) \eqdef \set{\alpha} \times \outN{G}(\alpha)$
      \item $\inE{G}(\alpha) \eqdef \outN{G^{-1}}(\alpha)$
   \end{itemize}
\end{definition}

\noindent Both notations extend to sets of vertices in the obvious way.

\begin{definition}[Subgraph, union]
   If $G = (V, E)$ and $G' = (V', E')$ define:
   \begin{enumerate}
      \item $G' \subseteq G$ to mean $G'$ is a \emph{subgraph} of $G$, i.e.~$V' \subseteq V$ and $E \subseteq E'$.
      \item $G \cup G'$ to mean the (non-disjoint) union ($V \cup V'$, $E \cup E'$) of $G$ and $G'$.
   \end{enumerate}
\end{definition}

\begin{definition}[Reachable]
   Define the \emph{reachable-from} relation $\reachable{G} \subseteq \V(G)^2$ that has $\reachable{G}(\alpha,
   \beta)$ iff there exists $\seq{\alpha}$ such that $\forall i < \length{\seq{\beta}}.\;(\beta_i,\beta_{i+1})
   \in \E(G)$ where $\seq{\beta} \eqdef \alpha \cons \seq{\alpha} \cons \beta$.
\end{definition}

\begin{definition}[Sources and sinks]
   Define:
   \begin{itemize}
      \item $\sources{G} \eqdef \set{\alpha \in \V(G) \mid \inN{G}(\alpha) = \emptyset}$ \hfill (sources)
      \item $\sinks{G} \eqdef \sources{G^{-1}}$ \hfill (sinks)
   \end{itemize}
\end{definition}

\begin{definition}[Extension]
   Define the \emph{extension} $\extension{G}$ of $G$ to be $\reachable{G}$ restricted to $\sources{G} \times \sinks{G}$.
\end{definition}

\subsection{Galois slicing over dependence graphs}

\begin{definition}[Slicing algorithms]
   \item Suppose $\hat{G}$ a graph.
   \begin{enumerate}
      \item \figref{graph:slicing:backward} defines the relation $\bwdAlg{\hat{G}}$. \hfill (backward slicing)
      \item \figref{graph:slicing:forward} defines the relation $\fwdAlg{\hat{G}}$. \hfill (forward slicing)
   \end{enumerate}
\end{definition}

\begin{theorem}[Correctness of slicing algorithms]
   \item
   \begin{enumerate}
   \item Suppose $V \subseteq \sources{\hat{G}}$. Then $\consumesF{\extension{\hat{G}}}(V) = V' \iff V \bwdAlg{\hat{G}} V'$.
   \item Suppose $V \subseteq \sinks{\hat{G}}$. Then $\producesF{\extension{\hat{G}}}(V) = V' \iff V \fwdAlg{\hat{G}} V'$.
   \end{enumerate}
\end{theorem}
