\subsection{Dependence graphs}

$G$ ranges over \emph{dependency graphs} $(V, E)$ where $V$ is a set of \emph{addresses} $\alpha, \beta
\in \Addr$ which serve as vertices.  Addresses include a distinguished address $\addrNull$ called the
\emph{null} address.

\begin{definition}[Opposite graph]
   For any $G = (V, E)$ define $G^{-1} \eqdef (V, E^{-1})$ where $\param^{-1}$ denotes relational inverse.
\end{definition}
\begin{definition}[Complement]
   For a set $X$ and $X' \subseteq X$, we write $X^c \subseteq X$ to denote the \textit{complement} 
   of $X'$ in $X$. Note that the type of $\param^{c}$ depends on the contextual graph.  
\end{definition}
\begin{definition}[Induced subgraph]
   For any $G = (V,E)$, and vertices $V' \subseteq V$, we write $G[V']$ to be the subgraph of $G$ induced
   by $V'$, that is $G[V'] = (V', E')$ where $E' = \set{(s,t) \in E \mid s,t \in V'}$.
\end{definition}

\begin{definition}[Path]
   For a graph $G = (V,E)$ and a pair of vertices $v, v' \in V$, we write $P_G(v, v')$, to mean that there is a
   \textit{path} from $v$ to $v'$ in $G$, meaning either $\exists \seq{v} \in V.\; (v_i,v_{i+1}) \in E \opAnd (v, v_1) \in E \opAnd (v_n, v') \in E$,
   or that $(v, v') \in E$. 
\end{definition}

\begin{definition}[Sources and Sinks]
   For a graph $G = (V,E)$, we define the sources $S(G) = \set{v \in V \mid \inN{G}(v) = \emptyset}$, and the sinks
   $T(G) = \set{v \in V \mid \outN{G}(v) = \emptyset}$
\end{definition}

\begin{definition}[Neighbourhoods]
   For a graph $G = (V,E)$ and a set of nodes $V' \subseteq V$, we write $N_G(V')$ to mean the \textit{neighbourhood}
   of $V'$, that is: $N_G(V') = \set{v \in V \mid v \notin V' \opAnd \exists v' \in V'.\; (v,v') \in E}$ 
\end{definition}
We will see later how the definition of a neighbourhood can be used to define a more efficient forward slicing algorithm.
\begin{definition}
   For any $G = (V, E)$, any $V' \subseteq V$ and any $\alpha \notin V$, define $\mkAddr{G}{V'}{\alpha} = (V \cup \set{\alpha}, E \cup (V' \times \set{\alpha}))$.
\end{definition}

\begin{definition}[Neighbours]
   For any $G = (V,E)$ define the \emph{in neighbours} and \emph{out neighbours} as the following functions of type $V \to \powerset V$:
   \begin{itemize}
      \item $\inN{G}(\alpha) = \set{\alpha' \in V \mid (\alpha', \alpha) \in E}$
      \item $\outN{G} = \inN{G^{-1}}$
   \end{itemize}
\end{definition}
We use $G' \subseteq G$ to mean that $G'$ is a subgraph of $G$.
\begin{definition}[Subgraph Extension]
   For any $G' = (V',E') \subseteq G = (V,E)$ and $V^{\dag} \subseteq V$, define $G' \disjunion V^{\dag} = G''$ where
   $G''$ is the induced subgraph $G[V' \cup V^{\dag}]$.
\end{definition}
We characterize slicing with 2 functions/relations, parameterized by the graph $G$, of the following types:
\begin{itemize}
   \item $\bwdSliceR{G} \;:\; \powerset{S(G)} \to \powerset{T(G)}$
   \item $\fwdSliceR{G} \;:\; \powerset{T(G)} \to \powerset{S(G)}$
\end{itemize}
We use $\bwdSliceR{G}$ to denote the backwards slicing relation. Intuitively, the relation 
$\bwdSliceR{G}$, when given a set of outputs of interest computes the subgraph of the computation that
contains only the nodes necessary for the computation of those outputs. More formally:
\begin{definition}[Backwards Slicing (Graph)]
   For a dependence graph $G = (V,E)$, and a set of output nodes $V_{out} \subseteq V$, we write $V_{out}, G \bwdSliceS G'$ to mean
   that $G'$ is the minimal slice of $G$ with respect to $V_{out}$, defined as
   $G' = (V', E')$ where $V' = \set{v \in V \mid \exists v' \in V_{out}.\;P_G(v', v)}$ and $E' = \set{(v,v') \in E \mid v,v' \in V'}$
\end{definition}
We analogously use $\fwdSliceR{G}$ to denote the forward slicing relation. In contrast to backwards
slicing, we provide the forward slicing relation with a set of input nodes, and compute the set of
internal and output nodes that the input nodes are sufficient to compute. Formally:
\begin{definition}[Forwards Slicing (Graph)]
   For a dependence graph $G = (V,E)$, and a set of input nodes $V_{in} \subseteq V$, we write $V_{in}, G \fwdSliceR{G} G'$ to mean that
   $G'$ is the slice of $G$ with respect to $V_{in}$, defined as $G' = (V', E')$ where we  define $V'$ as:
   $V' = \set{v \in V \mid v, G \bwdSliceR{G} G'' \opAnd T(G'') \subseteq V_{in}}$ and $E' = \set{(v,v')  \in E \mid v,v' \in V'}$.
\end{definition}

\begin{lemma}[De Morgan dual of Forward Slice]
      For a dependence graph $G = (V,E)$ and a given set of inputs, $V_{in} \subseteq T(G)$, we consider the complement
      of $V_{in}\;w.r.t\;T(G)$, then if we backwards slice in $G^{-1}$, and take the complement of the
      result $\bwdSliceR{G^{-1}}(V_{in}^c) w.r.t\; S(G)$, we obtain 
      a forward slice with respect to $V_{in}$:
      $$\fwdSliceR{G}(V_{in}) = (\bwdSliceR{G^{-1}}(V_{in}^c))^{c}$$
\end{lemma}
