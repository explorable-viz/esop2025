\section{Graphical Galois slicing}

\subsection{Preliminaries}

\begin{definition}[Complement]
   For any set $X$ and any $Y \subseteq X$, write $\compl{Y} \subseteq X$ to denote the \textit{complement} of $Y$
   with respect to $X$.
\end{definition}

\subsection{Galois connections for a relation}

The following Galois connection $\consumesF{D} \adjoint \producesF{D}: \powerset{X} \to \powerset{Y}$
generalises the usual image/preimage Galois connection for a function $f: X \to Y$ to a relation $D \subseteq
Y \times X$. (We flip the direction of the relation compared to $f$ so we can read $D$ as \emph{depends on}.)

\begin{definition}
   \item
   \begin{enumerate}
      \item $\consumesF{D}(Y') \eqdef \set{x \in X \mid \exists y \in Y'.\;Dyx}$ \hfill (Consumes)
      \item $\producesF{D}(X') \eqdef \set{y \in Y \mid \forall x \in X.\;Dyx \implies x \in X'}$ \hfill (Produces)
   \end{enumerate}
\end{definition}

\begin{definition}[De Morgan dual]
   Suppose $\mathcal{A}, \mathcal{B}$ are Boolean algebras.
   \begin{enumerate}
      \item For any $f: \mathcal{A} \to \mathcal{B}$, define $\dual{f}: \mathcal{A} \to \mathcal{B}$ to be $\neg_{\mathcal{B}} \after f \after \neg_{\mathcal{A}}$.
      \item For any $f \adjoint g: \mathcal{A} \to \mathcal{B}$, define $\dual{(f \adjoint g)}: \mathcal{B} \to \mathcal{B}$ to be $\dual{g} \adjoint \dual{f}$.
   \end{enumerate}
\end{definition}

\begin{definition} Define:
   \begin{enumerate}
      \item $\consumesDualF{D} := \dual{(\consumesF{D})}$
      \item $\producesDualF{D} := \dual{(\producesF{D})}$
   \end{enumerate}
\end{definition}

\begin{lemma}[Relationship between dual and adjoint]
   $\consumesF{D} \adjoint \producesF{D} = \producesDualF{D^{-1}} \adjoint \consumesDualF{D^{-1}}$
\end{lemma}

\begin{definition}[Related inputs and related outputs]
   Define
   \begin{enumerate}
      \item $\relatedOutputsF{D} \eqdef \consumesF{DD^{-1}} = \consumesF{D}\consumesF{D^{-1}} = \consumesF{D}\producesDualF{D}$ \hfill (Related outputs)
      \item $\relatedInputsF{D} \eqdef \consumesF{D^{-1}D} = \consumesF{D^{-1}}\consumesF{D} = \producesDualF{D}\consumesF{D}$ \hfill (Related inputs)
   \end{enumerate}
\end{definition}

\subsection{Dependence graphs}

We now introduce \emph{dependence graphs} $G = (V, E)$ with vertices $\alpha, \beta \in V$ and edges $(\alpha,
\beta) \in E$ along with some standard graph-theoretic notions. For the following definitions fix $G = (V,
E)$.

\begin{definition}[Opposite graph]
   Define $G^{-1} \eqdef (V, E^{-1})$ where $\param^{-1}$ denotes relational inverse.
\end{definition}

\begin{definition}[Induced subgraph]
   For any $V' \subseteq V$, write $\induced{G}{V'}$ for the subgraph of $G$ induced by $V'$, that is
   $\induced{G}{V'} \eqdef (V', E')$ where $E' = \set{(\alpha,\beta) \in E \mid \alpha, \beta \in V'}$.
\end{definition}

\begin{definition}[Reachable]
   For any pair of vertices $\alpha, \beta \in V$, write $\reachable{G}(\alpha, \beta)$, to mean there exists
   $\seq{\alpha}$ such that $\forall i < \length{\seq{\beta}}.\;(\beta_i,\beta_{i+1}) \in E$ where $\seq{\beta}
   \eqdef \alpha \cons \seq{\alpha} \cons \beta$.
\end{definition}

\begin{definition}[Neighbours]
   Define the \emph{in neighbours} and \emph{out neighbours} of $G$ as the following functions of type $V \to
   \powerset V$:
   \begin{itemize}
      \item $\inN{G}(\alpha) \eqdef \set{\alpha' \in V \mid (\alpha', \alpha) \in E}$
      \item $\outN{G} \eqdef \inN{G^{-1}}$
   \end{itemize}
\end{definition}

We extend the idea of neighbours of a node to \textit{neighbourhoods} of a set of vertices in the graph.

\begin{definition}[Neighbourhoods]
   For any $V' \subseteq V$, define:
   \begin{itemize}
      \item $\inN{G}(V') \eqdef \set{\alpha \in V \mid \exists \beta \in V'.\; \alpha \in \inN{G}(\beta) \opAnd \alpha \notin V'}$
      \item $\outN{G}(V') \eqdef \inN{G^{-1}}(V')$
   \end{itemize}
\end{definition}

\begin{definition}[Sources and Sinks]
   Define the \emph{sources} and \emph{sinks} of $G$ as:
   \begin{itemize}
      \item $\sources{G} \eqdef \set{v \in V \mid \inN{G}(v) = \emptyset}$
      \item $\sinks{G} \eqdef \sources{G^{-1}}$
   \end{itemize}
\end{definition}

\begin{definition}[Subgraph]
   Write $G' \subseteq G$ to mean that $G'$ is a subgraph of $G$.
\end{definition}

\begin{definition}[Subgraph Extension]
   For any $G' = (V',E') \subseteq G$ and any $V^{\dag} \subseteq V$, define $G' \disjunion V^{\dag} \eqdef \induced{G}{V' \cup V^{\dag}}$.
\end{definition}

\subsection{Galois slicing over dependence graphs}

We define two slicing functions over $G$ of the following types:
\begin{itemize}
   \item $\consumesF{G}: \powerset{\sources{G}} \to \powerset{\sinks{G}}$
   \item $\producesF{G}: \powerset{\sinks{G}} \to \powerset{\sources{G}}$
\end{itemize}

For a set of outputs (sources), the backward-slicing function $\consumesF{G}$ computes just the inputs needed
for those outputs. For a set of inputs (sinks), the forward-slicing function $\producesF{G}$ computes the set
of outputs that can be produced using only those inputs.

\begin{definition}[Forward slice]
   For any set of sources $V \subseteq \powerset{\sources{G}}$, define the \emph{backward slice}
   $\consumesF{G}(V)$ to be the set of sinks $\set{\alpha \in \sinks{G} \mid \exists \beta \in
   V.\;\reachable{G}(\beta, \alpha)}$.
\end{definition}

\begin{definition}[Backward slice]
   For any set of sinks $V \subseteq \powerset{\sinks{G}}$, define the \emph{forward slice}
   $\producesF{G}(V)$ of $V$ to be the set of sources $\set{\alpha \in \sources{G} \mid
   \consumesF{G}(\set{\alpha}) \subseteq V}$.
\end{definition}

\begin{lemma}[Adjointness of ${\producesAlgR{G}}$ and $\consumesAlgR{G}$]
   \item Suppose $U \subseteq \sources{G}$ and $V \subseteq \sinks{G}$. Then $U \subseteq {\producesAlgR{G}}(V)
   \iff {\consumesAlgR{G}}(U) \subseteq V$.
\end{lemma}

\begin{definition}[Backward slicing algorithm]
   \figref{graph:slicing:backward} defines the backward slicing relation $\producesAlgR{G}$.
\end{definition}

\begin{definition}[Forward slicing algorithm]
   \figref{graph:slicing:forward} defines the forward slicing relation $\producesAlgR{G}$.
\end{definition}

\begin{theorem}[Correctness of forward slicing algorithm]
For any $V \subseteq \powerset{\sinks{G}}$, we have $\induced{G}{V} \producesAlgR{G} G'$ iff $\sources{G'} = \producesF{G}(V)$.
\end{theorem}
