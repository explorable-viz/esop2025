\section{Graphical Galois slicing}

\subsection{Preliminaries}

\begin{definition}[Complement]
   For any set $X$ and any $Y \subseteq X$, write $\compl{Y} \subseteq X$ to denote the \textit{complement} of $Y$
   with respect to $X$.
\end{definition}

\subsection{Galois connections for a relation}

The following Galois connection $\consumesF{D} \adjoint \producesF{D}: \powerset{X} \to \powerset{Y}$
generalises the usual image/preimage Galois connection for a function $f: X \to Y$ to a relation $D \subseteq
Y \times X$. (We flip the direction of the relation compared to $f$ so we can read $D$ as \emph{depends on}.)

\begin{definition}
   \item
   \begin{enumerate}
      \item $\consumesF{D}(Y') \eqdef \set{x \in X \mid \exists y \in Y'.\;D(y,x)}$ \hfill (consumes)
      \item $\producesF{D}(X') \eqdef \set{y \in Y \mid \forall x \in X.\;D(y,x) \implies x \in X'}$ \hfill (produces)
   \end{enumerate}
\end{definition}

\noindent For a set of outputs, $\consumesF{D}$ collections just the inputs needed for those outputs. For a
set of inputs, $\producesF{G}$ collects just the outputs that can be produced using only those inputs.

\begin{definition}[De Morgan dual]
   Suppose $\mathcal{A}, \mathcal{B}$ are Boolean algebras.
   \begin{enumerate}
      \item For any $f: \mathcal{A} \to \mathcal{B}$, define $\dual{f}: \mathcal{A} \to \mathcal{B}$ to be $\neg_{\mathcal{B}} \after f \after \neg_{\mathcal{A}}$.
      \item For any $f \adjoint g: \mathcal{A} \to \mathcal{B}$, define $\dual{(f \adjoint g)}: \mathcal{B} \to \mathcal{B}$ to be $\dual{g} \adjoint \dual{f}$.
   \end{enumerate}
\end{definition}

\begin{definition} Define:
   \begin{enumerate}
      \item $\consumesDualF{D} := \dual{(\consumesF{D})}$ \hfill (produced by)
      \item $\producesDualF{D} := \dual{(\producesF{D})}$ \hfill (consumed by)
   \end{enumerate}
\end{definition}

\begin{lemma}[Relationship between dual and adjoint]
   $\consumesF{D} \adjoint \producesF{D} = \producesDualF{D^{-1}} \adjoint \consumesDualF{D^{-1}}$
\end{lemma}

\begin{definition}[Related inputs and related outputs]
   Define
   \begin{enumerate}
      \item $\relatedOutputsF{D} \eqdef \consumesF{DD^{-1}} = \consumesF{D}\consumesF{D^{-1}} = \consumesF{D}\producesDualF{D}$ \hfill (co-consumes)
      \item $\relatedInputsF{D} \eqdef \consumesF{D^{-1}D} = \consumesF{D^{-1}}\consumesF{D} = \producesDualF{D}\consumesF{D}$ \hfill (co-produces)
   \end{enumerate}
\end{definition}

\subsection{Dependence graphs}

We now introduce \emph{dependence graphs} $G = (V, E)$ with vertices $\alpha, \beta \in V$ and edges $(\alpha,
\beta) \in E$ along with some standard graph-theoretic notions. For the following definitions fix $G = (V,
E)$.

\begin{definition}[Opposite graph]
   Define $G^{-1} \eqdef (V, E^{-1})$ where $\param^{-1}$ denotes relational inverse.
\end{definition}

\begin{definition}[Neighbours]
   Define the following functions of type $V \to \powerset V$:
   \begin{itemize}
      \item $\inN{G}(\alpha) \eqdef \set{\alpha' \in V \mid (\alpha', \alpha) \in E}$ \hfill (in neighbours)
      \item $\outN{G} \eqdef \inN{G^{-1}}$ \hfill (out neighbours)
   \end{itemize}
\end{definition}

\begin{definition}[Induced subgraph]
   For any $V' \subseteq V$, write $\induced{G}{V'}$ for the subgraph of $G$ induced by $V'$, that is
   $\induced{G}{V'} \eqdef (V', E')$ where $E' = \set{(\alpha,\beta) \in E \mid \alpha, \beta \in V'}$.
\end{definition}

\begin{definition}[Reachable]
   Define the \emph{reachability} relation $\reachable{G} \subseteq V^2$ that has $\reachable{G}(\alpha,
   \beta)$ iff there exists $\seq{\alpha}$ such that $\forall i < \length{\seq{\beta}}.\;(\beta_i,\beta_{i+1})
   \in E$ where $\seq{\beta} \eqdef \alpha \cons \seq{\alpha} \cons \beta$.
\end{definition}

\begin{definition}[Sources and sinks]
   Define:
   \begin{itemize}
      \item $\sources{G} \eqdef \set{v \in V \mid \inN{G}(v) = \emptyset}$ \hfill (sources)
      \item $\sinks{G} \eqdef \sources{G^{-1}}$ \hfill (sinks)
   \end{itemize}
\end{definition}

\begin{definition}[Extension]
   Define the \emph{extension} $\extension{G}$ of $G$ to be $\reachable{G}$ restricted to $\sources{G} \times \sinks{G}$.
\end{definition}

\begin{definition}[Empty graph, subgraph, union]
   \item
   \begin{enumerate}
      \item Write $\emptyset$ to denote the empty graph $(\emptyset, \emptyset)$.
      \item Write $G' \subseteq G$ to mean that $G' = (V', E')$ is a \emph{subgraph} of $G$, i.e.~$V' \subseteq V$ and $E \subseteq E'$.
      \item Write $G \cup G'$ for the (non-disjoint) union ($V \cup V'$, $E \cup E'$) of $G$ and $G' = (V',
      E')$.
   \end{enumerate}
\end{definition}

\begin{definition}[Subgraph Extension]

\subsection{Galois slicing over dependence graphs}

Write $\consumesF{G} \adjoint \producesF{G}$ as shorthand for the Galois connection $\consumesF{\extension{G}}
\adjoint \producesF{\extension{G}}: \powerset{\sources{G}} \to \powerset{\sinks{G}}$. Note that distinct
graphs with the same extension will induce the same Galois connection.

\begin{definition}[Slicing algorithms]
   \item Suppose $\hat{G} = (\hat{V}, \hat{E})$.
   \begin{enumerate}
      \item \figref{graph:slicing:backward} defines the relation $\bwdAlg{\hat{G}}$ \hfill (backward slicing)
      \item \figref{graph:slicing:forward} defines the relation $\fwdAlg{\hat{G}}$ \hfill (forward slicing)
   \end{enumerate}
\end{definition}

\begin{theorem}[Correctness of slicing algorithms]
   \item
   \begin{enumerate}
   \item Suppose $V \subseteq \powerset{\sources{\hat{G}}}$. Then $\consumesF{\hat{G}}(V) = V' \iff V \fwdAlg{\hat{G}} V'$.
   \item Suppose $V \subseteq \powerset{\sinks{\hat{G}}}$. Then $\producesF{\hat{G}}(V) = V' \iff V \fwdAlg{\hat{G}} V'$.
   \end{enumerate}
\end{theorem}
