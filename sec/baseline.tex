\section{Baseline language}
\label{sec:baseline}
\subsection{Surface Language}

When presenting a program execution, we want to give the appearence that the user is executing parts of the
surface language directly. To that end we formalise the evaluation of surface terms and a corresponding notion
of value in the surface language, which corresponds to values in the core language except in the case of
closures, which contain surface expressions (and, recursively, surface values). Later we prove that directly
evaluating a surface program $s$ and desugaring the resulting value produces the same core language value as
desugaring $s$ to a core program and then evaluating, justifying our presentation of the surface semantics in
the debugger.

\subsubsection{Mutual recursion}

\figref{close-defs-surface} defines the relation $\closeDefsS$ which creates the environments associated with
a block of mutually recursively defined functions. Given an environment $\gamma$ and a sequence of recursively
defined functions $g = \seq{\bind{x}{\mu}}$ (read $x$ as the function name and $\mu$ as the clauses of the
function), we split the sequence into each individual function $\bind{x}{\mu}$ and form the closure
associated with the function.

\subsubsection{Piecewise function definitions}

We disallow certain combinations of patterns which are commonly allowed in other languages, such as:%
{\small
\begin{lstlisting}[language=Fluid]
   foo (Cons y ys) (Cons z zs) = ...
   foo  x           Nil        = ...

   bar x ... = ...
   bar y ... = ...
\end{lstlisting}}

\noindent In particular, ``aligned'' variable patterns must have the same name, and variable patterns and any other kind of pattern may only be aligned if we have already been able to distinguish both arguments. Thus the following is allowed:%
{\small
\begin{lstlisting}[language = Fluid]
   baz (Cons y ys) (Cons z zs) = ...
   baz  Nil         x          = ...
\end{lstlisting}}

\subsubsection{Desugaring Of Values}

To relate the surface semantics to the core semantics, we must extend desugaring to values, which in turn requires desugaring for surface environments $\upgamma$ and recursive functions $\seq{g}$ (\figref{desugar:values}).

\subsubsection{Equivalence of Semantics}

\begin{theorem}
   \label{thm:semantics-eq}
   Suppose $s \desugarFwdS e \; and \; \gamma \desugarFwdS \gamma'$, then:
\hfill
\begin{enumerate}
   \item $ If \; \gamma, s \evalSugS v \; then \; \gamma', e \evalFwdS v' \; where \; v \desugarFwdS v'$ \label{semantics-eq-1}
   \item $If \; \gamma',e \evalFwdS v' \; then \; \gamma, s \evalSugS v \; where \; v \desugarFwdS v'$   \label{semantics-eq-2}
\end{enumerate}
\end{theorem}
\begin{proof}
   For (1) see \appref{semantics-eq-1}. For (2) see \appref{semantics-eq-2}.
\end{proof}

\thmref{semantics-eq} relies on analogous lemmas for pattern-matching (\lemref{matching-bi}) and recursive
definitions (\lemref{closedefs-eq}), and also that environment lookup commutes with desugaring
(\lemref{lemma:varlookup}). Desugaring is not total, so we require that both $s$ and $\gamma$ can be desugared
as the precondition to \thmref{semantics-eq}.

\begin{lemma}
   \label{lem:lemma:varlookup}
   Suppose $\gamma \desugarFwdS \gamma'$ and $v \desugarFwdS v'$. Then $\bind{x}{v} \in \gamma$ iff $\bind{x}{v'} \in \gamma'$.
\end{lemma}
\begin{proof}
   See \appref{proofs:varlookup}
\end{proof}

\begin{lemma}
   \label{lem:matching-bi}
   Suppose $v \desugarFwdS v'$ and $\mu \desugarFwdS \sigma$.
   \hfill
   \begin{enumerate}
      \item If $u, \mu\matchFwdS \gamma, s$ then $\exists \gamma', e.v,\sigma \matchFwdS \gamma', e$ with $\gamma \desugarFwdS \gamma'$ and $s \desugarFwdS e$
      \item If $v, \sigma \matchFwdS \gamma', e$ then $u, \mu \matchFwdS \gamma, s$ and $\gamma \desugarFwdS \gamma'$ and $s \desugarFwdS e$
   \end{enumerate}

\end{lemma}
\begin{proof}
   For (1) see \appref{proofs:matching-bi-1}. For (2) see \appref{proofs:matching-bi-2}.
\end{proof}

\begin{lemma}
   \label{lem:closedefs-eq}
   Suppose $\gamma_1 \desugarFwdS \gamma_2$ and $g \desugarFwdS \rho$.
   \hfill
   \begin{enumerate}
      \item If $\gamma_1, g \closeDefsS \gamma_1'$ then $\gamma_2, \rho \closeDefsS \gamma_2'$ where $\gamma_1' \desugarFwdS \gamma_2'$
      \item If $\gamma_2, \rho \closeDefsS \gamma_2'$, then $\gamma_1, g, \closeDefsS \gamma_1'$ where $\gamma_1' \desugarFwdS \gamma_2'$
   \end{enumerate}
\end{lemma}
\begin{proof}
   See \appref{proofs:close-defs-eq}
\end{proof}

\begin{lemma}
   \label{lem:match-domains-agree}
   Suppose $u \desugar v$ and $\sigma = \mkElimF(p, \kappa)$.
   \begin{enumerate}
      \item If $u, p \match \gamma$ then $\exists \gamma'.v, \sigma \match \gamma', \kappa$ with $\gamma \desugar \gamma'$.
      \item If $u, p\;\matchNeg$ then $v, \sigma\;\matchNeg$.
   \end{enumerate}
\end{lemma}

\begin{proof}
   See \appref{proofs:match-domains-agree}.
\end{proof}

\begin{lemma}
   Suppose $\seq{u} \desugar \seq{v}$ and $\sigma = \elimArgsF(\seq{p}, \kappa)$.
   \begin{enumerate}
      \item If $\seq{u}, \seq{p} \match \gamma$ then $\exists \gamma'.\seq{v}, \sigma \match \gamma', \kappa$ with $\gamma \desugar \gamma'$.
      \item If $\seq{u}, \seq{p}\;\matchNeg$ then $\seq{v}, \sigma\;\matchNeg$.
   \end{enumerate}
\end{lemma}

\begin{lemma}
   \label{lem:match-increasing}
   Suppose $v, \sigma \match \gamma, \kappa_1$. Then $\exists \kappa.v, \sigma \disjjoin \tau \match \gamma, \kappa$ and:
   \begin{enumerate}
      \item If $v, \tau \match \gamma, \kappa_2$ then $\kappa = \kappa_1 \disjjoin \kappa_2$.
      \item If $v, \tau\;\matchNeg$ then $\kappa = \kappa_1$.
   \end{enumerate}
\end{lemma}

$\disjjoin$ is a partial commutative monoid. Write $x \eqif y$ to mean $x$ is defined iff $y$ is defined, in
which case $x = y$.

\begin{lemma}
   \label{lem:disjjoin-assoc-comm}
   \item
   \begin{enumerate}
   \item $\sigma_1 \disjjoin (\sigma_2 \disjjoin \sigma_3) \eqif (\sigma_1 \disjjoin \sigma_2) \disjjoin \sigma_3$
   \item $\sigma \disjjoin \tau \eqif \tau \disjjoin \sigma$
   \end{enumerate}
\end{lemma}
