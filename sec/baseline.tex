\section{Baseline language}
\label{sec:baseline}
\subsection{Surface Language}

To allow the user to execute surface programs directly, we formalise the evaluation of surface terms and
corresponding notion of surface value. Surface values differ from core values only in the case of closures,
which contain surface expressions other surface values. Later we prove that directly evaluating a surface
program $s$ and desugaring the resulting value produces the same core language value as desugaring $s$ to a
core program and then evaluating, justifying our presentation of the surface semantics in the debugger.

\subsubsection{Mutual recursion}

\figref{close-defs-surface} defines the relation $\closeDefsS$, called \emph{close defs}, which creates the
closures associated with a block of mutually recursive functions $g = \seq{\bind{x}{\mu}}$. (Read $x$ as the
function name and $\mu$ as the defining clauses of the function.)

\subsubsection{Piecewise function definitions}

We disallow certain combinations of patterns which are commonly allowed in other languages, such as:%
{\small
\begin{lstlisting}[language=Fluid]
   foo (Cons y ys) (Cons z zs) = ...
   foo  x           Nil        = ...

   bar x ... = ...
   bar y ... = ...
\end{lstlisting}}

\noindent In particular, ``aligned'' variable patterns must have the same name, and variable patterns and any other kind of pattern may only be aligned if we have already been able to distinguish both arguments. Thus the following is allowed:%
{\small
\begin{lstlisting}[language = Fluid]
   baz (Cons y ys) (Cons z zs) = ...
   baz  Nil         x          = ...
\end{lstlisting}}

\subsubsection{Continuations and Eliminators}

A \emph{continuation} $\kappa$ is either a term or an \textit{eliminator}. Eliminators are trie-like
structures into which surface-language pattern-matching constructs are desugared. An eliminator $\sigma$ is
said to be \textit{partial} if it is not defined for every element of its domain.

\subsubsection{Equivalence of Semantics}

\thmref{semantics-eq} is non-trivial only for desugarable $s$ and $\gamma$; for now we conjecture that
well-typed terms desugar.

\begin{theorem}
   \label{thm:semantics-eq}
   Suppose $s \desugar e \; and \; \gamma \desugar \gamma'$, then:
\hfill
\begin{enumerate}
   \item If $\gamma, s \evalSugS v$ then $\exists v'.\; \gamma', e \evalFwdS v'$ with $v \desugar v'$ \label{semantics-eq-1}
   \item If $\gamma',e \evalFwdS v'$ then $\exists v.\; \gamma, s \evalSugS v$ with $v \desugar v'$  \label{semantics-eq-2}
\end{enumerate}
\end{theorem}
\begin{proof}
   For (1) see \appref{semantics-eq-1}. For (2) see \appref{semantics-eq-2}.
\end{proof}

\thmref{semantics-eq} relies on analogous lemmas for pattern-matching (\lemref{match-eq}), recursive
definitions (\lemref{closedefs-eq}), and environment extension (\lemref{varlookup}).

\begin{lemma}
   \label{lem:varlookup}
   $\gamma \desugar \gamma'$ and $v \desugar v' \;\iff\; \gamma \cons (\bind{x}{v}) \desugar \gamma' \cons (\bind{x}{v'})$.
\end{lemma}
\begin{proof}
   Trivially follows from definition of environment desugaring.
\end{proof}
\begin{lemma}
   \label{lem:match-eq}
   Suppose $v \desugar v'$ and $\mu \desugar \sigma$.
   \hfill
   \begin{enumerate}
      \item If $u, \mu\match \gamma, s$ then $\exists \gamma', e.v,\sigma \match \gamma', e$ with $\gamma \desugar \gamma'$ and $s \desugar e$.
      \item If $v, \sigma \match \gamma, e$ then $\exists \gamma', s.u, \mu \match \gamma', s$ and $\gamma' \desugar \gamma$ and $s \desugar e$.
   \end{enumerate}

\end{lemma}
\begin{proof}
   For (1) see \appref{proofs:match-eq-1}. For (2) see \appref{proofs:match-eq-2}.
\end{proof}

\begin{lemma}
   \label{lem:closedefs-eq}
   Suppose $\gamma_1 \desugar \gamma_2$ and $g \desugar \rho$.
   \hfill
   \begin{enumerate}
      \item If $\gamma_1, g \closeDefsS \gamma_1'$ then $\exists \gamma_2'.\gamma_2, \rho \closeDefsS \gamma_2'$ with $\gamma_1' \desugar \gamma_2'$.
      \item If $\gamma_2, \rho \closeDefsS \gamma_2'$, then $\exists \gamma_1'.\gamma_1, g, \closeDefsS \gamma_1'$ with $\gamma_1' \desugar \gamma_2'$.
   \end{enumerate}
\end{lemma}
\begin{proof}
   See \appref{proofs:close-defs-eq}
\end{proof}

\begin{lemma}
   \label{lem:match-domains-agree}
   Suppose $u \desugar v$ and $\sigma = \elimF(p, \kappa)$.
   \begin{enumerate}
      \item If $u, p \match \gamma$ and $v, \sigma \match \gamma'$ then $\gamma \desugar \gamma'$.
      \item $u, p\;\matchNeg$ iff $v, \sigma\;\matchNeg$.
   \end{enumerate}
\end{lemma}

\begin{proof}
   See \appref{proofs:match-domains-agree}.
\end{proof}

\begin{lemma}
   \label{lem:match-domains-agree-list-rest}
   Suppose $u \desugar v$ and $\sigma = \elimF(o, \kappa)$.
   \begin{enumerate}
      \item If $u, o \match \gamma$ then $\exists \gamma'.v, \sigma \match \gamma', \kappa$ with $\gamma \desugar \gamma'$.
      \item If $u, o\;\matchNeg$ then $v, \sigma\;\matchNeg$.
   \end{enumerate}
\end{lemma}

\begin{proof}
   See \appref{proofs:match-domains-agree-list-rest}.
\end{proof}

\begin{lemma}
   \label{lem:match-domains-agree-seq}
   Suppose $\seq{u} \desugar \seq{v}$ and $\kappa' = \elimArgsF(\seq{p}, \kappa)$.
   \begin{enumerate}
      \item If $\seq{u}, \seq{p} \match \gamma$ then $\exists \gamma'.\seq{v}, \kappa' \match \gamma', \kappa$ with $\gamma \desugar \gamma'$.
      \item If $\seq{u}, \seq{p}\;\matchNeg$ then $\seq{v}, \kappa'\;\matchNeg$.
   \end{enumerate}
\end{lemma}

\begin{lemma}
   \label{lem:match-increasing}
   Suppose $v, \sigma \match \gamma, \kappa_1$. Then $\exists \kappa.v, \sigma \disjjoin \tau \match \gamma, \kappa$ and:
   \begin{enumerate}
      \item If $v, \tau \match \gamma, \kappa_2$ then $\kappa = \kappa_1 \disjjoin \kappa_2$.
      \item If $v, \tau\;\matchNeg$ then $\kappa = \kappa_1$.
   \end{enumerate}
\end{lemma}

\begin{lemma}
   \label{lem:match-increasing-two}
   Suppose $v, \sigma \match \gamma, \kappa$ where $s = \seqRangeOp{\sigma_1}{\sigma_j}{\disjjoin}$ and $\forall i \numleq j.\sigma_i = \elimF(p_i, \kappa_i)$.
   \begin{enumerate}
      \item $\forall i \numleq j.v, \sigma_i\;\matchNeg$ or $v, \sigma_i \match \gamma, \kappa_i$.
      \item $\exists i \numleq j.v, \sigma_i \match \gamma, \kappa_i$.
      \item $\exists p.(\forall i \numleq j.v, \sigma_{i} \match \gamma, \kappa_{i} \iff p_{i} = p)$.
      \item $\kappa = \disjjoin\set{\kappa_i \mid i \numleq j\text{ and }v, \sigma_i \match \gamma, \kappa_i}$
   \end{enumerate}
\end{lemma}

$\disjjoin$ is a partial commutative monoid.

\begin{lemma}
   \label{lem:disjjoin-assoc-comm}
   \item
   \begin{enumerate}
   \item $\sigma_1 \disjjoin (\sigma_2 \disjjoin \sigma_3) \eqif (\sigma_1 \disjjoin \sigma_2) \disjjoin \sigma_3$
   \item $\sigma \disjjoin \tau \eqif \tau \disjjoin \sigma$
   \end{enumerate}
\end{lemma}
