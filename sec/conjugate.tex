\section{Conjugate Operators For Program Slicing}
\label{sec:conjugate}

\subsection{Preliminaries}

\begin{definition}[Complement]
   For any set $X$ and any $Y \subseteq X$, write $\compl{Y} \subseteq X$ to denote the \textit{complement}
   of $Y$ with respect to $X$.
\end{definition}

\begin{definition}[Image and Preimage]
   For a relation $D \subseteq X \times Y$, and a subset $X' \subset X$, we write
   $D[X']$ to denote the image of $X'$ under $D$. For a subset $Y' \subseteq Y$
   we write $\preim{D}[Y']$ to denote the preimage of $Y'$ under $D$.
\end{definition}

\begin{definition}[Conjugate Functions]
   For Boolean algebras, $\mathcal{A}$, and $\mathcal{B}$, functions $f: \mathcal{A} \to \mathcal{B}$
   and $g: \mathcal{B} \to \mathcal{A}$, and elements $a \in \mathcal{A}$ and $b \in \mathcal{B}$,
   we say that $f$ and $g$ are \textit{conjugate} if the following holds:
   $$
      f(a) \wedge b = \bot \iff a \wedge g(b) = \bot 
   $$
\end{definition}

\begin{lemma}[Preimage and Image of a relation are conjugate]
   For a relation $D \subseteq X \times Y$, and subsets $X' \subseteq, Y' \subseteq Y$,
   $R[\param]$ and $\preim{R}[\param]$ are conjugate.
\end{lemma}
\begin{proof}
   Trivial.
\end{proof}
The notion of conjugate functions is due to \citet{jonssontarski51}. The authors
of that paper provide the following alternate definition of conjugacy:

\begin{definition}[Computation of Conjugate Function]\label{def:construct-conjugate}
   Given a function $f: \mathcal{A} \to \mathcal {B}$, there can exist only one
   conjugate function $g: \mathcal{B} \to \mathcal{A}$. If $g$ exists we can
   calculate it using the following:
   $$
      g(b) = \bigwedge_{f(a) \wedge b = \bot} \compl{a} \quad \text{for any $b \in \mathcal{B}$}
   $$
\end{definition}

In general, \defref{construct-conjugate} is inefficient, since it requires 
ranging over all possible values $a \in \mathcal{A}$, and computing their
meet with $b$.

It is not hard to see that the image and preimage functions of a relation
$D$ are conjugate and furthermore that the image function of a relation $D$ is
conjugate to the image function of the converse relation written $D^{-1}$.

We have already discussed how we realize the dependency relation described in
\secref{core}. For the rest of the section, we use sets of nodes.

\begin{definition}[Reachability]
   For a graph $G=(V,E)$, the \textit{reachability relation} $R_G$ is the transitive
   closure of $E$
\end{definition}

\begin{definition}[Demands]
   For a dependence graph $G = (V,E)$, a set of inputs $X' \subseteq V$ and a subset of outputs,
   $Y \subseteq Y' \subseteq V$, we write $\demandR(Y) = \bigwedge_{R_G[X] \wedge Y \neq \bot} \! X \quad (X \subseteq X')$ to mean the set of inputs that $Y$ demands.
\end{definition}

\begin{definition}{Demanded By}
   For a dependence graph $G = (V,E)$, a set of outputs $Y' \subseteq V$ and a subset of inputs
   $X  \subseteq X \subseteq V$, write $\demandByR(X) = \bigwedge_{R_G[Y] \wedge X \neq \bot} \! Y \quad (Y \subseteq Y')$ to mean the set of outputs that $X$ is demanded by.
\end{definition}
