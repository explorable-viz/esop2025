\begin{lemma}
   \label{lem:match-domains-agree}
   Suppose $u \desugar v$ and $\upsigma = \mkElimF(p, \upkappa)$.
   \begin{enumerate}
      \item If $u, p \desugar \upgamma$ then $\exists \upgamma'.v, \upsigma \match \upgamma', \upkappa$ with $\upgamma \desugar \upgamma'$.
      \item If $u, p\;\desugarNeg$ then $v, \upsigma\;\desugarNeg$.
   \end{enumerate}
\end{lemma}
